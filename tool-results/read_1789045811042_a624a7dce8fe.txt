   128→// ============================================================ hydrograph canvas
   129→export function Hydrograph({
   130→  data, peak, curT,
   131→}: {
   132→  data: { t: number; q: number }[];
   133→  peak: number;
   134→  curT: number;
   135→}) {
   136→  const ref = useRef<HTMLCanvasElement>(null);
   137→  useEffect(() => {
   138→    const c = ref.current;
   139→    if (!c) return;
   140→    const dpr = window.devicePixelRatio || 1;
   141→    const W = c.clientWidth || 262;
   142→    const H = c.clientHeight || 96;
   143→    c.width = W * dpr;
   144→    c.height = H * dpr;
   145→    const ctx = c.getContext('2d');
   146→    if (!ctx) return;
   147→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   148→    ctx.clearRect(0, 0, W, H);
   149→    const padL = 2, padR = 2, padT = 6, padB = 13;
   150→    const maxQ = Math.max(peak, 300);
   151→    const t1 = Math.max(data.length ? data[data.length - 1].t : 20, 20);
   152→    ctx.strokeStyle = 'rgba(255,255,255,0.07)';
   153→    for (let i = 1; i < 4; i++) {
   154→      const y = padT + ((H - padT - padB) * i) / 4;
   155→      ctx.beginPath(); ctx.moveTo(padL, y); ctx.lineTo(W - padR, y); ctx.stroke();
   156→    }
   157→    if (data.length > 1) {
   158→      const X = (t: number) => padL + (t / t1) * (W - padL - padR);
   159→      const Y = (q: number) => H - padB - (Math.min(q, maxQ) / maxQ) * (H - padT - padB);
   160→      ctx.beginPath();
   161→      data.forEach((p, i) => (i === 0 ? ctx.moveTo(X(p.t), Y(p.q)) : ctx.lineTo(X(p.t), Y(p.q))));
   162→      ctx.lineTo(X(data[data.length - 1].t), H - padB);
   163→      ctx.lineTo(X(data[0].t), H - padB);
   164→      ctx.closePath();
   165→      ctx.fillStyle = 'rgba(34,211,238,0.12)';
   166→      ctx.fill();
   167→      ctx.beginPath();
   168→      data.forEach((p, i) => (i === 0 ? ctx.moveTo(X(p.t), Y(p.q)) : ctx.lineTo(X(p.t), Y(p.q))));
   169→      ctx.strokeStyle = '#22d3ee';
   170→      ctx.lineWidth = 1.5;
   171→      ctx.stroke();
   172→      const pk = data.reduce((a, b) => (b.q > a.q ? b : a), data[0]);
   173→      if (pk.q > 50) {
   174→        ctx.fillStyle = '#fbbf24';
   175→        ctx.beginPath(); ctx.arc(X(pk.t), Y(pk.q), 2.4, 0, Math.PI * 2); ctx.fill();
   176→      }
   177→    }
   178→    // current time cursor
   179→    if (curT > 0 && curT <= t1) {
   180→      const x = padL + (curT / t1) * (W - padL - padR);
   181→      ctx.strokeStyle = 'rgba(255,255,255,0.45)';
   182→      ctx.setLineDash([3, 3]);
   183→      ctx.beginPath(); ctx.moveTo(x, padT); ctx.lineTo(x, H - padB); ctx.stroke();
   184→      ctx.setLineDash([]);
   185→    }
   186→    ctx.fillStyle = 'rgba(255,255,255,0.35)';
   187→    ctx.font = '9px ui-monospace, monospace';
   188→    ctx.fillText('0', padL, H - 2);
   189→    const tl = `${Math.round(t1)} s`;
   190→    ctx.fillText(tl, W - padR - ctx.measureText(tl).width, H - 2);
   191→    ctx.fillText(`${Math.round(maxQ)} m³/s`, padL + 2, padT + 7);
   192→  }, [data, peak, curT]);
   193→  return <canvas ref={ref} className="h-[96px] w-full" />;
   194→}

   277→function AlertLine({ kind, text }: { kind: 'ok' | 'warn' | 'crit'; text: string }) {
   278→  const map = {
   279→    ok: { icon: <CheckCircle2 className="h-3.5 w-3.5 text-emerald-400" />, cls: 'text-slate-300' },
   280→    warn: { icon: <AlertTriangle className="h-3.5 w-3.5 text-amber-400" />, cls: 'text-amber-200' },
   281→    crit: { icon: <AlertTriangle className="h-3.5 w-3.5 animate-pulse text-red-400" />, cls: 'text-red-200' },
   282→  };
   283→  return (
   284→    <div className={`flex items-start gap-1.5 text-[10.5px] leading-snug ${map[kind].cls}`}>
   285→      {map[kind].icon}
   286→      <span>{text}</span>

   408→export function ScenarioPanel({
   409→  form, onChange, onRun, onReset, running, stats, damId,
   410→}: {
   411→  form: ScenarioForm;
   412→  onChange: (p: Partial<ScenarioForm>) => void;
   413→  onRun: () => void;
   414→  onReset: () => void;
   415→  running: boolean;
   416→  stats: UIStats | null;
   417→  damId: DamId;
   418→}) {
   419→  const stages = ['Preparing terrain', 'Loading reservoir', 'Building breach', 'Running hydraulic model', 'Processing flood extent', 'Calculating impacts'];
   420→  const stageDone = (i: number): 'done' | 'active' | 'todo' => {
   421→    const s = stats?.stage ?? 0;
   422→    if (!running) return 'todo';
   423→    if (i + 1 < s) return 'done';
   424→    if (i + 1 === s) return 'active';
   425→    return 'todo';
   426→  };
   427→  return (
   428→    <div className="flex flex-col gap-2">
   429→      <Panel title="Failure scenario (assumed)" right={<Chip kind="simulated">WHAT-IF</Chip>}>
   430→        <SectionTitle>Failure mechanism</SectionTitle>
   431→        <RadioGroup value={form.mechanism} onValueChange={(v) => onChange({ mechanism: v as FailureMechanism })} className="gap-2">
   432→          {(['overtopping', 'piping', 'structural'] as FailureMechanism[]).map((m) => (
   433→            <label key={m} className="flex cursor-pointer items-start gap-2 text-[11px] text-slate-300">
   434→              <RadioGroupItem value={m} className="mt-0.5 h-3.5 w-3.5" />
