   159→function disposeSensorObject(root: THREE.Object3D): void {
   160→  root.traverse((o) => {
   161→    const mesh = o as THREE.Mesh;
   162→    if (mesh.geometry) mesh.geometry.dispose();
   163→    const mat = mesh.material as THREE.Material | THREE.Material[] | undefined;
   164→    if (Array.isArray(mat)) mat.forEach((x) => x.dispose());
   165→    else if (mat) mat.dispose();
   166→  });
   167→}

   413→  // ================================================================ terrain mesh
   414→  private buildTerrainMesh(): void {
   415→    // bed grid on cell centers
   416→    this.bedGrid = new Float32Array(NX * NZ);
   417→    for (let j = 0; j < NZ; j++)
   418→      for (let i = 0; i < NX; i++)
   419→        this.bedGrid[j * NX + i] = bedAt((i + 0.5) * DX, (j + 0.5) * DZ - LZ / 2);
   420→
   421→    const geo = new THREE.PlaneGeometry(LX, LZ, NX - 1, NZ - 1);
   422→    geo.rotateX(-Math.PI / 2);
   423→    geo.translate(LX / 2, 0, 0);
   424→    const pos = geo.attributes.position as THREE.BufferAttribute;
   425→    const colors = new Float32Array(pos.count * 3);
   426→    const col = new THREE.Color();
   427→    const tc = { r: 0, g: 0, b: 0 };
   428→    for (let i = 0; i < pos.count; i++) {
   429→      const x = pos.getX(i);
   430→      const z = pos.getZ(i);
   431→      const y = this.sampleBed(x, z);
   432→      pos.setY(i, y);
   433→      const e = 1.2;
   434→      const sx = (this.sampleBed(x + e, z) - this.sampleBed(x - e, z)) / (2 * e);
   435→      const sz = (this.sampleBed(x, z + e) - this.sampleBed(x, z - e)) / (2 * e);
   436→      const slope = Math.sqrt(sx * sx + sz * sz);
   437→      terrainColor(x, z, y, slope, tc);
   438→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   439→      colors[i * 3] = col.r;
   440→      colors[i * 3 + 1] = col.g;
   441→      colors[i * 3 + 2] = col.b;
   442→    }
   443→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   444→    geo.computeVertexNormals();
   445→    const mesh = new THREE.Mesh(
   446→      geo,
   447→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.96, metalness: 0, envMapIntensity: 0.22 }),
   448→    );
   449→    mesh.receiveShadow = true;
   450→    this.scene.add(mesh);
   451→
   452→    // far terrain: the valley runs on beyond the solver domain, then a mountain
   453→    // ring closes the horizon (replaces the old flat surround plane)
   454→    this.scene.add(buildFarTerrain().group);
   455→  }
   456→
   457→  private sampleBed(x: number, z: number): number {
   458→    const fu = clamp((x / DX) - 0.5, 0, NX - 1.001);
   459→    const fv = clamp((z + LZ / 2) / DZ - 0.5, 0, NZ - 1.001);
   460→    const i = Math.floor(fu), j = Math.floor(fv);
   461→    const fx = fu - i, fy = fv - j;
   462→    const i1 = Math.min(i + 1, NX - 1), j1 = Math.min(j + 1, NZ - 1);
   463→    const g = this.bedGrid;
   464→    const a = g[j * NX + i] + (g[j * NX + i1] - g[j * NX + i]) * fx;
   465→    const b = g[j1 * NX + i] + (g[j1 * NX + i1] - g[j1 * NX + i]) * fx;
   466→    return a + (b - a) * fy;
   467→  }

   726→  // per-frame storm driver: rain streaks, darkened sky/fog, lightning flashes
   727→  private updateWeather(dtReal: number): void {
   728→    const target = this.rainInflow > 0 ? (this.rainHeavy ? 1 : 0.55) : 0;
   729→    this.rainVis += (target - this.rainVis) * Math.min(1, dtReal * 1.4);
   730→    const rv = this.rainVis;
   731→
   732→    if (this.rainMesh) {
   733→      this.rainMesh.visible = rv > 0.02;
   734→      if (rv > 0.02) {
   735→        const p = this.rainPos;
   736→        const n = p.length / 6;
   737→        const fall = 46 * dtReal;
   738→        const drift = 8 * dtReal;
   739→        for (let i = 0; i < n; i++) {
   740→          const k = i * 6;
   741→          p[k] += drift; p[k + 1] -= fall; p[k + 2] += drift * 0.35;
   742→          p[k + 3] = p[k] + 0.26; p[k + 4] = p[k + 1] + 1.5; p[k + 5] = p[k + 2] + 0.09;
   743→          if (p[k + 1] < 2.5) {
   744→            const x = -30 + Math.random() * (LX + 110);
   745→            const y = 48 + Math.random() * 34;
   746→            const z = -LZ / 2 - 40 + Math.random() * (LZ + 80);
   747→            p[k] = x; p[k + 1] = y; p[k + 2] = z;
   748→            p[k + 3] = x + 0.26; p[k + 4] = y + 1.5; p[k + 5] = z + 0.09;
   749→          }
   750→        }
   751→        (this.rainMesh.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true;
   752→        (this.rainMesh.material as THREE.LineBasicMaterial).opacity = 0.38 * rv;
   753→      }
   754→    }
   755→
   756→    // occasional lightning during heavy rain
   757→    this.flash *= Math.exp(-dtReal * 6.5);
   758→    if (rv > 0.6) {
   759→      this.boltTimer -= dtReal;
   760→      if (this.boltTimer <= 0) {
   761→        this.boltTimer = 5 + Math.random() * 10;
   762→        this.flash = 0.7 + Math.random() * 0.5;
   763→      }
   764→    }
   765→
   766→    // dim / storm the atmosphere
   767→    this.sunLight.intensity = 2.35 * (1 - 0.7 * rv) + this.flash * 5.5;
   768→    this.sunLight.color.setHex(this.flash > 0.25 ? 0xdfe8ff : 0xffd9a8);
   769→    this.hemiLight.intensity = 0.32 * (1 - 0.4 * rv) + this.flash * 1.2;
   770→    this.scene.environmentIntensity = 0.22 * (1 - 0.5 * rv);
   771→    const fog = this.scene.fog as THREE.Fog;
   772→    fog.color.copy(this.fogDay).lerp(this.fogStorm, rv * 0.85);
   773→    fog.near = 470 - 220 * rv;
   774→    fog.far = 2300 - 1150 * rv;
   775→    this.renderer.toneMappingExposure = 0.85 - 0.13 * rv + this.flash * 0.12;
   776→    const su = this.skyU;
   777→    if (su) {
   778→      su.turbidity.value = 4.6 + 5.4 * rv;
   779→      su.rayleigh.value = Math.max(2.2 - 1.6 * rv, 0.3);
   780→      su.mieCoefficient.value = 0.006 + 0.02 * rv;
   781→    }
   782→
   783→    // water shader: choppier ripples + storm fog match
   784→    const wu = this.waterMat.uniforms;
   785→    wu.uRain.value = rv;
   786→    (wu.uFogColor.value as THREE.Color).copy(fog.color);
   787→    wu.uFogNear.value = fog.near;
   788→    wu.uFogFar.value = fog.far;
   789→  }

   798→  private simSubstep(dt: number): void {
   799→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   800→    eu.uState.value = this.curState.texture;
   801→    eu.uDt.value = dt;
   802→    eu.uG.value = this.gravity;
   803→    eu.uSrcRate.value = this.srcRate;
   804→    this.renderTo(this.etaPass, this.rtEta);
   805→
   806→    const vu = (this.velPass.material as THREE.ShaderMaterial).uniforms;
   807→    vu.uState.value = this.curState.texture;
   808→    vu.uEtaNew.value = this.rtEta.texture;
   809→    vu.uDt.value = dt;
   810→    vu.uG.value = this.gravity;
   811→    this.renderTo(this.velPass, this.nextState);
   812→    this.swapState();
   813→
   814→    if (this.foamOn) {
   815→      const fu = (this.foamPass.material as THREE.ShaderMaterial).uniforms;
   816→      fu.uState.value = this.curState.texture;
   817→      fu.uFoam.value = this.curFoam.texture;
   818→      fu.uDt.value = dt;
   819→      fu.uG.value = this.gravity;
   820→      this.renderTo(this.foamPass, this.nextFoam);
   821→      const t = this.curFoam;
   822→      this.curFoam = this.nextFoam;
   823→      this.nextFoam = t;
   824→    }
   825→
   826→    if (this.sprayOn) {
   827→      const su = (this.sprayPass.material as THREE.ShaderMaterial).uniforms;
   828→      su.uSpray.value = this.curSpray.texture;
   829→      su.uState.value = this.curState.texture;
   830→      su.uDt.value = dt;
   831→      su.uTime.value = this.simTime;
   832→      this.renderTo(this.sprayPass, this.nextSpray);
   833→      const t = this.curSpray;
   834→      this.curSpray = this.nextSpray;
   835→      this.nextSpray = t;
   836→    }
   837→  }

   884→  // wire up a failing block span: invert elevations + staggered failure delays
   885→  private initBreach(span: BreachSpan): void {
   886→    this.breachSpan = span;
   887→    this.breachInverts = [];
   888→    for (let b = span.start; b < span.start + span.count; b++) {
   889→      const z0 = BLOCK_Z0 + b * BLOCK_W;
   890→      this.breachInverts.push(this.blockInvert(z0 + BLOCK_W / 2));
   891→    }
   892→    this.breachDelays = [];
   893→    for (let k = 0; k < span.count; k++) {
   894→      this.breachDelays.push(Math.random() * this.breachTau * 0.3);
   895→    }
   896→  }

   898→  runScenario(p: ScenarioParams): void {
   899→    if (this.scenario) return;
   900→    this.scenario = p;
   901→    this.phase = 'scenario';
   902→    this.scenT = 0;
   903→    this.snapClock = 0;
   904→    this.overtopT = 0;
   905→    this.rainInflow = p.rain === 'none' ? 0 : p.rain === 'moderate' ? 26 : 64;
   906→    this.rainHeavy = p.rain === 'heavy';
   907→    this.breachSpan = null;
   908→    this.breachT = null;
   909→    this.breachDepth01 = 0;
   910→    this.breachFx = false;
   911→    this.structDirty = true;
   912→
   913→    // reservoir set to the scenario level (drive takes it there quickly)
   914→    const target = clamp(fracToSimLevel(p.levelFrac), 0, 24.4);
   915→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   916→    eu.uDriveEta.value = target;
   917→    eu.uDriveOn.value = 1;
   918→
   919→    if (p.mechanism === 'overtopping') {
   920→      // surge inflow pushes the lake above the crest; erosion follows
   921→      eu.uDriveEta.value = Math.max(target, CREST + 1.0);
   922→    } else {
   923→      // schedule breach after the reservoir reaches level (~2-6 demo s)
   924→      this.breachT = -clamp((target - this.lastLevel) / 1.4, 1.5, 5);
   925→      this.breachTau = Math.max(p.formationMin / this.damProfile.timeMinPerSec, 4) / (p.mechanism === 'piping' ? 0.65 : 1) / this.breachRate;
   926→      this.breachPre = p.mechanism === 'piping' ? 0.22 : 0;
   927→      this.initBreach(this.breachSpanFor(p.breachWidthM, p.location));
   928→    }
   929→    this.audio.burst(0.2);
   930→  }

   932→  // Starts the breach erosion clock and fires the failure effects (camera
   933→  // shake, audio burst, concrete debris). Runs when the scenario decides the
   934→  // dam begins to fail: overtopping reaches it after the 7 s surcharge, and
   935→  // structural / piping reach it through the armed negative breachT crossing
   936→  // zero in stepScenario.
   937→  private triggerBreachErosion(): void {
   938→    if (this.breachT !== null && this.breachT >= 0) return; // already eroding
