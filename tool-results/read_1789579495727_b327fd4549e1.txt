     7→// ---------------------------------------------------------------- CPU flow field
     8→// Downsampled readback of the GPU state, used for prop physics + stats.
     9→export class FlowField {
    10→  readonly W = 48;
    11→  readonly H = 28;
    12→  readonly data = new Float32Array(48 * 28 * 4); // (eta, u, v, h)
    13→
    14→  sample(x: number, z: number, out: { eta: number; u: number; v: number }): void {
    15→    const fu = Math.min(Math.max((x / LX) * this.W - 0.5, 0), this.W - 1.001);
    16→    const fv = Math.min(Math.max(((z + LZ / 2) / LZ) * this.H - 0.5, 0), this.H - 1.001);
    17→    const i = Math.floor(fu), j = Math.floor(fv);
    18→    const fx = fu - i, fy = fv - j;
    19→    const i1 = Math.min(i + 1, this.W - 1), j1 = Math.min(j + 1, this.H - 1);
    20→    const k00 = (j * this.W + i) * 4, k10 = (j * this.W + i1) * 4;
    21→    const k01 = (j1 * this.W + i) * 4, k11 = (j1 * this.W + i1) * 4;
    22→    const d = this.data;
    23→    const e0 = d[k00] + (d[k10] - d[k00]) * fx;
    24→    const e1 = d[k01] + (d[k11] - d[k01]) * fx;
    25→    const u0 = d[k00 + 1] + (d[k10 + 1] - d[k00 + 1]) * fx;
    26→    const u1 = d[k01 + 1] + (d[k11 + 1] - d[k01 + 1]) * fx;
    27→    const v0 = d[k00 + 2] + (d[k10 + 2] - d[k00 + 2]) * fx;
    28→    const v1 = d[k01 + 2] + (d[k11 + 2] - d[k01 + 2]) * fx;
    29→    out.eta = e0 + (e1 - e0) * fy;
    30→    out.u = u0 + (u1 - u0) * fy;
    31→    out.v = v0 + (v1 - v0) * fy;
    32→  }
    33→}

    35→// ------------------------------------------------------------------ textures
    36→function canvasTex(w: number, h: number, draw: (c: CanvasRenderingContext2D) => void): THREE.Texture {
    37→  const cv = document.createElement('canvas');
    38→  cv.width = w;
    39→  cv.height = h;
    40→  const ctx = cv.getContext('2d')!;
    41→  draw(ctx);
    42→  const tex = new THREE.CanvasTexture(cv);
    43→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    44→  tex.colorSpace = THREE.SRGBColorSpace;
    45→  tex.anisotropy = 4;
    46→  return tex;
    47→}

    49→function makeConcreteTex(base = '#9b948a'): THREE.Texture {
    50→  return canvasTex(256, 256, (c) => {
    51→    c.fillStyle = base;
    52→    c.fillRect(0, 0, 256, 256);
    53→    // aggregate speckle
    54→    for (let i = 0; i < 2600; i++) {
    55→      const v = 150 + Math.random() * 70;
    56→      c.fillStyle = `rgba(${v},${v - 4},${v - 10},${0.06 + Math.random() * 0.10})`;
    57→      const s = 1 + Math.random() * 2.4;
    58→      c.fillRect(Math.random() * 256, Math.random() * 256, s, s);
    59→    }
    60→    // formwork panel joints + tie holes
    61→    c.strokeStyle = 'rgba(0,0,0,0.14)';
    62→    c.lineWidth = 2;
    63→    for (let y = 0; y <= 256; y += 64) {
    64→      c.beginPath(); c.moveTo(0, y); c.lineTo(256, y); c.stroke();
    65→    }
    66→    for (let x = 0; x <= 256; x += 86) {
    67→      c.beginPath(); c.moveTo(x, 0); c.lineTo(x, 256); c.stroke();
    68→      c.fillStyle = 'rgba(0,0,0,0.22)';
    69→      for (let y = 32; y < 256; y += 64) {
    70→        c.beginPath(); c.arc(x, y, 3, 0, Math.PI * 2); c.fill();
    71→      }
    72→    }
    73→    // weather stains
    74→    for (let i = 0; i < 14; i++) {
    75→      const x = Math.random() * 256, y = Math.random() * 256;
    76→      const g = c.createLinearGradient(x, y, x, y + 30 + Math.random() * 50);
    77→      g.addColorStop(0, 'rgba(60,55,45,0.13)');
    78→      g.addColorStop(1, 'rgba(60,55,45,0)');
    79→      c.fillStyle = g;
    80→      c.fillRect(x, y, 4 + Math.random() * 10, 30 + Math.random() * 55);
    81→    }
    82→  });
    83→}

    85→function makeAsphaltTex(): THREE.Texture {
    86→  return canvasTex(128, 256, (c) => {
    87→    c.fillStyle = '#4a4a4c';
    88→    c.fillRect(0, 0, 128, 256);
    89→    for (let i = 0; i < 900; i++) {
    90→      const v = 60 + Math.random() * 40;
    91→      c.fillStyle = `rgba(${v},${v},${v + 3},${0.10 + Math.random() * 0.15})`;
    92→      c.fillRect(Math.random() * 128, Math.random() * 256, 1.6, 1.6);
    93→    }
    94→    // center dashes
    95→    c.fillStyle = 'rgba(230,225,190,0.85)';
    96→    for (let y = 8; y < 256; y += 42) c.fillRect(61, y, 6, 22);
    97→    // edge lines
    98→    c.fillStyle = 'rgba(225,220,200,0.5)';
    99→    c.fillRect(5, 0, 3, 256);
   100→    c.fillRect(120, 0, 3, 256);
   101→  });
   102→}

   118→export function buildDam(): DamProps {
   119→  const group = new THREE.Group();
   120→  const concTex = makeConcreteTex('#a09a8e');
   121→  concTex.repeat.set(2.2, 1.2);
   122→  const concTexDark = makeConcreteTex('#8f887b');
   123→  concTexDark.repeat.set(2.2, 1.2);
   124→  const concrete = new THREE.MeshStandardMaterial({ map: concTex, roughness: 0.88, metalness: 0.02, envMapIntensity: 0.35 });
   125→  const concreteDark = new THREE.MeshStandardMaterial({ map: concTexDark, roughness: 0.92, metalness: 0.02, envMapIntensity: 0.3 });
   126→  const steel = new THREE.MeshStandardMaterial({ color: 0x5d6870, roughness: 0.5, metalness: 0.55 });
   127→  const paintWhite = new THREE.MeshStandardMaterial({ color: 0xd8d5cc, roughness: 0.7, metalness: 0.05 });
   128→
   129→  const monolith = (z0: number, z1: number, crestY: number, mat: THREE.Material): THREE.Mesh => {
   130→    const s = new THREE.Shape();
   131→    s.moveTo(0, 6);
   132→    s.lineTo(0, crestY);
   133→    s.lineTo(3.4, crestY);
   134→    s.lineTo(8.6, 9.0);
   135→    s.lineTo(8.6, 6);
   136→    s.lineTo(0, 6);
   137→    const geo = new THREE.ExtrudeGeometry(s, { depth: z1 - z0, bevelEnabled: false, curveSegments: 1 });
   138→    geo.translate(DAM_X, 0, z0);
   139→    const m = new THREE.Mesh(geo, mat);
   140→    m.castShadow = true;
   141→    m.receiveShadow = true;
   142→    return m;
   143→  };
   144→
   145→  // crest roadway + railings + lamps (across the whole dam)
   146→  const CREST_Z0 = -38, CREST_Z1 = 38;
   147→  const roadTex = makeAsphaltTex();
   148→  roadTex.repeat.set(1, 10);
   149→  const crestRoad = new THREE.Mesh(
   150→    new THREE.BoxGeometry(3.1, 0.22, CREST_Z1 - CREST_Z0),
   151→    new THREE.MeshStandardMaterial({ map: roadTex, roughness: 0.85, metalness: 0.03 }),
   152→  );
   153→  crestRoad.position.set(DAM_X + 1.7, CREST + 0.1, 0);
   154→  crestRoad.castShadow = crestRoad.receiveShadow = true;
   155→  group.add(crestRoad);
   156→
   157→  // railing posts (instanced) + rails
   158→  const postCount = Math.floor((CREST_Z1 - CREST_Z0) / 2.4);
   159→  const postGeo = new THREE.BoxGeometry(0.09, 0.85, 0.09);
   160→  const posts = new THREE.InstancedMesh(postGeo, steel, postCount * 2);
   161→  const m4 = new THREE.Matrix4();
   162→  let pi = 0;
   163→  for (let i = 0; i < postCount; i++) {
   164→    const z = CREST_Z0 + 1 + i * 2.4;
   165→    for (const side of [DAM_X + 0.3, DAM_X + 3.1]) {
   166→      m4.makeTranslation(side, CREST + 0.62, z);
   167→      posts.setMatrixAt(pi++, m4);
   168→    }
   169→  }
   170→  posts.castShadow = true;
   171→  group.add(posts);
   172→  for (const side of [DAM_X + 0.3, DAM_X + 3.1]) {
   173→    const rail = new THREE.Mesh(new THREE.BoxGeometry(0.07, 0.07, CREST_Z1 - CREST_Z0), steel);
   174→    rail.position.set(side, CREST + 1.0, 0);
   175→    group.add(rail);
   176→    const rail2 = rail.clone();
   177→    rail2.position.y = CREST + 0.55;
   178→    group.add(rail2);
   179→  }
   180→
   181→  // lamp posts
   182→  for (let z = -35; z <= 35; z += 14) {
   183→    const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.07, 2.6, 6), steel);
   184→    pole.position.set(DAM_X + 3.0, CREST + 1.5, z);
   185→    pole.castShadow = true;
   186→    const arm = new THREE.Mesh(new THREE.BoxGeometry(0.7, 0.05, 0.05), steel);
   187→    arm.position.set(DAM_X + 2.65, CREST + 2.78, z);
   188→    const head = new THREE.Mesh(
   189→      new THREE.CylinderGeometry(0.12, 0.16, 0.14, 8),
   190→      new THREE.MeshStandardMaterial({ color: 0xe8e4d4, emissive: 0x535134, roughness: 0.5 }),
   191→    );
   192→    head.position.set(DAM_X + 2.3, CREST + 2.72, z);
   193→    group.add(pole, arm, head);
   194→  }
   195→
   196→  // intact outer monoliths
   197→  group.add(monolith(-38, BLOCK_Z0, CREST, concrete));
   198→  group.add(monolith(BLOCK_Z1, 18, CREST, concrete));
   199→  group.add(monolith(18, 30, 22.8, concreteDark)); // spillway sill (matches SPILL_CREST_CLOSED)
   200→  group.add(monolith(30, 38, CREST, concrete));
   201→
   202→  // --- spillway: 3 piers, 2 radial gates, hoist bridge, trunnion arms
   203→  const pierZ = [18.7, 24.0, 29.3];
   204→  for (const zc of pierZ) {
   205→    const pier = new THREE.Mesh(new THREE.BoxGeometry(2.6, 15.2, 1.5), concreteDark);
   206→    pier.position.set(DAM_X + 2.2, 17.4, zc);
   207→    pier.castShadow = pier.receiveShadow = true;
   208→    group.add(pier);
   209→    const nose = new THREE.Mesh(new THREE.CylinderGeometry(0.75, 0.75, 2.6, 8, 1, false, 0, Math.PI), concreteDark);
   210→    nose.rotation.set(Math.PI / 2, 0, Math.PI / 2);
   211→    nose.position.set(DAM_X + 0.9, 17.4, zc);
   212→    group.add(nose);
   213→  }
   214→  const hoistDeck = new THREE.Mesh(new THREE.BoxGeometry(2.2, 0.7, 12.4), concreteDark);
   215→  hoistDeck.position.set(DAM_X + 2.2, 25.3, 24);
   216→  hoistDeck.castShadow = true;
   217→  group.add(hoistDeck);
   218→  const hoistRoof = new THREE.Mesh(new THREE.BoxGeometry(2.6, 0.28, 12.8), paintWhite);
   219→  hoistRoof.position.set(DAM_X + 2.2, 26.6, 24);
   220→  group.add(hoistRoof);
   221→  for (const zh of [21.4, 26.6]) {
   222→    const col = new THREE.Mesh(new THREE.BoxGeometry(0.18, 1.1, 0.18), steel);
   223→    col.position.set(DAM_X + 1.15, 26.0, zh);
   224→    group.add(col, col.clone().translateX(2.1));
   225→  }
   226→
   227→  const gates: THREE.Mesh[] = [];
   228→  for (const zc of [21.3, 26.7]) {
   229→    const gate = new THREE.Mesh(new THREE.BoxGeometry(0.9, 7.5, 4.9), steel);
   230→    gate.position.set(DAM_X + 2.2, 19.05, zc); // closed gate top = 22.8 (sill elevation)
   231→    gate.castShadow = true;
   232→    group.add(gate);
   233→    gates.push(gate);
   234→    // trunnion arm + hub
   235→    const arm = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.09, 4.6, 6), steel);
   236→    arm.rotation.z = Math.PI / 2 - 0.62;
   237→    arm.position.set(DAM_X + 3.6, 16.4, zc);
   238→    group.add(arm);
   239→    const hub = new THREE.Mesh(new THREE.CylinderGeometry(0.24, 0.24, 0.6, 8), steel);
   240→    hub.rotation.x = Math.PI / 2;
   241→    hub.position.set(DAM_X + 5.2, 14.9, zc);
   242→    group.add(hub);
   243→  }
   244→
   245→  // upstream algal stain band (slightly proud of the face, subtle)
   246→  const stainBand = new THREE.Mesh(
   247→    new THREE.BoxGeometry(0.08, 2.0, 76),
   248→    new THREE.MeshStandardMaterial({ color: 0x6b7362, roughness: 0.95, transparent: true, opacity: 0.16 }),
   249→  );
   250→  stainBand.position.set(DAM_X - 0.05, RES_LEVEL - 0.6, 0);
   251→  group.add(stainBand);
   252→
   253→  // control building on the right abutment
   254→  const bld = new THREE.Group();
   255→  const bBody = new THREE.Mesh(new THREE.BoxGeometry(3.4, 2.6, 4.4), paintWhite);
   256→  bBody.position.y = 1.3;
   257→  bBody.castShadow = bBody.receiveShadow = true;
   258→  const bRoof = new THREE.Mesh(new THREE.BoxGeometry(3.8, 0.22, 4.8), concreteDark);
   259→  bRoof.position.y = 2.72;
   260→  const bWin = new THREE.Mesh(
   261→    new THREE.BoxGeometry(3.42, 0.8, 2.6),
   262→    new THREE.MeshStandardMaterial({ color: 0x27343c, roughness: 0.35, metalness: 0.4 }),
   263→  );
   264→  bWin.position.y = 1.55;
   265→  const bDoor = new THREE.Mesh(
   266→    new THREE.BoxGeometry(0.05, 1.5, 0.8),
   267→    new THREE.MeshStandardMaterial({ color: 0x3d5a52, roughness: 0.6 }),
   268→  );
   269→  bDoor.position.set(-1.72, 0.75, 0);
   270→  bld.add(bBody, bRoof, bWin, bDoor);
   271→  bld.position.set(DAM_X + 5.4, CREST, 33.4);
   272→  group.add(bld);
   273→  // mast + antenna
   274→  const mast = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.08, 3.4, 6), steel);
   275→  mast.position.set(DAM_X + 4.6, CREST + 4.6, 33.4);
   276→  group.add(mast);
   277→
   278→  // --- breach monolith blocks (each sinks individually during a scenario)
   279→  const breachBlocks: BreachBlock[] = [];
   280→  for (let b = 0; b < BLOCK_N; b++) {
   281→    const z0 = BLOCK_Z0 + b * BLOCK_W;
   282→    const z1 = z0 + BLOCK_W;
   283→    const bg = new THREE.Group();
   284→    bg.add(monolith(z0, z1, CREST, concrete));
   285→    // parapet segment on top of the block
   286→    const len = z1 - z0;
   287→    const p1 = new THREE.Mesh(new THREE.BoxGeometry(0.45, 1.0, len), concreteDark);
   288→    p1.position.set(DAM_X + 0.25, CREST + 0.5, z0 + len / 2);
   289→    const p2 = p1.clone();
   290→    p2.position.x = DAM_X + 3.15;
   291→    p1.castShadow = p2.castShadow = true;
   292→    bg.add(p1, p2);
   293→    // crest road patch belongs to the block so it sinks with it
   294→    const patch = new THREE.Mesh(
   295→      new THREE.BoxGeometry(3.1, 0.2, len),
   296→      new THREE.MeshStandardMaterial({ map: roadTex.clone(), roughness: 0.85 }),
   297→    );
   298→    (patch.material as THREE.MeshStandardMaterial).map!.repeat.set(1, 1);
   299→    (patch.material as THREE.MeshStandardMaterial).map!.needsUpdate = true;
   300→    patch.position.set(DAM_X + 1.7, CREST + 0.1, z0 + len / 2);
   301→    patch.castShadow = true;
   302→    bg.add(patch);
   303→    group.add(bg);
   304→    breachBlocks.push({ group: bg, z0, z1 });
   305→  }
   306→
   307→  // stilling-basin apron + baffle blocks downstream — sits ON the channel bed
   308→  // at the same elevation as the simulated apron shelf (APRON_TOP), so breach
   309→  // flow visibly crashes onto it and churns over the baffles instead of
   310→  // disappearing under a floating slab
   311→  const apron = new THREE.Mesh(new THREE.BoxGeometry(14, 0.35, 60), concreteDark);
   312→  apron.position.set(DAM_X + 15.6, APRON_TOP - 0.175, 0);
   313→  apron.receiveShadow = true;
   314→  group.add(apron);
   315→  for (let i = 0; i < 3; i++) {
   316→    for (let z = -24; z <= 24; z += 6) {
   317→      const baf = new THREE.Mesh(new THREE.BoxGeometry(0.9, 1.1, 1.4), concreteDark);
   318→      baf.position.set(DAM_X + 10 + i * 4.5, APRON_TOP + 0.55, z + (i % 2) * 3);
   319→      baf.castShadow = true;
   320→      group.add(baf);
   321→    }
   322→  }
   323→
   324→  return { group, breachBlocks, gates, stainBand };
   325→}

   334→function makeHouse(idx: number): THREE.Group {
   335→  const g = new THREE.Group();
   336→  const wallColors = [0xd9cfba, 0xcdb998, 0xdcd2c0, 0xc4b69c, 0xd7c2a7, 0xd0c7b5, 0xc9bda4, 0xd5cbb8];
   337→  const wall = new THREE.MeshStandardMaterial({ color: wallColors[idx % wallColors.length], roughness: 0.92 });
   338→  const trim = new THREE.MeshStandardMaterial({ color: 0x6e5a45, roughness: 0.85 });
   339→  const glass = new THREE.MeshStandardMaterial({ color: 0x2b3d46, roughness: 0.25, metalness: 0.5 });
   340→  const flat = idx % 3 === 2;
   341→
   342→  const plinth = new THREE.Mesh(new THREE.BoxGeometry(3.6, 0.3, 3.2), concreteGrey());
   343→  plinth.position.y = 0.15;
   344→  plinth.receiveShadow = true;
   345→  const body = new THREE.Mesh(new THREE.BoxGeometry(3.2, 2.4, 2.8), wall);
   346→  body.position.y = 1.5;
   347→  body.castShadow = body.receiveShadow = true;
   348→  g.add(plinth, body);
   349→
   350→  // windows front/back + door
   351→  for (const zs of [-0.75, 0.75]) {
   352→    for (const xo of [-0.95, 0.95]) {
   353→      const w = new THREE.Mesh(new THREE.BoxGeometry(0.55, 0.62, 0.06), glass);
   354→      w.position.set(xo, 1.7, 1.42);
   355→      w.castShadow = true;
   356→      g.add(w);
   357→      const fr = new THREE.Mesh(new THREE.BoxGeometry(0.67, 0.74, 0.04), trim);
   358→      fr.position.set(xo, 1.7, 1.40);
   359→      g.add(fr);
   360→    }
   361→    const wb = new THREE.Mesh(new THREE.BoxGeometry(0.55, 0.62, 0.06), glass);
   362→    wb.position.set(0, 1.7, -1.42);
   363→    g.add(wb);
   364→  }
   365→  const door = new THREE.Mesh(new THREE.BoxGeometry(0.72, 1.55, 0.07), trim);
   366→  door.position.set(-0.85, 1.075, 1.43);
   367→  g.add(door);
   368→
   369→  if (flat) {
   370→    const roof = new THREE.Mesh(new THREE.BoxGeometry(3.5, 0.18, 3.1), concreteGrey());
   371→    roof.position.y = 2.8;
   372→    roof.castShadow = true;
   373→    const tank = new THREE.Mesh(
   374→      new THREE.CylinderGeometry(0.4, 0.4, 0.62, 10),
   375→      new THREE.MeshStandardMaterial({ color: 0x1e2224, roughness: 0.6 }),
   376→    );
   377→    tank.position.set(-0.9, 3.2, -0.6);
   378→    tank.castShadow = true;
   379→    g.add(roof, tank);
   380→  } else {
   381→    // pitched gable roof (Kerala tile colour)
   382→    const roofMat = new THREE.MeshStandardMaterial({ color: idx % 2 ? 0x8a4a35 : 0x7a4438, roughness: 0.88 });
   383→    const slope = 0.62;
   384→    const slabG = new THREE.BoxGeometry(2.05, 0.09, 3.45);
   385→    const s1 = new THREE.Mesh(slabG, roofMat);
   386→    s1.position.set(-0.83, 3.32, 0);
   387→    s1.rotation.z = slope;
   388→    const s2 = new THREE.Mesh(slabG, roofMat);
   389→    s2.position.set(0.83, 3.32, 0);
   390→    s2.rotation.z = -slope;
   391→    s1.castShadow = s2.castShadow = true;
   392→    const ridge = new THREE.Mesh(new THREE.BoxGeometry(0.22, 0.14, 3.5), trim);
   393→    ridge.position.y = 3.93;
   394→    const gableG = new THREE.BufferGeometry();
   395→    gableG.setAttribute(
   396→      'position',
   397→      new THREE.BufferAttribute(
   398→        new Float32Array([
   399→          -1.6, 2.7, 0, 1.6, 2.7, 0, 0, 3.62, 0,
   400→          -1.6, 2.7, 0.02, 1.6, 2.7, 0.02, 0, 3.62, 0.02,
   401→        ]),
   402→        3,
   403→      ),
   404→    );
   405→    gableG.computeVertexNormals();
   406→    const gable = new THREE.Mesh(gableG, wall);
   407→    gable.position.z = 1.41;
   408→    g.add(s1, s2, ridge, gable);
   409→  }
   410→  return g;
   411→}

   442→function makePalm(rnd: number): THREE.Group {
   443→  const g = new THREE.Group();
   444→  const trunkMat = new THREE.MeshStandardMaterial({ color: 0x8a6a48, roughness: 0.95 });
   445→  const frondMat = new THREE.MeshStandardMaterial({ color: 0x3d7a34, roughness: 0.9, side: THREE.DoubleSide });
   446→  const lean = (rnd - 0.5) * 0.5;
   447→  let y = 0, x = 0;
   448→  const h = 4.4 + rnd * 2.2;
   449→  const seg = 4;
   450→  for (let i = 0; i < seg; i++) {
   451→    const t0 = i / seg, t1 = (i + 1) / seg;
   452→    const len = h / seg;
   453→    const cyl = new THREE.Mesh(new THREE.CylinderGeometry(0.13 * (1 - t1 * 0.45) + 0.05, 0.13 * (1 - t0 * 0.45) + 0.06, len, 6), trunkMat);
   454→    x += lean * len * 0.45;
   455→    y += len;
   456→    cyl.position.set(x, y - len / 2, 0);
   457→    cyl.rotation.z = -lean * 0.5;
   458→    cyl.castShadow = true;
   459→    g.add(cyl);
   460→  }
   461→  const crownY = h + 0.2;
   462→  for (let i = 0; i < 8; i++) {
   463→    const a = (i / 8) * Math.PI * 2 + rnd;
   464→    const frond = new THREE.Mesh(new THREE.ConeGeometry(0.42, 2.6, 4), frondMat);
   465→    frond.scale.y = 0.22;
   466→    frond.position.set(x + Math.cos(a) * 1.0, crownY + 0.25, Math.sin(a) * 1.0);
   467→    frond.rotation.set(Math.sin(a) * 0.9, 0, -Math.cos(a) * 0.9 - 0.25);
   468→    frond.castShadow = true;
   469→    g.add(frond);
   470→  }
   471→  for (let i = 0; i < 3; i++) {
   472→    const nut = new THREE.Mesh(
   473→      new THREE.SphereGeometry(0.12, 6, 5),
   474→      new THREE.MeshStandardMaterial({ color: 0x6b5433, roughness: 0.9 }),
   475→    );
   476→    nut.position.set(x + (i - 1) * 0.2, crownY - 0.05, (i % 2) * 0.2 - 0.1);
   477→    g.add(nut);
   478→  }
   479→  return g;
   480→}

   541→export function buildRoads(): { group: THREE.Group } {
   542→  const group = new THREE.Group();
   543→  const asphalt = new THREE.MeshStandardMaterial({ color: 0x46464a, roughness: 0.95, metalness: 0.0 });
   544→  const lineMat = new THREE.MeshStandardMaterial({ color: 0xd8d2b8, roughness: 0.8 });
   545→
   546→  const roadRibbon = (pts: [number, number][], w = 3.2): void => {
   547→    // sample terrain-following ribbon
   548→    const pos: number[] = [];
   549→    const idxArr: number[] = [];
   550→    for (let i = 0; i < pts.length - 1; i++) {
   551→      for (let s = 0; s < 4; s++) {
   552→        const t0 = s / 4, t1 = (s + 1) / 4;
   553→        const x0 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t0;
   554→        const z0 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t0;
   555→        const x1 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t1;
   556→        const z1 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t1;
   557→        const dx = z1 - z0, dz = -(x1 - x0);
   558→        const dl = Math.hypot(dx, dz) || 1;
   559→        const nx = (dx / dl) * w * 0.5, nz = (dz / dl) * w * 0.5;
   560→        const a = bedAt(x0 - nx, z0 - nz) + 0.16;
   561→        const b = bedAt(x0 + nx, z0 + nz) + 0.16;
   562→        const c = bedAt(x1 - nx, z1 - nz) + 0.16;
   563→        const d = bedAt(x1 + nx, z1 + nz) + 0.16;
   564→        const base = pos.length / 3;
   565→        pos.push(x0 - nx, a, z0 - nz, x0 + nx, b, z0 + nz, x1 - nx, c, z1 - nz, x1 + nx, d, z1 + nz);
   566→        idxArr.push(base, base + 2, base + 1, base + 1, base + 2, base + 3);
   567→      }
   568→    }
   569→    const geo = new THREE.BufferGeometry();
   570→    geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   571→    geo.setIndex(idxArr);
   572→    geo.computeVertexNormals();
   573→    const mesh = new THREE.Mesh(geo, asphalt);
   574→    mesh.receiveShadow = true;
   575→    group.add(mesh);
   576→  };
   577→
   578→  // main valley road along the +z bench
   579→  roadRibbon([
   580→    [117, 27.5], [124, 26], [132, 24.8], [140, 23.2], [148, 22.2], [156, 21.2],
   581→    [164, 20.2], [172, 19.6], [180, 20.2], [188, 21.8],
   582→  ]);
   583→  // spur along the -z bench
   584→  roadRibbon([
   585→    [117, -27.5], [124, -26.5], [132, -25.5], [140, -24.5], [148, -23.5], [156, -23],
   586→    [164, -23.5], [172, -24], [180, -24.5], [188, -25],
   587→  ]);
   588→  // access spur to the bridge head
   589→  roadRibbon([[148, 22.4], [150, 20], [150, 17]], 2.6);
   590→  roadRibbon([[150, -17], [150, -20], [148, -23.2]], 2.6);
   591→
   592→  // center dashes on the main road
   593→  for (let i = 0; i < 26; i++) {
   594→    const x = 119 + i * 2.7;
   595→    const z = 26.2 - (x - 119) * 0.075;
   596→    const dash = new THREE.Mesh(new THREE.BoxGeometry(1.5, 0.02, 0.22), lineMat);
   597→    dash.position.set(x, bedAt(x, z) + 0.18, z);
   598→    dash.rotation.y = -0.35;
   599→    group.add(dash);
   600→  }
   601→
   602→  // bridge over the channel at x = 150
   603→  const bridgeY = bedAt(150, 0) + 3.4;
   604→  const deck = new THREE.Mesh(new THREE.BoxGeometry(4.2, 0.5, 52), concreteGrey());
   605→  deck.position.set(150, bridgeY, 0);
   606→  deck.castShadow = deck.receiveShadow = true;
   607→  group.add(deck);
   608→  const deckRoad = new THREE.Mesh(new THREE.BoxGeometry(3.2, 0.06, 52), asphalt);
   609→  deckRoad.position.set(150, bridgeY + 0.28, 0);
   610→  group.add(deckRoad);
   611→  for (const s of [-1, 1]) {
   612→    const rail = new THREE.Mesh(new THREE.BoxGeometry(0.08, 0.5, 52), lineMat);
   613→    rail.position.set(150 + s * 1.95, bridgeY + 0.55, 0);
   614→    group.add(rail);
   615→    for (let z = -24; z <= 24; z += 4) {
   616→      const post = new THREE.Mesh(new THREE.BoxGeometry(0.1, 0.55, 0.1), lineMat);
   617→      post.position.set(150 + s * 1.95, bridgeY + 0.35, z);
   618→      group.add(post);
   619→    }
   620→  }
   621→  for (const zc of [-7, 7]) {
   622→    const pier = new THREE.Mesh(new THREE.CylinderGeometry(0.55, 0.7, bridgeY - bedAt(150, zc) + 0.6, 8), concreteGrey());
   623→    pier.position.set(150, bedAt(150, zc) + (bridgeY - bedAt(150, zc)) / 2, zc);
   624→    pier.castShadow = true;
   625→    group.add(pier);
   626→  }
   627→  return { group };
   628→}

   811→export class RiverAudio {
   812→  private ctx: AudioContext | null = null;
   813→  private master: GainNode | null = null;
   814→  private riverGain: GainNode | null = null;
   815→  private rumbleGain: GainNode | null = null;
   816→  enabled = false;
   817→
   818→  private ensure(): void {
   819→    if (this.ctx) return;
   820→    try {
   821→      const Ctx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   822→      const ctx = new Ctx();
   823→      this.ctx = ctx;
   824→      const len = ctx.sampleRate * 2;
   825→      const buf = ctx.createBuffer(1, len, ctx.sampleRate);
   826→      const d = buf.getChannelData(0);
   827→      for (let i = 0; i < len; i++) d[i] = Math.random() * 2 - 1;
   828→
   829→      const master = ctx.createGain();
   830→      master.gain.value = 0;
   831→      master.connect(ctx.destination);
   832→      this.master = master;
   833→
   834→      const mk = (freq: number, gain: number): { src: AudioBufferSourceNode; g: GainNode } => {
   835→        const src = ctx.createBufferSource();
   836→        src.buffer = buf;
   837→        src.loop = true;
   838→        const filt = ctx.createBiquadFilter();
   839→        filt.type = 'lowpass';
   840→        filt.frequency.value = freq;
   841→        const g = ctx.createGain();
   842→        g.gain.value = gain;
   843→        src.connect(filt);
   844→        filt.connect(g);
   845→        g.connect(master);
   846→        src.start();
   847→        return { src, g };
   848→      };
   849→      const river = mk(750, 0.05);
   850→      this.riverGain = river.g;
   851→      const rumble = mk(120, 0.0);
   852→      this.rumbleGain = rumble.g;
   853→    } catch {
   854→      this.ctx = null;
   855→    }
   856→  }
   857→
   858→  setEnabled(b: boolean): void {
   859→    this.enabled = b;
   860→    if (b) {
   861→      this.ensure();
   862→      if (this.ctx && this.master) {
   863→        void this.ctx.resume();
   864→        this.master.gain.setTargetAtTime(0.9, this.ctx.currentTime, 0.2);
   865→      }
   866→    } else if (this.ctx && this.master) {
   867→      this.master.gain.setTargetAtTime(0, this.ctx.currentTime, 0.1);
   868→    }
   869→  }
   870→
   871→  setIntensity(v01: number): void {
   872→    if (!this.ctx || !this.enabled || !this.riverGain || !this.rumbleGain) return;
   873→    const t = this.ctx.currentTime;
   874→    this.riverGain.gain.setTargetAtTime(0.03 + 0.30 * v01, t, 0.4);
   875→    this.rumbleGain.gain.setTargetAtTime(Math.max(this.rumbleGain.gain.value, 0.16 * v01 * v01), t, 0.6);
   876→  }
   877→
   878→  burst(power = 1): void {
   879→    if (!this.ctx || !this.enabled || !this.rumbleGain) return;
   880→    const t = this.ctx.currentTime;
   881→    const g = this.rumbleGain.gain;
   882→    g.cancelScheduledValues(t);
   883→    g.setValueAtTime(Math.max(g.value, 0.001), t);
   884→    g.linearRampToValueAtTime(0.65 * power, t + 0.15);
   885→    g.setTargetAtTime(0.02, t + 0.4, 1.4);
   886→  }
   887→
   888→  dispose(): void {
   889→    if (this.ctx) void this.ctx.close();
   890→    this.ctx = null;
   891→  }
