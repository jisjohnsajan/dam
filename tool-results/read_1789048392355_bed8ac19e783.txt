   334→  // ================================================================ terrain mesh
   335→  private buildTerrainMesh(): void {
   336→    // bed grid on cell centers
   337→    this.bedGrid = new Float32Array(NX * NZ);
   338→    for (let j = 0; j < NZ; j++)
   339→      for (let i = 0; i < NX; i++)
   340→        this.bedGrid[j * NX + i] = bedAt((i + 0.5) * DX, (j + 0.5) * DZ - LZ / 2);
   341→
   342→    const geo = new THREE.PlaneGeometry(LX, LZ, NX - 1, NZ - 1);
   343→    geo.rotateX(-Math.PI / 2);
   344→    geo.translate(LX / 2, 0, 0);
   345→    const pos = geo.attributes.position as THREE.BufferAttribute;
   346→    const colors = new Float32Array(pos.count * 3);
   347→    const col = new THREE.Color();
   348→    const tc = { r: 0, g: 0, b: 0 };
   349→    for (let i = 0; i < pos.count; i++) {
   350→      const x = pos.getX(i);
   351→      const z = pos.getZ(i);
   352→      const y = this.sampleBed(x, z);
   353→      pos.setY(i, y);
   354→      const e = 1.2;
   355→      const sx = (this.sampleBed(x + e, z) - this.sampleBed(x - e, z)) / (2 * e);
   356→      const sz = (this.sampleBed(x, z + e) - this.sampleBed(x, z - e)) / (2 * e);
   357→      const slope = Math.sqrt(sx * sx + sz * sz);
   358→      terrainColor(x, z, y, slope, tc);
   359→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   360→      colors[i * 3] = col.r;
   361→      colors[i * 3 + 1] = col.g;
   362→      colors[i * 3 + 2] = col.b;
   363→    }
   364→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   365→    geo.computeVertexNormals();
   366→    const mesh = new THREE.Mesh(
   367→      geo,
   368→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.96, metalness: 0, envMapIntensity: 0.22 }),
   369→    );
   370→    mesh.receiveShadow = true;
   371→    this.scene.add(mesh);
   372→
   373→    // surround plane so the domain doesn't float in the void
   374→    const surround = new THREE.Mesh(
   375→      new THREE.PlaneGeometry(3000, 3000),
   376→      new THREE.MeshStandardMaterial({ color: 0xa8ad97, roughness: 1, metalness: 0, envMapIntensity: 0.1 }),
   377→    );
   378→    surround.rotation.x = -Math.PI / 2;
   379→    surround.position.set(LX / 2, 1.2, 0);
   380→    this.scene.add(surround);
   381→  }

   383→  private sampleBed(x: number, z: number): number {
   384→    const fu = clamp((x / DX) - 0.5, 0, NX - 1.001);
   385→    const fv = clamp((z + LZ / 2) / DZ - 0.5, 0, NZ - 1.001);
   386→    const i = Math.floor(fu), j = Math.floor(fv);
   387→    const fx = fu - i, fy = fv - j;
   388→    const i1 = Math.min(i + 1, NX - 1), j1 = Math.min(j + 1, NZ - 1);
   389→    const g = this.bedGrid;
   390→    const a = g[j * NX + i] + (g[j * NX + i1] - g[j * NX + i]) * fx;
   391→    const b = g[j1 * NX + i] + (g[j1 * NX + i1] - g[j1 * NX + i]) * fx;
   392→    return a + (b - a) * fy;
   393→  }

   465→      uDriveOn: { value: 1 },
   466→      uDriveEta: { value: fracToSimLevel(this.liveFrac) },
   467→      uDriveRate: { value: 1.2 },

   645→  // per-frame storm driver: rain streaks, darkened sky/fog, lightning flashes
   646→  private updateWeather(dtReal: number): void {
   647→    const target = this.rainInflow > 0 ? (this.rainHeavy ? 1 : 0.55) : 0;
   648→    this.rainVis += (target - this.rainVis) * Math.min(1, dtReal * 1.4);
   649→    const rv = this.rainVis;
   650→
   651→    if (this.rainMesh) {
   652→      this.rainMesh.visible = rv > 0.02;
   653→      if (rv > 0.02) {
   654→        const p = this.rainPos;
   655→        const n = p.length / 6;
   656→        const fall = 46 * dtReal;
   657→        const drift = 8 * dtReal;
   658→        for (let i = 0; i < n; i++) {
   659→          const k = i * 6;
   660→          p[k] += drift; p[k + 1] -= fall; p[k + 2] += drift * 0.35;
   661→          p[k + 3] = p[k] + 0.26; p[k + 4] = p[k + 1] + 1.5; p[k + 5] = p[k + 2] + 0.09;
   662→          if (p[k + 1] < 2.5) {
   663→            const x = -30 + Math.random() * (LX + 110);
   664→            const y = 48 + Math.random() * 34;
   665→            const z = -LZ / 2 - 40 + Math.random() * (LZ + 80);
   666→            p[k] = x; p[k + 1] = y; p[k + 2] = z;
   667→            p[k + 3] = x + 0.26; p[k + 4] = y + 1.5; p[k + 5] = z + 0.09;
   668→          }
   669→        }
   670→        (this.rainMesh.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true;
   671→        (this.rainMesh.material as THREE.LineBasicMaterial).opacity = 0.38 * rv;
   672→      }
   673→    }
   674→
   675→    // occasional lightning during heavy rain
   676→    this.flash *= Math.exp(-dtReal * 6.5);
   677→    if (rv > 0.6) {
   678→      this.boltTimer -= dtReal;
   679→      if (this.boltTimer <= 0) {
   680→        this.boltTimer = 5 + Math.random() * 10;
   681→        this.flash = 0.7 + Math.random() * 0.5;
   682→      }
   683→    }
   684→
   685→    // dim / storm the atmosphere
   686→    this.sunLight.intensity = 2.1 * (1 - 0.7 * rv) + this.flash * 5.5;
   687→    this.sunLight.color.setHex(this.flash > 0.25 ? 0xdfe8ff : 0xffe8c8);
   688→    this.hemiLight.intensity = 0.45 * (1 - 0.4 * rv) + this.flash * 1.2;
   689→    this.scene.environmentIntensity = 0.32 * (1 - 0.5 * rv);
   690→    const fog = this.scene.fog as THREE.Fog;
   691→    fog.color.copy(this.fogDay).lerp(this.fogStorm, rv * 0.85);
   692→    fog.near = 380 - 200 * rv;
   693→    fog.far = 1600 - 700 * rv;
   694→    this.renderer.toneMappingExposure = 0.85 - 0.13 * rv + this.flash * 0.12;
   695→    const su = this.skyU;
   696→    if (su) {
   697→      su.turbidity.value = 5.5 + 4.5 * rv;
   698→      su.rayleigh.value = Math.max(2.4 - 1.7 * rv, 0.3);
   699→      su.mieCoefficient.value = 0.006 + 0.02 * rv;
   700→    }
   701→
   702→    // water shader: choppier ripples + storm fog match
   703→    const wu = this.waterMat.uniforms;
   704→    wu.uRain.value = rv;
   705→    (wu.uFogColor.value as THREE.Color).copy(fog.color);
   706→    wu.uFogNear.value = fog.near;
   707→    wu.uFogFar.value = fog.far;
   708→  }

   717→  private simSubstep(dt: number): void {
   718→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   719→    eu.uState.value = this.curState.texture;
   720→    eu.uDt.value = dt;
   721→    eu.uG.value = this.gravity;
   722→    eu.uSrcRate.value = this.srcRate;
   723→    this.renderTo(this.etaPass, this.rtEta);
   724→
   725→    const vu = (this.velPass.material as THREE.ShaderMaterial).uniforms;
   726→    vu.uState.value = this.curState.texture;
   727→    vu.uEtaNew.value = this.rtEta.texture;
   728→    vu.uDt.value = dt;
   729→    vu.uG.value = this.gravity;
   730→    this.renderTo(this.velPass, this.nextState);
   731→    this.swapState();
   732→
   733→    if (this.foamOn) {
   734→      const fu = (this.foamPass.material as THREE.ShaderMaterial).uniforms;
   735→      fu.uState.value = this.curState.texture;
   736→      fu.uFoam.value = this.curFoam.texture;
   737→      fu.uDt.value = dt;
   738→      fu.uG.value = this.gravity;
   739→      this.renderTo(this.foamPass, this.nextFoam);
   740→      const t = this.curFoam;
   741→      this.curFoam = this.nextFoam;
   742→      this.nextFoam = t;
   743→    }
   744→
   745→    if (this.sprayOn) {
   746→      const su = (this.sprayPass.material as THREE.ShaderMaterial).uniforms;
   747→      su.uSpray.value = this.curSpray.texture;
   748→      su.uState.value = this.curState.texture;
   749→      su.uDt.value = dt;
   750→      su.uTime.value = this.simTime;
   751→      this.renderTo(this.sprayPass, this.nextSpray);
   752→      const t = this.curSpray;
   753→      this.curSpray = this.nextSpray;
   754→      this.nextSpray = t;
   755→    }
   756→  }

   792→  private breachSpanFor(widthM: number, loc: BreachLocation): BreachSpan {

   803→  // wire up a failing block span: invert elevations + staggered failure delays
   804→  private initBreach(span: BreachSpan): void {
   805→    this.breachSpan = span;
   806→    this.breachInverts = [];
   807→    for (let b = span.start; b < span.start + span.count; b++) {
   808→      const z0 = BLOCK_Z0 + b * BLOCK_W;
   809→      this.breachInverts.push(this.blockInvert(z0 + BLOCK_W / 2));
   810→    }
   811→    this.breachDelays = [];
   812→    for (let k = 0; k < span.count; k++) {
   813→      this.breachDelays.push(Math.random() * this.breachTau * 0.3);
   814→    }
   815→  }

   817→  runScenario(p: ScenarioParams): void {
   818→    if (this.scenario) return;
   819→    this.scenario = p;
   820→    this.phase = 'scenario';
   821→    this.scenT = 0;
   822→    this.snapClock = 0;
   823→    this.overtopT = 0;
   824→    this.rainInflow = p.rain === 'none' ? 0 : p.rain === 'moderate' ? 26 : 64;
   825→    this.rainHeavy = p.rain === 'heavy';
   826→    this.breachSpan = null;
   827→    this.breachT = null;
   828→    this.breachDepth01 = 0;
   829→    this.structDirty = true;
   830→
   831→    // reservoir set to the scenario level (drive takes it there quickly)
   832→    const target = clamp(fracToSimLevel(p.levelFrac), 0, 24.4);
   833→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   834→    eu.uDriveEta.value = target;
   835→    eu.uDriveOn.value = 1;
   836→
   837→    if (p.mechanism === 'overtopping') {
   838→      // surge inflow pushes the lake above the crest; erosion follows
   839→      eu.uDriveEta.value = Math.max(target, CREST + 1.0);
   840→    } else {
   841→      // schedule breach after the reservoir reaches level (~2-6 demo s)
   842→      this.breachT = -clamp((target - this.lastLevel) / 1.4, 1.5, 5);
   843→      this.breachTau = Math.max(p.formationMin / this.damProfile.timeMinPerSec, 4) / (p.mechanism === 'piping' ? 0.65 : 1) / this.breachRate;
   844→      this.breachPre = p.mechanism === 'piping' ? 0.22 : 0;
   845→      this.initBreach(this.breachSpanFor(p.breachWidthM, p.location));
   846→    }
   847→    this.audio.burst(0.2);
   848→  }

   931→  // ================================================================ main loop
   932→  private tick = (): void => {
   933→    this.rafId = requestAnimationFrame(this.tick);
   934→    const now = performance.now();
   935→    const dtReal = Math.min((now - this.lastFrameTime) / 1000, 0.05);
   936→    this.lastFrameTime = now;
   937→    this.fpsEma = this.fpsEma * 0.95 + (1 / Math.max(dtReal, 1e-4)) * 0.05;
   938→    const dtSim = this.paused ? 0 : dtReal * this.timeScale;
   939→
   940→    if (dtSim > 0 && !this.scrubbing) {
   941→      this.stepScenario(dtSim);
   942→
   943→      const curInflow = this.baseInflow + this.rainInflow;
   944→      this.srcRate = curInflow / (this.stripCells * DX * DZ);
   945→
   946→      if (this.structDirty) {
   947→        this.applyStruct(this.breachSpan, this.gateElev);
   948→        this.structDirty = false;
   949→      }
   950→
   951→      const sub = clamp(Math.ceil(dtSim / DT_MAX), 1, 14);
   952→      const h = dtSim / sub;

   959→        this.snapClock += dtSim;
   960→        if (this.snapClock >= SNAP_EVERY) {
   961→          this.snapClock = 0;
   962→          this.recordSnapshot();
   963→        }
   964→      }
   965→    }
   966→
   967→    // readback for CPU physics + stats
   968→    if (this.frame % 2 === 0) {
   969→      (this.downPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   970→      this.renderTo(this.downPass, this.rtDown);
   971→      this.renderer.readRenderTargetPixels(this.rtDown, 0, 0, DOWN_W, DOWN_H, this.field.data);
   972→      this.updateArrival();
   973→    }
   974→
   975→    this.updateCamera(dtReal);
   976→    this.updateWeather(dtReal);

   978→    this.waterMat.uniforms.uFoam.value = this.curFoam.texture;

   987→    if (this.shake > 0.002) {
   988→      shook = new THREE.Vector3(
   989→        (Math.random() - 0.5) * this.shake * 0.9,
   990→        (Math.random() - 0.5) * this.shake * 0.9,
   991→        (Math.random() - 0.5) * this.shake * 0.9,
   992→      );
   993→      this.camera.position.add(shook);
   994→    }
   995→    this.composer.render();
   996→    if (shook) {
   997→      this.camera.position.sub(shook);
   998→      this.shake *= Math.exp(-dtReal * 1.15);

  1001→    // stats ~10 Hz

  1003→      const s = this.computeStats();

  1005→      const v01 = clamp(s.vmax / 10, 0, 1);

  1008→
  1009→    if (this.frame === 3) {
  1010→      (window as unknown as Record<string, unknown>).__damReady = true;
  1011→    }
  1012→    this.frame++;

  1015→  // ------------------------------------------------ scenario + gate animation
  1016→  private stepScenario(dt: number): void {
  1017→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
  1018→    const sc = this.scenario;
  1019→
  1020→    if (this.gateT !== null && this.gateT < 1) {
  1021→      this.gateT = Math.min(1, this.gateT + dt / 2.5);
  1022→      const e = easeInOut(this.gateT);
  1023→      this.gateElev = SPILL_CREST_CLOSED + (GATE_OPEN_ELEV - SPILL_CREST_CLOSED) * e;
  1024→      for (const g of this.dam.gates) g.position.y = 18.85 + 6.9 * e;
  1025→      this.structDirty = true;
  1026→    }
  1027→
  1028→    if (!sc) {
  1029→      // live monitoring: drive toward the user level target
  1030→      eu.uDriveOn.value = 1;
  1031→      return;
  1032→    }
  1033→
  1034→    this.scenT += dt;
  1035→
  1036→    if (sc.mechanism === 'overtopping') {
  1037→      this.overtopT += dt;
  1038→      // once overtopped for long enough, crest erosion begins
  1039→      if (this.overtopT > 7 && (this.breachT === null || this.breachT < 0)) {
  1040→        if (this.breachSpan === null) {
  1041→          this.breachTau = Math.max(sc.formationMin / this.damProfile.timeMinPerSec, 4) / this.breachRate;
  1042→          this.breachPre = 0;
  1043→          this.initBreach(this.breachSpanFor(sc.breachWidthM, sc.location));
  1044→        }
  1045→        this.triggerBreachErosion();
  1046→      }
  1047→    }
  1048→
  1049→    // breach formation — each monolith block fails on a staggered delay, and the
  1050→    // simulated structure field descends per block with the SAME curve as the
  1051→    // visuals, so water always pours exactly through the visibly-open gap.
  1052→    if (this.breachT !== null && this.breachSpan) {
  1053→      this.breachT += dt;
  1054→      if (this.breachT > 0) {
  1055→        const tau = this.breachTau;
  1056→        const depths: number[] = [];
  1057→        let minP = 1;
  1058→        let maxD = 0;
  1059→        for (let k = 0; k < this.breachSpan.count; k++) {
  1060→          const tt = Math.max(this.breachT - (this.breachDelays[k] ?? 0), 0);
  1061→          const pk = 1 - Math.exp(-Math.pow(tt / tau, 1.7));
  1062→          const d = clamp(this.breachPre + (1 - this.breachPre) * pk, 0, 1);
  1063→          depths.push(d);
  1064→          if (d < minP) minP = d;
  1065→          if (d > maxD) maxD = d;
  1066→        }
  1067→        this.breachDepth01 = maxD;
  1068→        this.breachSpan.depth01 = maxD;
  1069→        this.breachSpan.depths = depths;
  1070→        this.structDirty = true;
  1071→        // sink the failing blocks with the SAME per-block curve as the struct field
  1072→        const bi = this.breachInverts;
  1073→        for (let k = 0; k < this.breachSpan.count; k++) {
  1074→          const blockIdx = this.breachSpan.start + k;
  1075→          const blk = this.dam.breachBlocks[blockIdx];
  1076→          if (!blk) continue;
  1077→          const d = depths[k];
  1078→          const drop = (CREST - bi[k]) * d;
  1079→          blk.group.position.y = -drop;
  1080→          blk.group.rotation.z = -0.06 * d;
  1081→          blk.group.rotation.x = (blockIdx % 2 === 0 ? 1 : -1) * 0.016 * d;
  1082→          blk.group.rotation.y = (blockIdx % 2 === 0 ? -1 : 1) * 0.012 * d;
  1083→        }
  1084→        if (minP > 0.995 && this.breachT > tau) {
  1085→          this.breachT = null; // formation complete; struct stays at full depth
  1086→        }
  1087→      }
  1088→      // stop the level drive once the breach starts draining the lake
  1089→      if (this.breachDepth01 > 0.02 && sc.mechanism !== 'overtopping') {
  1090→        eu.uDriveOn.value = 0;
  1091→      }
  1092→    }
  1093→  }

  1167→  // ================================================================ props update
  1168→  private updateProps(dt: number): void {
  1169→    const f = this.field;
  1170→    const s = this.sample;
  1171→
  1172→    for (const fl of this.floaters) {
  1173→      f.sample(fl.pos.x, fl.pos.z, s);
  1174→      const bed = bedAt(fl.pos.x, fl.pos.z);
  1175→      const dep = s.eta - bed;
  1176→      const isBoat = !((fl.mesh as THREE.Mesh).isMesh); // boats are Groups, barrels are Meshes
  1177→      if (dep > 0.35) {
  1178→        const bob = Math.sin(this.simTime * 1.7 + fl.pos.x) * 0.06;
  1179→        fl.pos.y += (s.eta + 0.15 + bob - fl.pos.y) * Math.min(1, dt * 5);
  1180→        const follow = isBoat ? 0.0 : 0.92; // moored boats only bob, don't drift
  1181→        fl.vel.x += (s.u * follow - fl.vel.x) * Math.min(1, dt * 2.2);
  1182→        fl.vel.z += (s.v * follow - fl.vel.z) * Math.min(1, dt * 2.2);
  1183→        fl.pos.x += fl.vel.x * dt;
  1184→        fl.pos.z += fl.vel.z * dt;
  1185→      } else {
  1186→        fl.vel.multiplyScalar(1 - Math.min(1, dt * 2.5));
  1187→        fl.pos.y += (bed + 0.3 - fl.pos.y) * Math.min(1, dt * 4);
  1188→      }
  1189→      fl.pos.x = clamp(fl.pos.x, 1, LX - 1);
  1190→      fl.pos.z = clamp(fl.pos.z, -LZ / 2 + 1, LZ / 2 - 1);
  1191→      const spd = Math.sqrt(fl.vel.x * fl.vel.x + fl.vel.z * fl.vel.z);
  1192→      fl.mesh.position.copy(fl.pos);
  1193→      fl.mesh.rotation.x = clamp(fl.vel.z * 0.1, -0.5, 0.5) + Math.sin(this.simTime * 2.1 + fl.pos.z) * 0.04;
  1194→      fl.mesh.rotation.z = clamp(-fl.vel.x * 0.1, -0.5, 0.5);
  1195→      if (!isBoat) fl.mesh.rotation.y += (0.2 + spd * 0.3) * dt;
  1196→    }
  1197→
  1198→    for (const h of this.houses) {
  1199→      if (h.prog >= 1) continue;
  1200→      f.sample(h.group.position.x, h.group.position.z, s);
  1201→      const wd = s.eta - h.ground;
  1202→      if (wd > 0.55) {
  1203→        h.prog = Math.min(1, h.prog + dt * 0.4);
  1204→        const p = h.prog;
  1205→        const dir = Math.atan2(s.v, s.u);
  1206→        h.group.rotation.x = -Math.cos(dir) * p * 1.35;
  1207→        h.group.rotation.z = Math.sin(dir) * p * 1.35;
  1208→        h.group.position.x += s.u * dt * 0.45 * p;
  1209→        h.group.position.z += s.v * dt * 0.45 * p;
  1210→        h.group.position.y = h.ground - p * 0.5;
  1211→      }
  1212→    }
  1213→
  1214→    for (const t of this.trees) {
  1215→      if (t.prog >= 1) continue;
  1216→      f.sample(t.group.position.x, t.group.position.z, s);
  1217→      const wd = s.eta - t.ground;
  1218→      if (wd > 0.7) {
  1219→        t.prog = Math.min(1, t.prog + dt * 0.6);
  1220→        const p = t.prog;
  1221→        const dir = Math.atan2(s.v, s.u);
  1222→        t.group.rotation.x = -Math.cos(dir) * p * 1.3;
  1223→        t.group.rotation.z = Math.sin(dir) * p * 1.3;
  1224→        t.group.position.x += s.u * dt * 0.25 * p;
  1225→        t.group.position.z += s.v * dt * 0.25 * p;
  1226→      }
  1227→    }
  1228→
  1229→    for (const c of this.chunks) {
  1230→      if (!c.active) continue;
  1231→      c.vel.y -= 11 * dt;
  1232→      c.mesh.position.addScaledVector(c.vel, dt);
  1233→      c.mesh.rotation.x += c.vel.z * dt * 0.15;
  1234→      c.mesh.rotation.z -= c.vel.x * dt * 0.15;
  1235→      const ground = bedAt(c.mesh.position.x, c.mesh.position.z) + 0.25;
  1236→      if (c.mesh.position.y < ground) {
  1237→        c.mesh.position.y = ground;
  1238→        c.vel.y *= -0.35;
  1239→        c.vel.x *= 0.6;
  1240→        c.vel.z *= 0.6;
  1241→        if (c.vel.lengthSq() < 0.4) c.active = false;
  1242→      }
  1243→      if (c.mesh.position.x > LX - 1) c.active = false;
  1244→    }
  1245→
  1246→    // keep the algal stain band near the waterline
  1247→    this.dam.stainBand.position.y = this.lastLevel - 0.6;
  1248→  }

  1250→  // ================================================================ stats
  1251→  private computeStats(): DamStats {
  1252→    const d = this.field.data;
  1253→    const dzC = LZ / DOWN_H;
  1254→    const gauges: GaugeStat[] = [];
  1255→    let vmax = 0;
  1256→    let level = 0;
  1257→    let floodedCells = 0;
  1258→    for (let i = 0; i < DOWN_W * DOWN_H; i++) {
  1259→      const k = i * 4;
  1260→      const sp = Math.sqrt(d[k + 1] * d[k + 1] + d[k + 2] * d[k + 2]);
  1261→      vmax = Math.max(vmax, sp);
  1262→      const col = i % DOWN_W;
  1263→      // reservoir pool level: deepest wet cells (depth > 0.5 m) so shelf puddles
  1264→      // cannot distort the stage reading
  1265→      if (col < 26 && d[k + 3] > 0.5) level = Math.max(level, d[k]);
  1266→      if (d[k + 3] > 0.12) floodedCells++;
  1267→    }
  1268→    // gauges
  1269→    for (let gi = 0; gi < GAUGES.length; gi++) {
  1270→      const g = GAUGES[gi];
  1271→      const gx = clamp(Math.round((g.x / LX) * DOWN_W), 1, DOWN_W - 2);
  1272→      let q = 0;
  1273→      let stage = 0;
  1274→      let gv = 0;
  1275→      let froude = 0;
  1276→      for (let j = 0; j < DOWN_H; j++) {
  1277→        const k = (j * DOWN_W + gx) * 4;
  1278→        const h = Math.max(d[k + 3], 0);
  1279→        const u = d[k + 1];
  1280→        const v = d[k + 2];
  1281→        q += u * h * dzC;
  1282→        stage = Math.max(stage, h);
  1283→        gv = Math.max(gv, Math.sqrt(u * u + v * v));
  1284→        if (h > 0.05) froude = Math.max(froude, Math.abs(u) / Math.sqrt(this.gravity * h));
  1285→      }
  1286→      q = Math.max(q, 0);
  1287→      if (q > this.gaugePeaks[gi]) this.gaugePeaks[gi] = q;
  1288→      if (this.gaugeArr[gi] === null && stage > 0.2 && g.x > DAM_X) this.gaugeArr[gi] = this.simTime;
  1289→      gauges.push({
  1290→        id: g.id,
  1291→        label: g.label,
  1292→        q,
  1293→        qPeak: this.gaugePeaks[gi],
  1294→        depth: stage,
  1295→        vel: gv,
  1296→        froude,
  1297→        arrivalMin: this.gaugeArr[gi] !== null ? this.gaugeArr[gi]! * this.damProfile.timeMinPerSec : null,
  1298→        distKm: (g.x - DAM_X) * this.damProfile.lenScale / 1000,
  1299→      });
  1300→    }
  1301→    const frG = gauges[0] ? gauges[0].q : 0;
  1302→    if (frG > this.qPeak) this.qPeak = frG;
  1303→
  1304→    // level rate of rise (demo m/s, ema over stats ticks)
  1305→    const dtSimS = this.simTime - this.lastStatSimT;
  1306→    const riseMs = dtSimS > 0.05 && this.lastStatSimT > 0
  1307→      ? Math.max((level - this.lastLevel) / dtSimS, -2)
  1308→      : 0;
  1309→    this.levelRise = this.levelRise * 0.7 + riseMs * 0.3;
  1310→    this.lastStatSimT = this.simTime;
  1311→    this.lastLevel = level;
  1312→
  1313→    // scenario progress + stage
  1314→    let progress01 = 0;
  1315→    let stage = 0;
  1316→    if (this.scenario) {
  1317→      const tau = this.breachTau || 15;
  1318→      const breachDone = this.breachDepth01;
  1319→      const floodT = clamp((this.scenT - tau) / 55, 0, 1);
  1320→      progress01 = clamp(0.12 * clamp(this.scenT / 4, 0, 1) + 0.38 * breachDone + 0.42 * floodT + 0.08 * clamp(floodT * 1.2, 0, 1), 0, 1);
  1321→      stage = this.scenT < 2 ? 1 : breachDone <= 0 ? 2 : breachDone < 0.98 ? 3 : floodT < 1 ? 4 : 5;
  1322→    }
  1323→
  1324→    return {
  1325→      t: this.simTime,
  1326→      realMin: this.simTime * this.damProfile.timeMinPerSec,
  1327→      level,
  1328→      levelFrac: simLevelToFrac(level),
  1329→      riseMPerHr: this.levelRise * (60 / this.damProfile.timeMinPerSec),
  1330→      vmax,
  1331→      froude: gauges[0]?.froude ?? 0,
  1332→      fps: this.fpsEma,
  1333→      phase: this.scrubbing ? 'scrub' : this.scenario ? 'scenario' : 'live',
  1334→      scenarioActive: !!this.scenario,
  1335→      breach01: this.breachDepth01,
  1336→      progress01,
  1337→      stage,
  1338→      gauges,
  1339→      floodedCells,
  1340→      qOut: gauges[0]?.q ?? 0,
  1341→    };
  1342→  }
