    18→function hashRnd(i: number): number {
    19→  const h = Math.sin(i * 127.1 + 311.7) * 43758.5453;
    20→  return h - Math.floor(h);
    21→}

    23→function canvasTex(w: number, h: number, draw: (c: CanvasRenderingContext2D) => void): THREE.Texture {
    24→  const cv = document.createElement('canvas');
    25→  cv.width = w;
    26→  cv.height = h;
    27→  const ctx = cv.getContext('2d')!;
    28→  draw(ctx);
    29→  const tex = new THREE.CanvasTexture(cv);
    30→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    31→  tex.colorSpace = THREE.SRGBColorSpace;
    32→  tex.anisotropy = 4;
    33→  return tex;
    34→}

    56→function makePylon(h: number): THREE.Group {
    57→  const g = new THREE.Group();
    58→  const mat = matGalvanised();
    59→  const base = 1.5;
    60→  const top = 0.45;
    61→  // 4 splayed legs
    62→  for (const sx of [-1, 1]) {
    63→    for (const sz of [-1, 1]) {
    64→      const leg = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.09, h, 5), mat);
    65→      const lx = (sx * (base + (top - base) * 0.5)) / 2;
    66→      const lz = (sz * (base + (top - base) * 0.5)) / 2;
    67→      leg.position.set(lx, h / 2, lz);
    68→      leg.rotation.z = -sx * 0.115;
    69→      leg.rotation.x = sz * 0.115;
    70→      leg.castShadow = true;
    71→      g.add(leg);
    72→    }
    73→  }
    74→  // cross-braces on both faces (2 levels)
    75→  for (const lvl of [0.3, 0.62]) {
    76→    for (const sz of [-1, 1]) {
    77→      const brace = new THREE.Mesh(new THREE.BoxGeometry(base * (1 - lvl * 0.55), 0.06, 0.06), mat);
    78→      brace.position.set(0, h * lvl, (sz * base * (1 - lvl * 0.55)) / 2);
    79→      brace.rotation.z = sz * 0.5;
    80→      g.add(brace);
    81→    }
    82→  }
    83→  // cross-arms (two levels) + peak
    84→  const arm1 = new THREE.Mesh(new THREE.BoxGeometry(3.6, 0.09, 0.09), mat);
    85→  arm1.position.y = h * 0.8;
    86→  const arm2 = new THREE.Mesh(new THREE.BoxGeometry(2.7, 0.09, 0.09), mat);
    87→  arm2.position.y = h * 0.93;
    88→  const peak = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.05, h * 0.1, 5), mat);
    89→  peak.position.y = h * 1.04;
    90→  arm1.castShadow = arm2.castShadow = true;
    91→  g.add(arm1, arm2, peak);
    92→  return g;
    93→}

    95→function makeCable(a: THREE.Vector3, b: THREE.Vector3, sag: number, mat: THREE.Material): THREE.Mesh {
    96→  const mid = a.clone().add(b).multiplyScalar(0.5);
    97→  mid.y -= sag;
    98→  const curve = new THREE.QuadraticBezierCurve3(a, mid, b);
    99→  const geo = new THREE.TubeGeometry(curve, 20, 0.035, 5, false);
   100→  const m = new THREE.Mesh(geo, mat);
   101→  m.castShadow = false;
   102→  return m;
   103→}

   437→function distToPaths(x: number, z: number): number {
   438→  let best = 1e9;
   439→  for (const s of CLEAR_SEGS) {
   440→    const dx = s.bx - s.ax, dz = s.bz - s.az;
   441→    const t = clamp(((x - s.ax) * dx + (z - s.az) * dz) / (dx * dx + dz * dz || 1), 0, 1);
   442→    const d = Math.hypot(x - (s.ax + dx * t), z - (s.az + dz * t));
   443→    if (d < best) best = d;
   444→  }
   445→  return best;
   446→}

   471→// deck ribbon following an arbitrary elevation profile; returns the sampled
   472→// centreline so callers can hang piers / dashes off it
   473→function profileRibbon(
   474→  group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material,
   475→  yAt: (x: number, z: number, t: number) => number,
   476→  castShadow = true,
   477→): { x: number; z: number; y: number }[] {
   478→  const n = Math.max(10, pts.length * 5);
   479→  const P: { x: number; z: number; y: number }[] = [];
   480→  for (let k = 0; k <= n; k++) {
   481→    const t = k / n;
   482→    const seg = Math.min(Math.floor(t * (pts.length - 1)), pts.length - 2);
   483→    const lt = t * (pts.length - 1) - seg;
   484→    const x = pts[seg][0] + (pts[seg + 1][0] - pts[seg][0]) * lt;
   485→    const z = pts[seg][1] + (pts[seg + 1][1] - pts[seg][1]) * lt;
   486→    P.push({ x, z, y: yAt(x, z, t) });
   487→  }
   488→  // smooth the elevation profile (endpoints pinned)
   489→  for (let p = 0; p < 4; p++)
   490→    for (let i = 1; i < P.length - 1; i++)
   491→      P[i].y = (P[i - 1].y + 2 * P[i].y + P[i + 1].y) / 4;
   492→  const pos: number[] = [];
   493→  const idx: number[] = [];
   494→  for (let i = 0; i < P.length; i++) {
   495→    const a = P[Math.max(i - 1, 0)];
   496→    const b = P[Math.min(i + 1, P.length - 1)];
   497→    const dx = b.x - a.x, dz = b.z - a.z;
   498→    const dl = Math.hypot(dx, dz) || 1;
   499→    const nx = (-dz / dl) * w * 0.5, nz = (dx / dl) * w * 0.5;
   500→    pos.push(P[i].x - nx, P[i].y, P[i].z - nz, P[i].x + nx, P[i].y, P[i].z + nz);
   501→    if (i > 0) {
   502→      const q = (i - 1) * 2;
   503→      idx.push(q, q + 2, q + 1, q + 1, q + 2, q + 3);
   504→    }
   505→  }
   506→  const geo = new THREE.BufferGeometry();
   507→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   508→  geo.setIndex(idx);
   509→  geo.computeVertexNormals();
   510→  const mesh = new THREE.Mesh(geo, mat);
   511→  mesh.castShadow = castShadow;
   512→  mesh.receiveShadow = true;
   513→  group.add(mesh);
   514→  return P;
   515→}

   517→function deckPiers(
   518→  group: THREE.Group, P: { x: number; z: number; y: number }[],
   519→  r: number, mat: THREE.Material, minH = 1.4,

   527→    pier.castShadow = true;
   528→    group.add(pier);
   529→  }

   532→// elevated expressway: deck on pylons following ground + hAbove, edge beams,
   533→// centre dashes, plus sloped ramps down to street level
   534→function buildExpressway(group: THREE.Group): void {
   535→  const deckMat = new THREE.MeshStandardMaterial({ color: 0x58585e, roughness: 0.88, metalness: 0.04, side: THREE.DoubleSide });
   536→  const beamMat = matConcreteDark();
   537→  const lineMat = new THREE.MeshStandardMaterial({ color: 0xd8d2b8, roughness: 0.8 });
   538→
   539→  const road = (pts: [number, number][], h0: number, h1: number, w: number): void => {
   540→    const P = profileRibbon(group, pts, w, deckMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t);
   541→    // edge beams slightly under the deck
   542→    profileRibbon(group, pts, w + 0.7, beamMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t - 0.16, false);
   543→    // pylons where the deck is well above ground (never in the river channel)
   544→    for (let i = 3; i < P.length - 2; i += 4) {
   545→      const g = bedAt(P[i].x, P[i].z);
   546→      const hh = P[i].y - g;
   547→      if (hh < 1.8 || Math.abs(P[i].z - axisAt(P[i].x)) < 14) continue;
   548→      const pylon = new THREE.Mesh(new THREE.BoxGeometry(0.72, hh - 0.1, 0.72), beamMat);
   549→      pylon.position.set(P[i].x, g + (hh - 0.1) / 2, P[i].z);
   550→      pylon.castShadow = true;
   551→      group.add(pylon);
   552→      const cap = new THREE.Mesh(new THREE.BoxGeometry(w + 0.5, 0.3, 1.0), beamMat);
   553→      cap.position.set(P[i].x, P[i].y - 0.3, P[i].z);
   554→      group.add(cap);
   555→    }
   556→    // centre dashes (instanced)
   557→    const dashT: THREE.Matrix4[] = [];
   558→    for (let i = 2; i < P.length - 2; i += 3) {
   559→      const a = P[i], b = P[i + 1];
   560→      const yaw = Math.atan2(b.x - a.x, b.z - a.z);
   561→      const mm = new THREE.Matrix4();
   562→      mm.makeRotationY(yaw);
   563→      mm.setPosition(a.x, a.y + 0.04, a.z);
   564→      dashT.push(mm);
   565→    }
   566→    if (dashT.length) {
   567→      const dashMesh = new THREE.InstancedMesh(new THREE.BoxGeometry(0.16, 0.04, 1.7), lineMat, dashT.length);
   568→      dashT.forEach((mm, i) => dashMesh.setMatrixAt(i, mm));
   569→      group.add(dashMesh);
   570→    }
   571→  };
   572→
   573→  road(EXP_A, 5.4, 5.4, 5.0);    // corridor expressway (north bench, full length)
   574→  road(RAMP_A1, 5.4, 0.16, 3.4); // west down-ramp to the dam abutment bench
   575→  road(RAMP_A2, 5.4, 0.16, 3.4); // east exit ramp toward the rim villages
   576→  road(RAMP_A3, 5.4, 0.16, 3.0); // spur down to the north-bench service road
   577→}

   579→function flatRibbon(group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material, lift = 0.14): void {
   580→  const pos: number[] = [];
   581→  const idx: number[] = [];
   582→  for (let i = 0; i < pts.length - 1; i++) {
   583→    for (let s = 0; s < 3; s++) {
   584→      const t0 = s / 3, t1 = (s + 1) / 3;
   585→      const x0 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t0;
   586→      const z0 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t0;
   587→      const x1 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t1;
   588→      const z1 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t1;
   589→      const dx = z1 - z0, dz = -(x1 - x0);
   590→      const dl = Math.hypot(dx, dz) || 1;
   591→      const nx = (dx / dl) * w * 0.5, nz = (dz / dl) * w * 0.5;
   592→      const a = bedAt(x0 - nx, z0 - nz) + lift;
   593→      const b = bedAt(x0 + nx, z0 + nz) + lift;
   594→      const c = bedAt(x1 - nx, z1 - nz) + lift;
   595→      const d = bedAt(x1 + nx, z1 + nz) + lift;
   596→      const base = pos.length / 3;
   597→      pos.push(x0 - nx, a, z0 - nz, x0 + nx, b, z0 + nz, x1 - nx, c, z1 - nz, x1 + nx, d, z1 + nz);
   598→      idx.push(base, base + 2, base + 1, base + 1, base + 2, base + 3);
   599→    }
   600→  }
   601→  const geo = new THREE.BufferGeometry();
   602→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   603→  geo.setIndex(idx);
   604→  geo.computeVertexNormals();
   605→  const mesh = new THREE.Mesh(geo, mat);
   606→  mesh.receiveShadow = true;
   607→  group.add(mesh);
   608→}

  1467→    const pump = new THREE.Mesh(new THREE.BoxGeometry(1.6, 1.5, 1.5), matGalvanised());
  1468→    pump.position.set(fx + 2.2, g + 0.75, fz);
  1469→    pump.castShadow = true;

  1471→  }
  1472→
  1473→  // substation — gravel pad, transformers, gantry (east district)
  1474→  {
  1475→    const vx = 136.2, vz = -2.9;
  1476→    const g = bedAt(vx, vz);
  1477→    const pad = new THREE.Mesh(new THREE.BoxGeometry(4.8, 0.14, 5.6), new THREE.MeshStandardMaterial({ color: 0x9a968c, roughness: 0.98 }));
  1478→    pad.position.set(vx, g + 0.07, vz);
  1479→    pad.receiveShadow = true;
  1480→    for (const [tx, tz] of [[vx - 1.1, vz - 1.2], [vx - 1.1, vz + 1.2]] as const) {
  1481→      const xfmr = new THREE.Mesh(new THREE.BoxGeometry(1.5, 1.3, 1.1), matDark());
  1482→      xfmr.position.set(tx, g + 0.8, tz);

  1485→    }
  1486→    for (const gz2 of [vz - 2.2, vz + 2.2]) {
  1487→      const post = new THREE.Mesh(new THREE.CylinderGeometry(0.08, 0.1, 4.4, 6), matGalvanised());
  1488→      post.position.set(vx + 1.6, g + 2.2, gz2);

  1492→    const beam = new THREE.Mesh(new THREE.BoxGeometry(0.1, 0.12, 4.6), matGalvanised());

  1496→
  1497→  // civic hall — columned portico + pediment (south civic band)
  1498→  {
  1499→    const cx = 127, cz = 26.4;
  1500→    const g = bedAt(cx, cz);
  1501→    const hall = new THREE.Mesh(new THREE.BoxGeometry(5.2, 3.4, 3.6), matWhite());
  1502→    hall.position.set(cx, g + 1.7, cz);
  1503→    hall.castShadow = hall.receiveShadow = true;
  1504→    const ped = new THREE.Mesh(new THREE.CylinderGeometry(0.02, 1.6, 1.1, 3), matWhite());
  1505→    ped.rotation.x = Math.PI / 2;
  1506→    ped.rotation.y = Math.PI / 2;
  1507→    ped.position.set(cx, g + 4.0, cz - 1.95);
  1508→    ped.castShadow = true;
  1509→    group.add(hall, ped);
  1510→    for (let k = 0; k < 4; k++) {
  1511→      const column = new THREE.Mesh(new THREE.CylinderGeometry(0.16, 0.18, 2.6, 8), matWhite());
  1512→      column.position.set(cx - 1.8 + k * 1.2, g + 1.3, cz - 2.05);

  1586→// ============================================================ FAR TERRAIN
  1587→// The solver domain is a hard-edged plane; this dresses the world beyond it so
  1588→// the square canvas never shows its cut: the downstream valley keeps running
  1589→// east past the exit gorge (the flood river visibly continues toward the
  1590→// horizon), and a rounded forested mountain ring closes the view on all four
  1591→// sides. Heights blend out of bedAt() so the seam against the solver mesh is
  1592→// invisible, and colour comes from the same terrainColor painter.
  1593→export function buildFarTerrain(): { group: THREE.Group } {
  1594→  const group = new THREE.Group();
  1595→
  1596→  const sstep = (a: number, b: number, v: number): number => {
  1597→    const t = clamp((v - a) / (b - a), 0, 1);
  1598→    return t * t * (3 - 2 * t);
  1599→  };
  1600→
  1601→  // distant backdrop: forested mountain ring — tall enough to frame the
  1602→  // valley like the reference map (green wooded slopes, tan high tops)
  1603→  const ringH = (x: number, z: number): number =>
  1604→    12 + 14 * fbm(x * 0.021 + 40.7, z * 0.021 - 13.3, 4)
  1605→     + 3.5 * fbm(x * 0.065 - 8.1, z * 0.065 + 21.4, 3);
  1606→
  1607→  const farH = (x: number, z: number): number => {
  1608→    // east corridor: bedAt stays well-behaved east of the canvas, so the
  1609→    // valley (channel + widening walls) keeps running out toward the horizon
  1610→    // and the range closes over it further downstream
  1611→    const dzo = z - axisAt(LX);
  1612→    if (x > LX && Math.abs(dzo) < 46) {
  1613→      const valley = bedAt(x, z);
  1614→      const t = Math.max(sstep(175, 255, x), sstep(34, 46, Math.abs(dzo)));
  1615→      return valley + (ringH(x, z) + 2 - valley) * t;
  1616→    }
  1617→    // everywhere else: hold the boundary profile, fade into the range
  1618→    const dOut = Math.max(-x, x - LX, -LZ / 2 - z, z - LZ / 2, 0);
  1619→    const base = bedAt(clamp(x, 1.5, LX - 1.5), clamp(z, -LZ / 2 + 1, LZ / 2 - 1));
  1620→    const t = sstep(2, 70, dOut);
  1621→    return base + (ringH(x, z) - base) * t;
  1622→  };
  1623→
  1624→  const mat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.97, metalness: 0, envMapIntensity: 0.18 });
  1625→  const tc = { r: 0, g: 0, b: 0 };
  1626→  const col = new THREE.Color();
  1627→
  1628→  const strip = (x0: number, x1: number, z0: number, z1: number, nx: number, nz: number): void => {
  1629→    const geo = new THREE.PlaneGeometry(x1 - x0, z1 - z0, nx, nz);
  1630→    geo.rotateX(-Math.PI / 2);
  1631→    geo.translate((x0 + x1) / 2, 0, (z0 + z1) / 2);
  1632→    const pos = geo.attributes.position as THREE.BufferAttribute;
  1633→    const colors = new Float32Array(pos.count * 3);
  1634→    for (let i = 0; i < pos.count; i++) {
  1635→      const x = pos.getX(i);
  1636→      const z = pos.getZ(i);
  1637→      const y = farH(x, z);
  1638→      pos.setY(i, y);
  1639→      const e = 3;
  1640→      const sx = (farH(x + e, z) - farH(x - e, z)) / (2 * e);
  1641→      const sz = (farH(x, z + e) - farH(x, z - e)) / (2 * e);
  1642→      terrainColor(x, z, y, Math.sqrt(sx * sx + sz * sz), tc);
  1643→      // keep the far ring lush: only a whisper of aerial haze so the backdrop
  1644→      // reads as forested mountains, not a grey wall
  1645→      const gr = (tc.r + tc.g + tc.b) / 3;
  1646→      const mu = 0.14;
  1647→      tc.r = tc.r * (1 - mu) + (gr * 0.78 + 0.1) * mu;
  1648→      tc.g = tc.g * (1 - mu) + (gr * 0.86 + 0.12) * mu;
  1649→      tc.b = tc.b * (1 - mu) + (gr * 0.84 + 0.14) * mu;
  1650→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
  1651→      colors[i * 3] = col.r;
  1652→      colors[i * 3 + 1] = col.g;
  1653→      colors[i * 3 + 2] = col.b;
  1654→    }
  1655→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
  1656→    geo.computeVertexNormals();
  1657→    group.add(new THREE.Mesh(geo, mat));
  1658→  };
  1659→
  1660→  // four aprons framing the canvas: north / south rims, west headwall, east
  1661→  // valley continuation (the east strip carries the river out of frame)
  1662→  strip(-200, 340, -210, -LZ / 2, 128, 30);
  1663→  strip(-200, 340, LZ / 2, 210, 128, 30);
  1664→  strip(-200, 0, -LZ / 2, LZ / 2, 46, 64);
  1665→  strip(LX, 340, -LZ / 2, LZ / 2, 50, 64);
  1666→
  1667→  // haze backstop far below the terrain so no sky peeks under the outer ranges
  1668→  const back = new THREE.Mesh(
  1669→    new THREE.CircleGeometry(2600, 40),
  1670→    new THREE.MeshBasicMaterial({ color: 0xa8bcc8 }),
  1671→  );
  1672→  back.rotation.x = -Math.PI / 2;
  1673→  back.position.set(LX / 2, 0.4, 0);
  1674→  group.add(back);
  1675→
  1676→  return { group };
  1677→}
