    14→function hashRnd(i: number): number {
    15→  const h = Math.sin(i * 127.1 + 311.7) * 43758.5453;
    16→  return h - Math.floor(h);
    17→}

    19→function canvasTex(w: number, h: number, draw: (c: CanvasRenderingContext2D) => void): THREE.Texture {
    20→  const cv = document.createElement('canvas');
    21→  cv.width = w;
    22→  cv.height = h;
    23→  const ctx = cv.getContext('2d')!;
    24→  draw(ctx);
    25→  const tex = new THREE.CanvasTexture(cv);
    26→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    27→  tex.colorSpace = THREE.SRGBColorSpace;
    28→  tex.anisotropy = 4;
    29→  return tex;
    30→}

    52→function makePylon(h: number): THREE.Group {
    53→  const g = new THREE.Group();
    54→  const mat = matGalvanised();
    55→  const base = 1.5;
    56→  const top = 0.45;
    57→  // 4 splayed legs
    58→  for (const sx of [-1, 1]) {
    59→    for (const sz of [-1, 1]) {
    60→      const leg = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.09, h, 5), mat);
    61→      const lx = (sx * (base + (top - base) * 0.5)) / 2;
    62→      const lz = (sz * (base + (top - base) * 0.5)) / 2;
    63→      leg.position.set(lx, h / 2, lz);
    64→      leg.rotation.z = -sx * 0.115;
    65→      leg.rotation.x = sz * 0.115;
    66→      leg.castShadow = true;
    67→      g.add(leg);
    68→    }
    69→  }
    70→  // cross-braces on both faces (2 levels)
    71→  for (const lvl of [0.3, 0.62]) {
    72→    for (const sz of [-1, 1]) {
    73→      const brace = new THREE.Mesh(new THREE.BoxGeometry(base * (1 - lvl * 0.55), 0.06, 0.06), mat);
    74→      brace.position.set(0, h * lvl, (sz * base * (1 - lvl * 0.55)) / 2);
    75→      brace.rotation.z = sz * 0.5;
    76→      g.add(brace);
    77→    }
    78→  }
    79→  // cross-arms (two levels) + peak
    80→  const arm1 = new THREE.Mesh(new THREE.BoxGeometry(3.6, 0.09, 0.09), mat);
    81→  arm1.position.y = h * 0.8;
    82→  const arm2 = new THREE.Mesh(new THREE.BoxGeometry(2.7, 0.09, 0.09), mat);
    83→  arm2.position.y = h * 0.93;
    84→  const peak = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.05, h * 0.1, 5), mat);
    85→  peak.position.y = h * 1.04;
    86→  arm1.castShadow = arm2.castShadow = true;
    87→  g.add(arm1, arm2, peak);
    88→  return g;
    89→}

    91→function makeCable(a: THREE.Vector3, b: THREE.Vector3, sag: number, mat: THREE.Material): THREE.Mesh {
    92→  const mid = a.clone().add(b).multiplyScalar(0.5);
    93→  mid.y -= sag;
    94→  const curve = new THREE.QuadraticBezierCurve3(a, mid, b);
    95→  const geo = new THREE.TubeGeometry(curve, 20, 0.035, 5, false);
    96→  const m = new THREE.Mesh(geo, mat);
    97→  m.castShadow = false;
    98→  return m;
    99→}

   395→function arcPts(r: number, half: number, n: number): [number, number][] {
   396→  const pts: [number, number][] = [];
   397→  for (let i = 0; i <= n; i++) {
   398→    const a = -half + (2 * half * i) / n;
   399→    pts.push([TOWN_C[0] + r * Math.cos(a), TOWN_C[1] + r * Math.sin(a)]);
   400→  }
   401→  return pts;
   402→}
   403→function rayPts(a: number, r0: number, r1: number, n: number): [number, number][] {
   404→  const pts: [number, number][] = [];
   405→  for (let i = 0; i <= n; i++) {
   406→    const r = r0 + ((r1 - r0) * i) / n;

   417→    CLEAR_SEGS.push({ ax: pts[i][0], az: pts[i][1], bx: pts[i + 1][0], bz: pts[i + 1][1] });

   445→function distToPaths(x: number, z: number): number {
   446→  let best = 1e9;
   447→  for (const s of CLEAR_SEGS) {
   448→    const dx = s.bx - s.ax, dz = s.bz - s.az;
   449→    const t = clamp(((x - s.ax) * dx + (z - s.az) * dz) / (dx * dx + dz * dz || 1), 0, 1);
   450→    const d = Math.hypot(x - (s.ax + dx * t), z - (s.az + dz * t));
   451→    if (d < best) best = d;
   452→  }
   453→  return best;
   454→}

   464→function clearOfSites(x: number, z: number, m = 1.4): boolean {
   465→  for (const [cx, cz, w, d] of CLEAR_RECTS)
   466→    if (Math.abs(x - cx) < w / 2 + m && Math.abs(z - cz) < d / 2 + m) return false;
   467→  return true;
   468→}

   470→// deck ribbon following an arbitrary elevation profile; returns the sampled
   471→// centreline so callers can hang piers / dashes off it
   472→function profileRibbon(
   473→  group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material,
   474→  yAt: (x: number, z: number, t: number) => number,
   475→  castShadow = true,
   476→): { x: number; z: number; y: number }[] {
   477→  const n = Math.max(10, pts.length * 5);
   478→  const P: { x: number; z: number; y: number }[] = [];
   479→  for (let k = 0; k <= n; k++) {
   480→    const t = k / n;
   481→    const seg = Math.min(Math.floor(t * (pts.length - 1)), pts.length - 2);
   482→    const lt = t * (pts.length - 1) - seg;
   483→    const x = pts[seg][0] + (pts[seg + 1][0] - pts[seg][0]) * lt;
   484→    const z = pts[seg][1] + (pts[seg + 1][1] - pts[seg][1]) * lt;
   485→    P.push({ x, z, y: yAt(x, z, t) });
   486→  }
   487→  // smooth the elevation profile (endpoints pinned)
   488→  for (let p = 0; p < 4; p++)
   489→    for (let i = 1; i < P.length - 1; i++)
   490→      P[i].y = (P[i - 1].y + 2 * P[i].y + P[i + 1].y) / 4;
   491→  const pos: number[] = [];
   492→  const idx: number[] = [];
   493→  for (let i = 0; i < P.length; i++) {
   494→    const a = P[Math.max(i - 1, 0)];
   495→    const b = P[Math.min(i + 1, P.length - 1)];
   496→    const dx = b.x - a.x, dz = b.z - a.z;
   497→    const dl = Math.hypot(dx, dz) || 1;
   498→    const nx = (-dz / dl) * w * 0.5, nz = (dx / dl) * w * 0.5;
   499→    pos.push(P[i].x - nx, P[i].y, P[i].z - nz, P[i].x + nx, P[i].y, P[i].z + nz);
   500→    if (i > 0) {
   501→      const q = (i - 1) * 2;
   502→      idx.push(q, q + 2, q + 1, q + 1, q + 2, q + 3);
   503→    }
   504→  }
   505→  const geo = new THREE.BufferGeometry();
   506→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   507→  geo.setIndex(idx);
   508→  geo.computeVertexNormals();
   509→  const mesh = new THREE.Mesh(geo, mat);
   510→  mesh.castShadow = castShadow;
   511→  mesh.receiveShadow = true;
   512→  group.add(mesh);
   513→  return P;
   514→}

   516→function deckPiers(
   517→  group: THREE.Group, P: { x: number; z: number; y: number }[],
   518→  r: number, mat: THREE.Material, minH = 1.4,
   519→): void {
   520→  for (let i = 3; i < P.length - 2; i += 4) {
   521→    const g = bedAt(P[i].x, P[i].z);
   522→    const hh = P[i].y - g;
   523→    if (hh < minH) continue;
   524→    const pier = new THREE.Mesh(new THREE.CylinderGeometry(r * 0.78, r, hh - 0.12, 8), mat);
   525→    pier.position.set(P[i].x, g + (hh - 0.12) / 2, P[i].z);
   526→    pier.castShadow = true;
   527→    group.add(pier);
   528→  }
   529→}

   531→// elevated expressway: deck on pylons following ground + hAbove, edge beams,
   532→// centre dashes, plus sloped ramps down to street level
   533→function buildExpressway(group: THREE.Group): void {
   534→  const deckMat = new THREE.MeshStandardMaterial({ color: 0x58585e, roughness: 0.88, metalness: 0.04, side: THREE.DoubleSide });
   535→  const beamMat = matConcreteDark();
   536→  const lineMat = new THREE.MeshStandardMaterial({ color: 0xd8d2b8, roughness: 0.8 });
   537→
   538→  const road = (pts: [number, number][], h0: number, h1: number, w: number): void => {
   539→    const P = profileRibbon(group, pts, w, deckMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t);
   540→    // edge beams slightly under the deck
   541→    profileRibbon(group, pts, w + 0.7, beamMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t - 0.16, false);
   542→    // pylons where the deck is well above ground (never in the river channel)
   543→    for (let i = 3; i < P.length - 2; i += 4) {
   544→      const g = bedAt(P[i].x, P[i].z);
   545→      const hh = P[i].y - g;
   546→      if (hh < 1.8 || Math.abs(P[i].z) < 13) continue;
   547→      const pylon = new THREE.Mesh(new THREE.BoxGeometry(0.72, hh - 0.1, 0.72), beamMat);
   548→      pylon.position.set(P[i].x, g + (hh - 0.1) / 2, P[i].z);
   549→      pylon.castShadow = true;
   550→      group.add(pylon);
   551→      const cap = new THREE.Mesh(new THREE.BoxGeometry(w + 0.5, 0.3, 1.0), beamMat);
   552→      cap.position.set(P[i].x, P[i].y - 0.3, P[i].z);
   553→      group.add(cap);
   554→    }
   555→    // centre dashes (instanced)
   556→    const dashT: THREE.Matrix4[] = [];
   557→    for (let i = 2; i < P.length - 2; i += 3) {
   558→      const a = P[i], b = P[i + 1];
   559→      const yaw = Math.atan2(b.x - a.x, b.z - a.z);
   560→      const mm = new THREE.Matrix4();
   561→      mm.makeRotationY(yaw);
   562→      mm.setPosition(a.x, a.y + 0.04, a.z);
   563→      dashT.push(mm);
   564→    }
   565→    if (dashT.length) {
   566→      const dashMesh = new THREE.InstancedMesh(new THREE.BoxGeometry(0.16, 0.04, 1.7), lineMat, dashT.length);
   567→      dashT.forEach((mm, i) => dashMesh.setMatrixAt(i, mm));
   568→      group.add(dashMesh);
   569→    }
   570→  };
   571→
   572→  road(EXP_A, 5.6, 5.6, 5.0);   // +z bank elevated highway
   573→  road(EXP_B, 5.2, 5.2, 5.0);   // signature river-crossing elevated highway
   574→  road(RAMP_A1, 5.6, 0.16, 3.6); // down-ramp to the outer ring
   575→  road(RAMP_A2, 5.6, 0.16, 3.6); // down-ramp near the dam bench
   576→  road(RAMP_B1, 5.2, 0.16, 3.6); // down-ramp on the left bank
   577→  road(RAMP_AB, 5.6, 5.2, 3.6);  // flyover connector A → B
   578→}

   580→function flatRibbon(group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material, lift = 0.14): void {
   581→  const pos: number[] = [];
   582→  const idx: number[] = [];
   583→  for (let i = 0; i < pts.length - 1; i++) {
   584→    for (let s = 0; s < 3; s++) {
   585→      const t0 = s / 3, t1 = (s + 1) / 3;
   586→      const x0 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t0;
   587→      const z0 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t0;
   588→      const x1 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t1;
   589→      const z1 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t1;
   590→      const dx = z1 - z0, dz = -(x1 - x0);
   591→      const dl = Math.hypot(dx, dz) || 1;
   592→      const nx = (dx / dl) * w * 0.5, nz = (dz / dl) * w * 0.5;
   593→      const a = bedAt(x0 - nx, z0 - nz) + lift;
   594→      const b = bedAt(x0 + nx, z0 + nz) + lift;
   595→      const c = bedAt(x1 - nx, z1 - nz) + lift;
   596→      const d = bedAt(x1 + nx, z1 + nz) + lift;
   597→      const base = pos.length / 3;
   598→      pos.push(x0 - nx, a, z0 - nz, x0 + nx, b, z0 + nz, x1 - nx, c, z1 - nz, x1 + nx, d, z1 + nz);
   599→      idx.push(base, base + 2, base + 1, base + 1, base + 2, base + 3);
   600→    }
   601→  }
   602→  const geo = new THREE.BufferGeometry();
   603→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   604→  geo.setIndex(idx);
   605→  geo.computeVertexNormals();
   606→  const mesh = new THREE.Mesh(geo, mat);
   607→  mesh.receiveShadow = true;
   608→  group.add(mesh);
   609→}

   709→export function buildTown(): { group: THREE.Group } {
   710→  const group = new THREE.Group();
   711→  const asphalt = new THREE.MeshStandardMaterial({ color: 0x45454a, roughness: 0.95 });
   712→  const paving = new THREE.MeshStandardMaterial({ color: 0x9b968c, roughness: 0.95 });
   713→
   714→  // old-town grid streets
   715→  for (const st of TOWN_STREETS) flatRibbon(group, st.pts, st.w, asphalt);
   716→
   717→  // ---- ring boulevards — arched decks over the river channel (bridges);
   718→  // lighter asphalt so the concentric arcs read from the air like the
   719→  // reference map's ring highways
   720→  const ringMat = new THREE.MeshStandardMaterial({ color: 0x585862, roughness: 0.9 });
   721→  RING_PTS.forEach((pts, ri) => {
   722→    const P = profileRibbon(group, pts, RINGS[ri].w, ringMat, (x, z) => {
   723→      const g = bedAt(x, z);
   724→      const arch = Math.max(0, 1 - Math.abs(z) / 14);
   725→      return Math.max(g + 0.16, bedAt(x, 0) + 3.6 + 1.15 * arch * arch);
   726→    });
   727→    deckPiers(group, P, 0.4, matConcreteDark());
   728→  });
   729→
   730→  // radial avenues linking the rings to the old core
   731→  for (const pts of RADIAL_PTS) flatRibbon(group, pts, 2.5, asphalt);
   732→
   733→  // riverside promenades (trees + lamps line these)
   734→  flatRibbon(group, [[124, 12.4], [140, 12.2], [158, 12.2], [176, 12.4], [190, 12.8]], 1.9, paving);
   735→  flatRibbon(group, [[124, -12.4], [140, -12.3], [158, -12.3], [176, -12.5], [190, -12.9]], 1.9, paving);
   736→
   737→  // ---- elevated expressway + interchange
   738→  buildExpressway(group);
   739→
   740→  // ---- dense districts (instanced, three building classes)
   741→  const bldTex = makeBuildingTex();
   742→  const lowMat = new THREE.MeshStandardMaterial({ map: bldTex, roughness: 0.85, metalness: 0.05 });
   743→  const midMat = new THREE.MeshStandardMaterial({ map: makeTowerTex(false), roughness: 0.68, metalness: 0.1 });
   744→  const glassMat = new THREE.MeshStandardMaterial({ map: makeTowerTex(true), roughness: 0.38, metalness: 0.35, envMapIntensity: 0.35 });
   745→  interface BSpot { x: number; z: number; w: number; d: number; h: number; rot: number; tone: number }
   746→  const spotsL: BSpot[] = [];
   747→  const spotsM: BSpot[] = [];
   748→  const spotsH: BSpot[] = [];
   749→  const tones = [0xd9cdb4, 0xcbb99b, 0xd8d2c4, 0xc2b49a, 0xd3c1a6, 0xbfae94, 0xded6c6, 0xb7a68c];
   750→  const glassTones = [0x8fa4b2, 0xa3b8c4, 0x7f98a8, 0xb0c2cc];
   751→  // deterministic scatter — CONCENTRIC RING DISTRICTS: buildings arrayed in
   752→  // tangentially-oriented arcs around the bowl centre (the reference map's
   753→  // circular city), towers toward the core, terraced low-rise at the rim
   754→  let seed = 1;
   755→  const nextRnd = () => hashRnd(seed++ * 12.9898);
   756→  const RING_BUILD_R = [16, 19.5, 23, 26.5, 30, 33.5, 37, 40.5, 44];
   757→  for (let ri = 0; ri < RING_BUILD_R.length; ri++) {
   758→    const r0 = RING_BUILD_R[ri];
   759→    const arc = 2 * RING_HALF * r0;
   760→    const n = Math.max(8, Math.floor(arc / 4.2));
   761→    for (let i = 0; i < n; i++) {
   762→      const a = -RING_HALF + (2 * RING_HALF * i) / (n - 1) + (nextRnd() - 0.5) * (2.6 / r0);
   763→      const r = r0 + (nextRnd() - 0.5) * 3.0;
   764→      const x = TOWN_C[0] + r * Math.cos(a);
   765→      const z = r * Math.sin(a);
   766→      if (Math.abs(z) < 12.4 || Math.abs(z) > 52) continue; // river corridor
   767→      if (distToPaths(x, z) < 3.0) continue;
   768→      if (!clearOfSites(x, z)) continue;
   769→      const ground = bedAt(x, z);
   770→      if (ground < 3.3 || ground > 16.5) continue;
   771→      const slope = Math.abs(bedAt(x + 2, z) - ground) + Math.abs(bedAt(x, z + 2) - ground);
   772→      if (slope > 2.1) continue;
   773→      if (nextRnd() < 0.13) continue;
   774→      // tangential orientation — street fronts follow the ring curvature
   775→      const rot = a + Math.PI / 2 + (nextRnd() - 0.5) * 0.26;
   776→      // skyline weight — towers cluster on the inner rings near the dam
   777→      const core = clamp(1 - (r - 15) / 30, 0, 1);
   778→      const roll = nextRnd();
   779→      if (core > 0.35 && roll < 0.06 + 0.22 * core) {
   780→        spotsH.push({ x, z, w: 3.9 + nextRnd() * 1.6, d: 3.5 + nextRnd() * 1.5, h: 13.5 + nextRnd() * 9.5, rot, tone: Math.floor(nextRnd() * glassTones.length) });
   781→      } else if (roll < 0.52) {

  1127→// Visual-only continuation of the world beyond the solver domain: the valley
  1128→// runs on, then a broad amphitheater of SMOOTH ROLLING HILLS closes the
  1129→// horizon — the user asked for NO jagged mountains, so every wall is a soft
  1130→// exponential dome modulated by gentle fbm (no ridged-noise crest lines).
  1131→function sstep(a: number, b: number, x: number): number {
  1132→  const t = Math.min(Math.max((x - a) / (b - a), 0), 1);
  1133→  return t * t * (3 - 2 * t);
  1134→}
  1135→function farBed(x: number, z: number): number {
  1136→  const cx = clamp(x, 0.2, LX - 0.2);
  1137→  const cz = clamp(z, -LZ / 2 + 0.2, LZ / 2 - 0.2);
  1138→  let h = bedAt(cx, cz);
  1139→  // distance beyond each solver-domain edge
  1140→  const dUp = Math.max(-x, 0);
  1141→  const dDown = Math.max(x - LX, 0);
  1142→  const dNorth = Math.max(-LZ / 2 - z, 0);
  1143→  const dSouth = Math.max(z - LZ / 2, 0);
  1144→  // rolling amphitheater — each wall eases in exponentially and caps far below
  1145→  // the old jagged 260 m peaks: the north wall (behind the dam) is tallest
  1146→  // (~85 m over the valley), the upstream headwall ~65 m, the downstream east
  1147→  // wall ~75 m, and the south rim stays low (~40 m) so the default aerial
  1148→  // camera still looks over it into the city bowl
  1149→  const rise = Math.max(
  1150→    dUp > 0 ? (1 - Math.exp(-dUp / 55)) * 60 : 0,
  1151→    dDown > 0 ? (dDown > 34 ? (1 - Math.exp(-(dDown - 34) / 60)) * 70 : 0) : 0,
  1152→    dNorth > 0 ? (1 - Math.exp(-dNorth / 48)) * 72 : 0,
  1153→    dSouth > 0 ? (1 - Math.exp(-dSouth / 70)) * 38 : 0,
  1154→  );
  1155→  if (rise > 0) {
  1156→    // smooth fbm only — frequencies kept well below the terrain mesh sampling
  1157→    // rate (anything finer aliases into triangular spikes); the squared massif
  1158→    // mask gives broad dome-like ridges instead of sharp crests
  1159→    const m1 = fbm(x * 0.012 + 7.7, z * 0.012 + 2.9, 3); // broad massifs
  1160→    const m2 = fbm(x * 0.027 + 1.2, z * 0.027 + 8.8, 3); // gentle undulation
  1161→    const hills = 0.52 + 0.58 * m1 * m1; // 0.52..1.10 — rounded domes
  1162→    h += rise * hills * (0.7 + 0.55 * m2);
  1163→    h += (fbm(x * 0.033 + 3.3, z * 0.033 + 6.1, 2) - 0.5) * Math.min(2.2 + rise * 0.05, 5.5);
  1164→  }
  1165→  // Circular continuation of the city-bowl rim: beyond the solver domain the
  1166→  // amphitheater keeps curving around the map (downstream half) instead of
  1167→  // reading as a rectangular valley. Phased in over ~30 m past each domain
  1168→  // edge so the seam stays flush with the domain mesh, and carved along the
  1169→  // river corridor (|z| ≲ 10-26 m) so the waterway runs on to the horizon.
  1170→  const rC = Math.hypot(x - BOWL_CX, z);
  1171→  if (x > 104 && rC > BOWL_R) {
  1172→    const beyond = Math.max(dUp, dDown, dNorth, dSouth);
  1173→    const phase = sstep(1.5, 32, beyond);
  1174→    const gorge = sstep(10, 26, Math.abs(z));
  1175→    const rim = (1 - Math.exp(-(rC - BOWL_R) / 20)) * 30;
  1176→    const m2 = fbm(x * 0.027 + 1.2, z * 0.027 + 8.8, 3);
  1177→    h += rim * gorge * (0.7 + 0.55 * m2) * phase;
  1178→  }
  1179→  return h;
  1180→}

  1182→function farPatch(
  1183→  x0: number, x1: number, z0: number, z1: number, sx: number, sz: number, sink = 0,
  1184→): THREE.Group {
  1185→  const geo = new THREE.PlaneGeometry(x1 - x0, z1 - z0, sx, sz);
  1186→  geo.rotateX(-Math.PI / 2);
  1187→  geo.translate((x0 + x1) / 2, 0, (z0 + z1) / 2);
  1188→  const pos = geo.attributes.position as THREE.BufferAttribute;
  1189→  const colors = new Float32Array(pos.count * 3);
  1190→  const col = new THREE.Color();
  1191→  const tc = { r: 0, g: 0, b: 0 };
  1192→  for (let i = 0; i < pos.count; i++) {
  1193→    const x = pos.getX(i);
  1194→    const z = pos.getZ(i);
  1195→    const y = farBed(x, z) - sink;
  1196→    pos.setY(i, y);
  1197→    const e = 3.2;
  1198→    const sl = Math.hypot(farBed(x + e, z) - farBed(x - e, z), farBed(x, z + e) - farBed(x, z - e)) / (2 * e);
  1199→    terrainColor(x, z, y, sl, tc);
  1200→    col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
  1201→    colors[i * 3] = col.r;
  1202→    colors[i * 3 + 1] = col.g;
  1203→    colors[i * 3 + 2] = col.b;
  1204→  }
  1205→
  1206→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
  1207→  geo.computeVertexNormals();
  1208→
  1209→  // Perimeter skirt: neighbouring patches share edge heights but have different
  1210→  // vertex spacings, so T-junction interpolation opens hairline cracks that
  1211→  // read as white specks from afar. Dropping a terrain-colored ribbon around
  1212→  // every edge hides any gap, whatever the viewing angle.
  1213→  const SKIRT = 7;
  1214→  const gridW = sx + 1;
  1215→  const gridH = sz + 1;
  1216→  const boundary: number[] = [];
  1217→  for (let ix = 0; ix < gridW; ix++) boundary.push(ix); // north row (z0)
  1218→  for (let iz = 1; iz < gridH; iz++) boundary.push(iz * gridW + gridW - 1); // east col
  1219→  for (let ix = gridW - 2; ix >= 0; ix--) boundary.push((gridH - 1) * gridW + ix); // south row
  1220→  for (let iz = gridH - 2; iz >= 1; iz--) boundary.push(iz * gridW); // west col
  1221→  const baseCount = pos.count;
  1222→  const skirtVerts = boundary.length * 2;
  1223→  const posArr = new Float32Array((baseCount + skirtVerts) * 3);
  1224→  posArr.set(pos.array as Float32Array);
  1225→  const colArr = new Float32Array((baseCount + skirtVerts) * 3);
  1226→  colArr.set(colors);
  1227→  for (let k = 0; k < boundary.length; k++) {
  1228→    const b = boundary[k];
  1229→    for (let c = 0; c < 3; c++) {
  1230→      posArr[(baseCount + k * 2) * 3 + c] = posArr[b * 3 + c];
  1231→      colArr[(baseCount + k * 2) * 3 + c] = colArr[b * 3 + c];
  1232→      posArr[(baseCount + k * 2 + 1) * 3 + c] = posArr[b * 3 + c];
  1233→      colArr[(baseCount + k * 2 + 1) * 3 + c] = colArr[b * 3 + c];
  1234→    }
  1235→    posArr[(baseCount + k * 2 + 1) * 3 + 1] -= SKIRT;
  1236→  }
  1237→  const idx = new Uint32Array(boundary.length * 6);
  1238→  for (let k = 0; k < boundary.length; k++) {
  1239→    const k2 = (k + 1) % boundary.length;
  1240→    const tA = baseCount + k * 2, bA = tA + 1, tB = baseCount + k2 * 2, bB = tB + 1;
  1241→    idx[k * 6] = tA; idx[k * 6 + 1] = tB; idx[k * 6 + 2] = bA;
  1242→    idx[k * 6 + 3] = bA; idx[k * 6 + 4] = tB; idx[k * 6 + 5] = bB;
  1243→  }
  1244→  const skirtGeo = new THREE.BufferGeometry();
  1245→  skirtGeo.setAttribute('position', new THREE.BufferAttribute(posArr, 3));
  1246→  skirtGeo.setAttribute('color', new THREE.BufferAttribute(colArr, 3));
  1247→  skirtGeo.setIndex(new THREE.BufferAttribute(idx, 1));
  1248→  skirtGeo.computeVertexNormals();
  1249→
  1250→  const mat = new THREE.MeshStandardMaterial({
  1251→    vertexColors: true, roughness: 0.97, metalness: 0, envMapIntensity: 0.18, side: THREE.DoubleSide,
  1252→  });
  1253→  const top = new THREE.Mesh(geo, mat);
  1254→  const skirt = new THREE.Mesh(skirtGeo, mat);
  1255→  skirt.receiveShadow = false;
  1256→  const group = new THREE.Group();
  1257→  group.add(top, skirt);
  1258→  return group;
  1259→}

  1261→export function buildFarTerrain(): { group: THREE.Group } {
  1262→  const group = new THREE.Group();
  1263→  const E = 260; // how far the ring extends sideways
  1264→  const Z0 = -LZ / 2;
  1265→  const Z1 = LZ / 2;
  1266→  // Seam apron: the domain terrain mesh samples bedAt at CELL CENTERS while a
  1267→  // far patch at the exact edge samples the analytic bed — the half-cell offset
  1268→  // used to open hairline cracks (white streaks) along every domain edge once
  1269→  // the jagged peaks that hid them were removed. Each patch now extends ~2.5 m
  1270→  // INTO the domain, sunk 0.8 m below the bed, so the domain surface always
  1271→  // renders on top of a tucked-under apron — no crack can ever open.
  1272→  const OV = 2.5;
  1273→  const SINK = 0.8;
  1274→  // upstream continuation — rises into the headwall range that hugs the lake
  1275→  // (also covers both far corners)
  1276→  group.add(farPatch(-260, OV, -E, E, 60, 100, SINK));
  1277→  // downstream continuation toward the horizon (also covers both far corners)
  1278→  group.add(farPatch(LX - OV, LX + 260, -E, E, 60, 100, SINK));
  1279→  // side ranges — inset in x so they meet, never overlap, the upstream and
  1280→  // downstream patches
  1281→  group.add(farPatch(OV, LX - OV, Z0 - E, Z0 + OV, 76, 64, SINK));
  1282→  group.add(farPatch(OV, LX - OV, Z1 - OV, Z1 + E, 76, 64, SINK));
  1283→  return { group };
  1284→}
