   191→   726→  // per-frame storm driver: rain streaks, darkened sky/fog, lightning flashes
   192→   727→  private updateWeather(dtReal: number): void {
   193→   728→    const target = this.rainInflow > 0 ? (this.rainHeavy ? 1 : 0.55) : 0;
   194→   729→    this.rainVis += (target - this.rainVis) * Math.min(1, dtReal * 1.4);
   195→   730→    const rv = this.rainVis;
   196→   731→
   197→   732→    if (this.rainMesh) {
   198→   733→      this.rainMesh.visible = rv > 0.02;
   199→   734→      if (rv > 0.02) {
   200→   735→        const p = this.rainPos;
   201→   736→        const n = p.length / 6;
   202→   737→        const fall = 46 * dtReal;
   203→   738→        const drift = 8 * dtReal;
   204→   739→        for (let i = 0; i < n; i++) {
   205→   740→          const k = i * 6;
   206→   741→          p[k] += drift; p[k + 1] -= fall; p[k + 2] += drift * 0.35;
   207→   742→          p[k + 3] = p[k] + 0.26; p[k + 4] = p[k + 1] + 1.5; p[k + 5] = p[k + 2] + 0.09;
   208→   743→          if (p[k + 1] < 2.5) {
   209→   744→            const x = -30 + Math.random() * (LX + 110);
   210→   745→            const y = 48 + Math.random() * 34;
   211→   746→            const z = -LZ / 2 - 40 + Math.random() * (LZ + 80);
   212→   747→            p[k] = x; p[k + 1] = y; p[k + 2] = z;
   213→   748→            p[k + 3] = x + 0.26; p[k + 4] = y + 1.5; p[k + 5] = z + 0.09;
   214→   749→          }
   215→   750→        }
   216→   751→        (this.rainMesh.geometry.attributes.position as THREE.BufferAttribute).needsUpdate = true;
   217→   752→        (this.rainMesh.material as THREE.LineBasicMaterial).opacity = 0.38 * rv;
   218→   753→      }
   219→   754→    }
   220→   755→
   221→   756→    // occasional lightning during heavy rain
   222→   757→    this.flash *= Math.exp(-dtReal * 6.5);
   223→   758→    if (rv > 0.6) {
   224→   759→      this.boltTimer -= dtReal;
   225→   760→      if (this.boltTimer <= 0) {
   226→   761→        this.boltTimer = 5 + Math.random() * 10;
   227→   762→        this.flash = 0.7 + Math.random() * 0.5;
   228→   763→      }
   229→   764→    }
   230→   765→
   231→   766→    // dim / storm the atmosphere
   232→   767→    this.sunLight.intensity = 2.35 * (1 - 0.7 * rv) + this.flash * 5.5;
   233→   768→    this.sunLight.color.setHex(this.flash > 0.25 ? 0xdfe8ff : 0xffd9a8);
   234→   769→    this.hemiLight.intensity = 0.32 * (1 - 0.4 * rv) + this.flash * 1.2;
   235→   770→    this.scene.environmentIntensity = 0.22 * (1 - 0.5 * rv);
   236→   771→    const fog = this.scene.fog as THREE.Fog;
   237→   772→    fog.color.copy(this.fogDay).lerp(this.fogStorm, rv * 0.85);
   238→   773→    fog.near = 470 - 220 * rv;
   239→   774→    fog.far = 2300 - 1150 * rv;
   240→   775→    this.renderer.toneMappingExposure = 0.85 - 0.13 * rv + this.flash * 0.12;
   241→   776→    const su = this.skyU;
   242→   777→    if (su) {
   243→   778→      su.turbidity.value = 4.6 + 5.4 * rv;
   244→   779→      su.rayleigh.value = Math.max(2.2 - 1.6 * rv, 0.3);
   245→   780→      su.mieCoefficient.value = 0.006 + 0.02 * rv;
   246→   781→    }
   247→   782→
   248→   783→    // water shader: choppier ripples + storm fog match
   249→   784→    const wu = this.waterMat.uniforms;
   250→   785→    wu.uRain.value = rv;
   251→   786→    (wu.uFogColor.value as THREE.Color).copy(fog.color);
   252→   787→    wu.uFogNear.value = fog.near;
   253→   788→    wu.uFogFar.value = fog.far;
   254→   789→  }

   349→   884→  // wire up a failing block span: invert elevations + staggered failure delays
   350→   885→  private initBreach(span: BreachSpan): void {
   351→   886→    this.breachSpan = span;
   352→   887→    this.breachInverts = [];
   353→   888→    for (let b = span.start; b < span.start + span.count; b++) {
   354→   889→      const z0 = BLOCK_Z0 + b * BLOCK_W;
   355→   890→      this.breachInverts.push(this.blockInvert(z0 + BLOCK_W / 2));
   356→   891→    }
   357→   892→    this.breachDelays = [];
   358→   893→    for (let k = 0; k < span.count; k++) {
   359→   894→      this.breachDelays.push(Math.random() * this.breachTau * 0.3);
   360→   895→    }

   363→   898→  runScenario(p: ScenarioParams): void {
   364→   899→    if (this.scenario) return;
   365→   900→    this.scenario = p;
   366→   901→    this.phase = 'scenario';
   367→   902→    this.scenT = 0;
   368→   903→    this.snapClock = 0;
   369→   904→    this.overtopT = 0;
   370→   905→    this.rainInflow = p.rain === 'none' ? 0 : p.rain === 'moderate' ? 26 : 64;
   371→   906→    this.rainHeavy = p.rain === 'heavy';
   372→   907→    this.breachSpan = null;
   373→   908→    this.breachT = null;
   374→   909→    this.breachDepth01 = 0;
   375→   910→    this.breachFx = false;
   376→   911→    this.structDirty = true;
   377→   912→
   378→   913→    // reservoir set to the scenario level (drive takes it there quickly)
   379→   914→    const target = clamp(fracToSimLevel(p.levelFrac), 0, 24.4);
   380→   915→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   381→   916→    eu.uDriveEta.value = target;
   382→   917→    eu.uDriveOn.value = 1;
   383→   918→
   384→   919→    if (p.mechanism === 'overtopping') {
   385→   920→      // surge inflow pushes the lake above the crest; erosion follows
   386→   921→      eu.uDriveEta.value = Math.max(target, CREST + 1.0);
   387→   922→    } else {
   388→   923→      // schedule breach after the reservoir reaches level (~2-6 demo s)
   389→   924→      this.breachT = -clamp((target - this.lastLevel) / 1.4, 1.5, 5);
   390→   925→      this.breachTau = Math.max(p.formationMin / this.damProfile.timeMinPerSec, 4) / (p.mechanism === 'piping' ? 0.65 : 1) / this.breachRate;
   391→   926→      this.breachPre = p.mechanism === 'piping' ? 0.22 : 0;
