   222→  // ================================================================ terrain mesh
   223→  private buildTerrainMesh(): void {
   224→    // bed grid on cell centers
   225→    this.bedGrid = new Float32Array(NX * NZ);
   226→    for (let j = 0; j < NZ; j++)
   227→      for (let i = 0; i < NX; i++)
   228→        this.bedGrid[j * NX + i] = bedAt((i + 0.5) * DX, (j + 0.5) * DZ - LZ / 2);
   229→
   230→    const geo = new THREE.PlaneGeometry(LX, LZ, NX - 1, NZ - 1);
   231→    geo.rotateX(-Math.PI / 2);
   232→    geo.translate(LX / 2, 0, 0);
   233→    const pos = geo.attributes.position as THREE.BufferAttribute;
   234→    const colors = new Float32Array(pos.count * 3);
   235→    const col = new THREE.Color();
   236→    const tc = { r: 0, g: 0, b: 0 };
   237→    for (let i = 0; i < pos.count; i++) {
   238→      const x = pos.getX(i);
   239→      const z = pos.getZ(i);
   240→      const y = this.sampleBed(x, z);
   241→      pos.setY(i, y);
   242→      const e = 1.2;
   243→      const sx = (this.sampleBed(x + e, z) - this.sampleBed(x - e, z)) / (2 * e);
   244→      const sz = (this.sampleBed(x, z + e) - this.sampleBed(x, z - e)) / (2 * e);
   245→      const slope = Math.sqrt(sx * sx + sz * sz);
   246→      terrainColor(x, z, y, slope, tc);
   247→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   248→      colors[i * 3] = col.r;
   249→      colors[i * 3 + 1] = col.g;
   250→      colors[i * 3 + 2] = col.b;
   251→    }
   252→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   253→    geo.computeVertexNormals();
   254→    const mesh = new THREE.Mesh(
   255→      geo,
   256→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.96, metalness: 0, envMapIntensity: 0.18 }),
   257→    );
   258→    mesh.receiveShadow = true;
   259→    this.scene.add(mesh);
   260→
   261→    // surround plane so the domain doesn't float in the void
   262→    const surround = new THREE.Mesh(
   263→      new THREE.PlaneGeometry(3000, 3000),
   264→      new THREE.MeshStandardMaterial({ color: 0xa8ad97, roughness: 1, metalness: 0, envMapIntensity: 0.1 }),
   265→    );
   266→    surround.rotation.x = -Math.PI / 2;
   267→    surround.position.set(LX / 2, 1.2, 0);
   268→    this.scene.add(surround);
   269→  }

   271→  private sampleBed(x: number, z: number): number {
   272→    const fu = clamp((x / DX) - 0.5, 0, NX - 1.001);
   273→    const fv = clamp((z + LZ / 2) / DZ - 0.5, 0, NZ - 1.001);
   274→    const i = Math.floor(fu), j = Math.floor(fv);
   275→    const fx = fu - i, fy = fv - j;
   276→    const i1 = Math.min(i + 1, NX - 1), j1 = Math.min(j + 1, NZ - 1);
   277→    const g = this.bedGrid;
   278→    const a = g[j * NX + i] + (g[j * NX + i1] - g[j * NX + i]) * fx;
   279→    const b = g[j1 * NX + i] + (g[j1 * NX + i1] - g[j1 * NX + i]) * fx;
   280→    return a + (b - a) * fy;
   281→  }

   295→  private buildGpuResources(): void {
   296→    const gl = this.renderer.getContext();
   297→    gl.getExtension('EXT_color_buffer_float');
   298→    const floatLinear = !!gl.getExtension('OES_texture_float_linear');
   299→    const foamFilter = floatLinear ? THREE.LinearFilter : THREE.NearestFilter;
   300→
   301→    this.rtStateA = this.mkRT(NX, NZ, THREE.NearestFilter);
   302→    this.rtStateB = this.mkRT(NX, NZ, THREE.NearestFilter);
   303→    this.rtEta = this.mkRT(NX, NZ, THREE.NearestFilter);
   304→    this.rtFoamA = this.mkRT(NX, NZ, foamFilter);
   305→    this.rtFoamB = this.mkRT(NX, NZ, foamFilter);
   306→    this.rtSprayA = this.mkRT(SPRAY_N, SPRAY_N, THREE.NearestFilter);
   307→    this.rtSprayB = this.mkRT(SPRAY_N, SPRAY_N, THREE.NearestFilter);
   308→    this.rtDown = this.mkRT(DOWN_W, DOWN_H, THREE.NearestFilter);
   309→
   310→    this.structBase = buildStructBase();
   311→    this.structArr = new Float32Array(this.structBase);
   312→    this.texBed = new THREE.DataTexture(this.bedGrid, NX, NZ, THREE.RedFormat, THREE.FloatType);
   313→    this.texBed.minFilter = this.texBed.magFilter = THREE.NearestFilter;
   314→    this.texBed.needsUpdate = true;
   315→    this.texStruct = new THREE.DataTexture(this.structArr, NX, NZ, THREE.RedFormat, THREE.FloatType);
   316→    this.texStruct.minFilter = this.texStruct.magFilter = THREE.NearestFilter;
   317→    this.texStruct.needsUpdate = true;
   318→    this.texInit = new THREE.DataTexture(buildInitState(), NX, NZ, THREE.RGBAFormat, THREE.FloatType);
   319→    this.texInit.minFilter = this.texInit.magFilter = THREE.NearestFilter;
   320→    this.texInit.needsUpdate = true;
   321→
   322→    const sprayInit = new Float32Array(SPRAY_N * SPRAY_N * 4);
   323→    for (let i = 0; i < sprayInit.length; i += 4) {
   324→      sprayInit[i] = 0; sprayInit[i + 1] = 0; sprayInit[i + 2] = 1; sprayInit[i + 3] = 1;
   325→    }
   326→    this.texSprayInit = new THREE.DataTexture(sprayInit, SPRAY_N, SPRAY_N, THREE.RGBAFormat, THREE.FloatType);
   327→    this.texSprayInit.needsUpdate = true;
   328→    this.texBlack = new THREE.DataTexture(new Float32Array([0, 0, 0, 1]), 1, 1, THREE.RGBAFormat, THREE.FloatType);
   329→    this.texBlack.needsUpdate = true;
   330→
   331→    const mat = (frag: string, uniforms: Record<string, THREE.IUniform>) =>
   332→      new THREE.ShaderMaterial({ vertexShader: QUAD_VERT, fragmentShader: frag, uniforms, depthTest: false, depthWrite: false });
   333→
   334→    const common = {
   335→      uState: { value: null },
   336→      uBed: { value: this.texBed },
   337→      uStruct: { value: this.texStruct },
   338→      uTexel: { value: new THREE.Vector2(TEXEL[0], TEXEL[1]) },
   339→      uCell: { value: new THREE.Vector2(CELL[0], CELL[1]) },
   340→      uG: { value: this.gravity },
   341→    };
   342→
   343→    this.etaPass = new FullScreenQuad(mat(ETA_FRAG, {
   344→      ...common,
   345→      uDt: { value: 0 },
   346→      uSrcRate: { value: 0 },
   347→      uSrcBox: { value: new THREE.Vector4(2 / LX, 8 / LX, (-34 + LZ / 2) / LZ, (34 + LZ / 2) / LZ) },
   348→    }));
   349→    this.velPass = new FullScreenQuad(mat(VEL_FRAG, {
   350→      ...common,
   351→      uEtaNew: { value: null },
   352→      uDt: { value: 0 },
   353→      uManning: { value: MANNING },
   354→    }));
   355→    this.foamPass = new FullScreenQuad(mat(FOAM_FRAG, {
   356→      ...common,
   357→      uFoam: { value: null },
   358→      uDt: { value: 0 },
   359→    }));
   360→    this.sprayPass = new FullScreenQuad(mat(SPRAY_FRAG, {
   361→      uSpray: { value: null },
   362→      uState: { value: null },
   363→      uBed: { value: this.texBed },
   364→      uStruct: { value: this.texStruct },
   365→      uTexel: { value: new THREE.Vector2(TEXEL[0], TEXEL[1]) },
   366→      uDt: { value: 0 },
   367→      uTime: { value: 0 },
   368→      uSpawnSpeed: { value: 3.2 },
   369→      uDomain: { value: new THREE.Vector2(LX, LZ) },
   370→    }));
   371→    this.copyPass = new FullScreenQuad(mat(COPY_FRAG, { uInit: { value: null } }));
   372→    this.downPass = new FullScreenQuad(mat(DOWN_FRAG, { uState: { value: null } }));
   373→
   374→    // count inflow strip cells (x ∈ [2,8], bed < 15)
   375→    let n = 0;
   376→    for (let j = 0; j < NZ; j++)
   377→      for (let i = Math.ceil(2 / DX); i < Math.floor(8 / DX); i++)
   378→        if (this.bedGrid[j * NX + i] < 15) n++;
   379→    this.stripCells = Math.max(n, 1);
   380→  }

   441→        uDomain: { value: new THREE.Vector2(LX, LZ) },
   442→        uPixelRatio: { value: this.renderer.getPixelRatio() },

   444→    });
   445→    const points = new THREE.Points(pg, this.sprayMat);
   446→    points.renderOrder = 11;

   449→  }

   476→  private simSubstep(dt: number): void {
   477→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   478→    eu.uState.value = this.curState.texture;
   479→    eu.uDt.value = dt;
   480→    eu.uG.value = this.gravity;
   481→    eu.uSrcRate.value = this.srcRate;
   482→    this.renderTo(this.etaPass, this.rtEta);
   483→
   484→    const vu = (this.velPass.material as THREE.ShaderMaterial).uniforms;
   485→    vu.uState.value = this.curState.texture;
   486→    vu.uEtaNew.value = this.rtEta.texture;
   487→    vu.uDt.value = dt;
   488→    vu.uG.value = this.gravity;
   489→    this.renderTo(this.velPass, this.nextState);
   490→    this.swapState();
   491→
   492→    if (this.foamOn) {
   493→      const fu = (this.foamPass.material as THREE.ShaderMaterial).uniforms;
   494→      fu.uState.value = this.curState.texture;
   495→      fu.uFoam.value = this.curFoam.texture;
   496→      fu.uDt.value = dt;
   497→      fu.uG.value = this.gravity;
   498→      this.renderTo(this.foamPass, this.nextFoam);
   499→      const t = this.curFoam;
   500→      this.curFoam = this.nextFoam;
   501→      this.nextFoam = t;
   502→    }
   503→
   504→    if (this.sprayOn) {
   505→      const su = (this.sprayPass.material as THREE.ShaderMaterial).uniforms;
   506→      su.uSpray.value = this.curSpray.texture;
   507→      su.uState.value = this.curState.texture;
   508→      su.uDt.value = dt;
   509→      su.uTime.value = this.simTime;
   510→      this.renderTo(this.sprayPass, this.nextSpray);
   511→      const t = this.curSpray;
   512→      this.curSpray = this.nextSpray;
   513→      this.nextSpray = t;
   514→    }
   515→  }

   531→    this.curFoam = this.rtFoamA;
   532→    this.nextFoam = this.rtFoamB;
   533→    this.curSpray = this.rtSprayA;
   534→    this.nextSpray = this.rtSprayB;

   536→    this.qPeak = 0;
   537→    this.applyStruct(null, null);

   619→  // ================================================================ main loop
   620→  private tick = (): void => {
   621→    this.rafId = requestAnimationFrame(this.tick);
   622→    const now = performance.now();
   623→    const dtReal = Math.min((now - this.lastFrameTime) / 1000, 0.05);
   624→    this.lastFrameTime = now;
   625→    this.fpsEma = this.fpsEma * 0.95 + (1 / Math.max(dtReal, 1e-4)) * 0.05;
   626→    const dtSim = this.paused ? 0 : dtReal * this.timeScale;
   627→
   628→    if (dtSim > 0) {
   629→      // scenario animation timers
   630→      if (this.breachT !== null) {
   631→        this.breachT += dtSim;
   632→        const tau = 2.6 / this.breachRate;
   633→        const p = 1 - Math.exp(-Math.pow(this.breachT / tau, 1.7));
   634→        this.breachElev = CREST + (BREACH_BOTTOM - CREST) * p;
   635→        this.dam.breachGroup.position.y = this.breachElev - CREST;
   636→        this.dam.breachGroup.rotation.z = -0.05 * p;
   637→        this.structDirty = true;
   638→        if (p > 0.995) this.breachT = null;
   639→      }
   640→      if (this.gateT !== null && this.gateT < 1) {
   641→        this.gateT = Math.min(1, this.gateT + dtSim / 2.5);
   642→        const e = easeInOut(this.gateT);
   643→        this.gateElev = SPILL_CREST_CLOSED + (GATE_OPEN_ELEV - SPILL_CREST_CLOSED) * e;
   644→        this.dam.gate.position.y = 18.85 + 6.9 * e;
   645→        this.structDirty = true;
   646→      }
   647→      if (this.overtopping) {
   648→        this.overtopT += dtSim;
   649→      }
   650→      const curInflow = this.overtopping
   651→        ? this.inflowQ + 272 * Math.min(this.overtopT / 18, 1)
   652→        : this.inflowQ;
   653→      this.srcRate = curInflow / (this.stripCells * DX * DZ);
   654→
   655→      if (this.structDirty) {
   656→        this.applyStruct(this.breachElev, this.gateElev);
   657→        this.structDirty = false;
   658→      }
   659→
   660→      const sub = clamp(Math.ceil(dtSim / DT_MAX), 1, 8);
   661→      const h = dtSim / sub;
   662→      for (let i = 0; i < sub; i++) this.simSubstep(h);
   663→      this.simTime += dtSim;
   664→      this.updateProps(dtSim);
   665→    }
   666→
   667→    // readback for CPU physics + stats
   668→    if (this.frame % 2 === 0) {
   669→      (this.downPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   670→      this.renderTo(this.downPass, this.rtDown);
   671→      this.renderer.readRenderTargetPixels(this.rtDown, 0, 0, DOWN_W, DOWN_H, this.field.data);
   672→    }
   673→
   674→    this.updateCamera(dtReal);
   675→
   676→    // shader uniforms
   677→    this.waterMat.uniforms.uState.value = this.curState.texture;
   678→    this.waterMat.uniforms.uFoam.value = this.curFoam.texture;
   679→    this.waterMat.uniforms.uTime.value = this.simTime;
   680→    this.waterMat.uniforms.uCamPos.value.copy(this.camera.position);
   681→    this.sprayMat.uniforms.uSpray.value = this.curSpray.texture;
   682→    this.sprayMat.uniforms.uState.value = this.curState.texture;
   683→    (this.sprayPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   684→
   685→    // render with camera shake
   686→    let shook: THREE.Vector3 | null = null;
   687→    if (this.shake > 0.002) {
   688→      shook = new THREE.Vector3(
   689→        (Math.random() - 0.5) * this.shake * 0.9,
   690→        (Math.random() - 0.5) * this.shake * 0.9,
   691→        (Math.random() - 0.5) * this.shake * 0.9,
   692→      );
   693→      this.camera.position.add(shook);
   694→    }
   695→    this.renderer.render(this.scene, this.camera);
   696→    if (shook) {
   697→      this.camera.position.sub(shook);
   698→      this.shake *= Math.exp(-dtReal * 1.15);
   699→    }
   700→
   701→    // stats ~10 Hz
   702→    if (this.frame % 6 === 0 && this.onStats) {
   703→      const s = this.computeStats();
   704→      this.onStats(s);
   705→      const v01 = clamp(s.vmax / 10, 0, 1);
   706→      this.audio.setIntensity(v01);
   707→    }
   708→
   709→    if (this.frame === 3) {
   710→      (window as unknown as Record<string, unknown>).__damReady = true;
   711→    }
   712→    this.frame++;
   713→  };

   715→  // ================================================================ props update
   716→  private updateProps(dt: number): void {
   717→    const f = this.field;
   718→    const s = this.sample;
   719→
   720→    for (const b of this.barrels) {
   721→      f.sample(b.pos.x, b.pos.z, s);
   722→      const bed = bedAt(b.pos.x, b.pos.z);
   723→      const dep = s.eta - bed;
   724→      if (dep > 0.4) {
   725→        const bob = Math.sin(this.simTime * 1.7 + b.pos.x) * 0.06;
   726→        b.pos.y += (s.eta + 0.15 + bob - b.pos.y) * Math.min(1, dt * 5);
   727→        b.vel.x += (s.u * 0.92 - b.vel.x) * Math.min(1, dt * 2.2);
   728→        b.vel.z += (s.v * 0.92 - b.vel.z) * Math.min(1, dt * 2.2);
   729→        b.pos.x += b.vel.x * dt;
   730→        b.pos.z += b.vel.z * dt;
   731→      } else {
   732→        b.vel.multiplyScalar(1 - Math.min(1, dt * 2.5));
   733→        b.pos.y += (bed + 0.3 - b.pos.y) * Math.min(1, dt * 4);
   734→      }
   735→      b.pos.x = clamp(b.pos.x, 1, LX - 1);
   736→      b.pos.z = clamp(b.pos.z, -LZ / 2 + 1, LZ / 2 - 1);
   737→      const spd = Math.sqrt(b.vel.x * b.vel.x + b.vel.z * b.vel.z);
   738→      b.mesh.position.copy(b.pos);
   739→      b.mesh.rotation.x = clamp(b.vel.z * 0.1, -0.5, 0.5) + Math.sin(this.simTime * 2.1 + b.pos.z) * 0.04;
   740→      b.mesh.rotation.z = clamp(-b.vel.x * 0.1, -0.5, 0.5);
   741→      b.mesh.rotation.y += (0.2 + spd * 0.3) * dt;
   742→    }
   743→
   744→    for (const h of this.houses) {
   745→      if (h.prog >= 1) continue;
   746→      f.sample(h.group.position.x, h.group.position.z, s);
   747→      const wd = s.eta - h.ground;
   748→      if (wd > 0.55) {
   749→        h.prog = Math.min(1, h.prog + dt * 0.4);
   750→        const p = h.prog;
   751→        const dir = Math.atan2(s.v, s.u);
   752→        h.group.rotation.x = -Math.cos(dir) * p * 1.35;
   753→        h.group.rotation.z = Math.sin(dir) * p * 1.35;
   754→        h.group.position.x += s.u * dt * 0.45 * p;
   755→        h.group.position.z += s.v * dt * 0.45 * p;
   756→        h.group.position.y = h.ground - p * 0.5;
   757→      }
   758→    }
   759→
   760→    for (const t of this.trees) {
   761→      if (t.prog >= 1) continue;
   762→      f.sample(t.group.position.x, t.group.position.z, s);
   763→      const wd = s.eta - t.ground;
   764→      if (wd > 0.7) {
   765→        t.prog = Math.min(1, t.prog + dt * 0.6);
   766→        const p = t.prog;
   767→        const dir = Math.atan2(s.v, s.u);
   768→        t.group.rotation.x = -Math.cos(dir) * p * 1.3;
   769→        t.group.rotation.z = Math.sin(dir) * p * 1.3;
   770→        t.group.position.x += s.u * dt * 0.25 * p;
   771→        t.group.position.z += s.v * dt * 0.25 * p;
   772→      }
   773→    }
   774→
   775→    for (const c of this.chunks) {
   776→      if (!c.active) continue;
   777→      c.vel.y -= 11 * dt;
   778→      c.mesh.position.addScaledVector(c.vel, dt);
   779→      c.mesh.rotation.x += c.vel.z * dt * 0.15;
   780→      c.mesh.rotation.z -= c.vel.x * dt * 0.15;
   781→      const ground = bedAt(c.mesh.position.x, c.mesh.position.z) + 0.25;
   782→      if (c.mesh.position.y < ground) {
   783→        c.mesh.position.y = ground;
   784→        c.vel.y *= -0.35;
   785→        c.vel.x *= 0.6;
   786→        c.vel.z *= 0.6;
   787→        if (c.vel.lengthSq() < 0.4) c.active = false;
   788→      }
   789→      if (c.mesh.position.x > LX - 1) c.active = false;
   790→    }
   791→  }

   793→  // ================================================================ stats
   794→  private computeStats(): DamStats {
   795→    const d = this.field.data;
   796→    const dzC = LZ / DOWN_H;
   797→    const gx = 37;
   798→    let q = 0;
   799→    let stage = 0;
   800→    let froude = 0;
   801→    for (let j = 0; j < DOWN_H; j++) {
   802→      const k = (j * DOWN_W + gx) * 4;
   803→      const h = Math.max(d[k + 3], 0);
   804→      const u = d[k + 1];
   805→      q += u * h * dzC;
   806→      stage = Math.max(stage, h);
   807→      if (h > 0.05) {
   808→        froude = Math.max(froude, Math.abs(u) / Math.sqrt(this.gravity * h));
   809→      }
   810→    }
   811→    let vmax = 0;
   812→    let level = 0;
   813→    for (let i = 0; i < DOWN_W * DOWN_H; i++) {
   814→      const k = i * 4;
   815→      const sp = Math.sqrt(d[k + 1] * d[k + 1] + d[k + 2] * d[k + 2]);
   816→      vmax = Math.max(vmax, sp);
   817→      if (i % DOWN_W < 18 && d[k + 3] > 0.1) level = Math.max(level, d[k]);
   818→    }
   819→    if (q > this.qPeak) this.qPeak = q;
   820→    return {
   821→      t: this.simTime,
   822→      q: Math.max(q, 0),
   823→      qPeak: this.qPeak,
   824→      level,
   825→      vmax,
   826→      froude,
   827→      fps: this.fpsEma,
   828→      phase: this.phase,
   829→      overtopping: this.overtopping,
   830→    };
   831→  }

   850→      this.controls.target.lerpVectors(this.tweenFromT, this.tweenToT, e);
