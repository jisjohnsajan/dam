    14→function hashRnd(i: number): number {
    15→  const h = Math.sin(i * 127.1 + 311.7) * 43758.5453;
    16→  return h - Math.floor(h);
    17→}

    19→function canvasTex(w: number, h: number, draw: (c: CanvasRenderingContext2D) => void): THREE.Texture {
    20→  const cv = document.createElement('canvas');
    21→  cv.width = w;
    22→  cv.height = h;
    23→  const ctx = cv.getContext('2d')!;
    24→  draw(ctx);
    25→  const tex = new THREE.CanvasTexture(cv);
    26→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    27→  tex.colorSpace = THREE.SRGBColorSpace;
    28→  tex.anisotropy = 4;
    29→  return tex;
    30→}

    52→function makePylon(h: number): THREE.Group {
    53→  const g = new THREE.Group();
    54→  const mat = matGalvanised();
    55→  const base = 1.5;
    56→  const top = 0.45;
    57→  // 4 splayed legs
    58→  for (const sx of [-1, 1]) {
    59→    for (const sz of [-1, 1]) {
    60→      const leg = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.09, h, 5), mat);
    61→      const lx = (sx * (base + (top - base) * 0.5)) / 2;
    62→      const lz = (sz * (base + (top - base) * 0.5)) / 2;
    63→      leg.position.set(lx, h / 2, lz);
    64→      leg.rotation.z = -sx * 0.115;
    65→      leg.rotation.x = sz * 0.115;
    66→      leg.castShadow = true;
    67→      g.add(leg);
    68→    }
    69→  }
    70→  // cross-braces on both faces (2 levels)
    71→  for (const lvl of [0.3, 0.62]) {
    72→    for (const sz of [-1, 1]) {
    73→      const brace = new THREE.Mesh(new THREE.BoxGeometry(base * (1 - lvl * 0.55), 0.06, 0.06), mat);
    74→      brace.position.set(0, h * lvl, (sz * base * (1 - lvl * 0.55)) / 2);
    75→      brace.rotation.z = sz * 0.5;
    76→      g.add(brace);
    77→    }
    78→  }
    79→  // cross-arms (two levels) + peak
    80→  const arm1 = new THREE.Mesh(new THREE.BoxGeometry(3.6, 0.09, 0.09), mat);
    81→  arm1.position.y = h * 0.8;
    82→  const arm2 = new THREE.Mesh(new THREE.BoxGeometry(2.7, 0.09, 0.09), mat);
    83→  arm2.position.y = h * 0.93;
    84→  const peak = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.05, h * 0.1, 5), mat);
    85→  peak.position.y = h * 1.04;
    86→  arm1.castShadow = arm2.castShadow = true;
    87→  g.add(arm1, arm2, peak);
    88→  return g;
    89→}

    91→function makeCable(a: THREE.Vector3, b: THREE.Vector3, sag: number, mat: THREE.Material): THREE.Mesh {
    92→  const mid = a.clone().add(b).multiplyScalar(0.5);
    93→  mid.y -= sag;
    94→  const curve = new THREE.QuadraticBezierCurve3(a, mid, b);
    95→  const geo = new THREE.TubeGeometry(curve, 20, 0.035, 5, false);
    96→  const m = new THREE.Mesh(geo, mat);
    97→  m.castShadow = false;
    98→  return m;
    99→}

   383→function distToStreets(x: number, z: number): number {
   384→  let best = 1e9;
   385→  for (const st of TOWN_STREETS) {
   386→    for (let i = 0; i < st.pts.length - 1; i++) {
   387→      const [ax, az] = st.pts[i];
   388→      const [bx, bz] = st.pts[i + 1];
   389→      const dx = bx - ax, dz = bz - az;
   390→      const t = clamp(((x - ax) * dx + (z - az) * dz) / (dx * dx + dz * dz), 0, 1);
   391→      const d = Math.hypot(x - (ax + dx * t), z - (az + dz * t));
   392→      best = Math.min(best, d);
   393→    }
   394→  }
   395→  return best;
   396→}

   398→function flatRibbon(group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material, lift = 0.14): void {
   399→  const pos: number[] = [];
   400→  const idx: number[] = [];
   401→  for (let i = 0; i < pts.length - 1; i++) {
   402→    for (let s = 0; s < 3; s++) {
   403→      const t0 = s / 3, t1 = (s + 1) / 3;
   404→      const x0 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t0;
   405→      const z0 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t0;
   406→      const x1 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t1;
   407→      const z1 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t1;
   408→      const dx = z1 - z0, dz = -(x1 - x0);
   409→      const dl = Math.hypot(dx, dz) || 1;
   410→      const nx = (dx / dl) * w * 0.5, nz = (dz / dl) * w * 0.5;
   411→      const a = bedAt(x0 - nx, z0 - nz) + lift;
   412→      const b = bedAt(x0 + nx, z0 + nz) + lift;
   413→      const c = bedAt(x1 - nx, z1 - nz) + lift;
   414→      const d = bedAt(x1 + nx, z1 + nz) + lift;
   415→      const base = pos.length / 3;
   416→      pos.push(x0 - nx, a, z0 - nz, x0 + nx, b, z0 + nz, x1 - nx, c, z1 - nz, x1 + nx, d, z1 + nz);
   417→      idx.push(base, base + 2, base + 1, base + 1, base + 2, base + 3);
   418→    }
   419→  }
   420→  const geo = new THREE.BufferGeometry();
   421→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   422→  geo.setIndex(idx);
   423→  geo.computeVertexNormals();
   424→  const mesh = new THREE.Mesh(geo, mat);
   425→  mesh.receiveShadow = true;
   426→  group.add(mesh);
   427→}

   795→// ============================================================ FAR TERRAIN
   796→// Visual-only continuation of the world beyond the solver domain: the valley
   797→// runs on, then a mountain ring closes the horizon (fog blends it out).
   798→function farBed(x: number, z: number): number {
   799→  const cx = clamp(x, 0.2, LX - 0.2);
   800→  const cz = clamp(z, -LZ / 2 + 0.2, LZ / 2 - 0.2);
   801→  let h = bedAt(cx, cz);
   802→  // upstream / downstream closures rise steeply (they sit behind the valley
   803→  // axis); the side ranges rise gently so overview sight-lines stay open
   804→  const riseX = Math.max((6 - x) * 0.4, (x - (LX + 26)) * 0.42, 0);
   805→  const riseZ = Math.max((-LZ / 2 - 30 - z) * 0.26, (z - (LZ / 2 + 30)) * 0.26, 0);
   806→  const rise = Math.max(riseX, riseZ);
   807→  h += Math.min(rise, 95) * (0.62 + 0.75 * fbm(x * 0.035 + 5.5, z * 0.035 + 1.1, 4));
   808→  h += (fbm(x * 0.06 + 9.1, z * 0.06 + 3.3, 3) - 0.5) * 9;
   809→  return h;
   810→}

   812→function farPatch(x0: number, x1: number, z0: number, z1: number, sx: number, sz: number): THREE.Mesh {
   813→  const geo = new THREE.PlaneGeometry(x1 - x0, z1 - z0, sx, sz);
   814→  geo.rotateX(-Math.PI / 2);
   815→  geo.translate((x0 + x1) / 2, 0, (z0 + z1) / 2);
   816→  const pos = geo.attributes.position as THREE.BufferAttribute;
   817→  const colors = new Float32Array(pos.count * 3);
   818→  const col = new THREE.Color();
   819→  const tc = { r: 0, g: 0, b: 0 };
   820→  for (let i = 0; i < pos.count; i++) {
   821→    const x = pos.getX(i);
   822→    const z = pos.getZ(i);
   823→    const y = farBed(x, z);
   824→    pos.setY(i, y);
   825→    const e = 2.4;
   826→    const sl = Math.hypot(farBed(x + e, z) - farBed(x - e, z), farBed(x, z + e) - farBed(x, z - e)) / (2 * e);
   827→    terrainColor(x, z, y, sl, tc);
   828→    col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   829→    colors[i * 3] = col.r;
   830→    colors[i * 3 + 1] = col.g;
   831→    colors[i * 3 + 2] = col.b;
   832→  }
   833→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   834→  geo.computeVertexNormals();
   835→  const mesh = new THREE.Mesh(
   836→    geo,
   837→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.97, metalness: 0, envMapIntensity: 0.18 }),
   838→  );
   839→  mesh.receiveShadow = false;
   840→  return mesh;
   841→}

   843→export function buildFarTerrain(): { group: THREE.Group } {
   844→  const group = new THREE.Group();
   845→  const E = 170; // how far the ring extends sideways
   846→  const Z0 = -LZ / 2;
   847→  const Z1 = LZ / 2;
   848→  // upstream continuation (covers the corners too)
   849→  group.add(farPatch(-150, 0, -E, E, 30, 68));
   850→  // downstream continuation toward the horizon
   851→  group.add(farPatch(LX, LX + 150, -E, E, 30, 68));
   852→  // side ranges
   853→  group.add(farPatch(0, LX, Z0 - E, Z0, 44, 30));
   854→  group.add(farPatch(0, LX, Z1, Z1 + E, 44, 30));
   855→  return { group };
   856→}
