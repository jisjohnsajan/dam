   335→   335→  private buildTerrainMesh(): void {
   336→   336→    // bed grid on cell centers
   337→   337→    this.bedGrid = new Float32Array(NX * NZ);
   338→   338→    for (let j = 0; j < NZ; j++)
   339→   339→      for (let i = 0; i < NX; i++)
   340→   340→        this.bedGrid[j * NX + i] = bedAt((i + 0.5) * DX, (j + 0.5) * DZ - LZ / 2);
   341→   341→
   342→   342→    const geo = new THREE.PlaneGeometry(LX, LZ, NX - 1, NZ - 1);
   343→   343→    geo.rotateX(-Math.PI / 2);
   344→   344→    geo.translate(LX / 2, 0, 0);
   345→   345→    const pos = geo.attributes.position as THREE.BufferAttribute;
   346→   346→    const colors = new Float32Array(pos.count * 3);
   347→   347→    const col = new THREE.Color();
   348→   348→    const tc = { r: 0, g: 0, b: 0 };
   349→   349→    for (let i = 0; i < pos.count; i++) {
   350→   350→      const x = pos.getX(i);
   351→   351→      const z = pos.getZ(i);
   352→   352→      const y = this.sampleBed(x, z);
   353→   353→      pos.setY(i, y);
   354→   354→      const e = 1.2;
   355→   355→      const sx = (this.sampleBed(x + e, z) - this.sampleBed(x - e, z)) / (2 * e);
   356→   356→      const sz = (this.sampleBed(x, z + e) - this.sampleBed(x, z - e)) / (2 * e);
   357→   357→      const slope = Math.sqrt(sx * sx + sz * sz);
   358→   358→      terrainColor(x, z, y, slope, tc);
   359→   359→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   360→   360→      colors[i * 3] = col.r;
   361→   361→      colors[i * 3 + 1] = col.g;
   362→   362→      colors[i * 3 + 2] = col.b;
   363→   363→    }
   364→   364→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   365→   365→    geo.computeVertexNormals();
   366→   366→    const mesh = new THREE.Mesh(
   367→   367→      geo,
   368→   368→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.96, metalness: 0, envMapIntensity: 0.22 }),
   369→   369→    );
   370→   370→    mesh.receiveShadow = true;
   371→   371→    this.scene.add(mesh);
   372→   372→
   373→   373→    // surround plane so the domain doesn't float in the void
   374→   374→    const surround = new THREE.Mesh(
   375→   375→      new THREE.PlaneGeometry(3000, 3000),
   376→   376→      new THREE.MeshStandardMaterial({ color: 0xa8ad97, roughness: 1, metalness: 0, envMapIntensity: 0.1 }),
   377→   377→    );
   378→   378→    surround.rotation.x = -Math.PI / 2;
   379→   379→    surround.position.set(LX / 2, 1.2, 0);
   380→   380→    this.scene.add(surround);
   381→   381→  }

   383→   383→  private sampleBed(x: number, z: number): number {
   384→   384→    const fu = clamp((x / DX) - 0.5, 0, NX - 1.001);
   385→   385→    const fv = clamp((z + LZ / 2) / DZ - 0.5, 0, NZ - 1.001);
   386→   386→    const i = Math.floor(fu), j = Math.floor(fv);
   387→   387→    const fx = fu - i, fy = fv - j;
   388→   388→    const i1 = Math.min(i + 1, NX - 1), j1 = Math.min(j + 1, NZ - 1);
   389→   389→    const g = this.bedGrid;
   390→   390→    const a = g[j * NX + i] + (g[j * NX + i1] - g[j * NX + i]) * fx;
   391→   391→    const b = g[j1 * NX + i] + (g[j1 * NX + i1] - g[j1 * NX + i]) * fx;
   392→   392→    return a + (b - a) * fy;
   393→   393→  }
