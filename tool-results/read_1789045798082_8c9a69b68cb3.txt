   315→   315→  // ================================================================ terrain mesh
   316→   316→  private buildTerrainMesh(): void {
   317→   317→    // bed grid on cell centers
   318→   318→    this.bedGrid = new Float32Array(NX * NZ);
   319→   319→    for (let j = 0; j < NZ; j++)
   320→   320→      for (let i = 0; i < NX; i++)
   321→   321→        this.bedGrid[j * NX + i] = bedAt((i + 0.5) * DX, (j + 0.5) * DZ - LZ / 2);
   322→   322→
   323→   323→    const geo = new THREE.PlaneGeometry(LX, LZ, NX - 1, NZ - 1);
   324→   324→    geo.rotateX(-Math.PI / 2);
   325→   325→    geo.translate(LX / 2, 0, 0);
   326→   326→    const pos = geo.attributes.position as THREE.BufferAttribute;
   327→   327→    const colors = new Float32Array(pos.count * 3);
   328→   328→    const col = new THREE.Color();
   329→   329→    const tc = { r: 0, g: 0, b: 0 };
   330→   330→    for (let i = 0; i < pos.count; i++) {
   331→   331→      const x = pos.getX(i);
   332→   332→      const z = pos.getZ(i);
   333→   333→      const y = this.sampleBed(x, z);
   334→   334→      pos.setY(i, y);
   335→   335→      const e = 1.2;
   336→   336→      const sx = (this.sampleBed(x + e, z) - this.sampleBed(x - e, z)) / (2 * e);
   337→   337→      const sz = (this.sampleBed(x, z + e) - this.sampleBed(x, z - e)) / (2 * e);
   338→   338→      const slope = Math.sqrt(sx * sx + sz * sz);
   339→   339→      terrainColor(x, z, y, slope, tc);
   340→   340→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   341→   341→      colors[i * 3] = col.r;
   342→   342→      colors[i * 3 + 1] = col.g;
   343→   343→      colors[i * 3 + 2] = col.b;
   344→   344→    }
   345→   345→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   346→   346→    geo.computeVertexNormals();
   347→   347→    const mesh = new THREE.Mesh(
   348→   348→      geo,
   349→   349→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.96, metalness: 0, envMapIntensity: 0.22 }),
   350→   350→    );
   351→   351→    mesh.receiveShadow = true;
   352→   352→    this.scene.add(mesh);
   353→   353→
   354→   354→    // surround plane so the domain doesn't float in the void
   355→   355→    const surround = new THREE.Mesh(
   356→   356→      new THREE.PlaneGeometry(3000, 3000),
   357→   357→      new THREE.MeshStandardMaterial({ color: 0xa8ad97, roughness: 1, metalness: 0, envMapIntensity: 0.1 }),
   358→   358→    );
   359→   359→    surround.rotation.x = -Math.PI / 2;
   360→   360→    surround.position.set(LX / 2, 1.2, 0);
   361→   361→    this.scene.add(surround);
   362→   362→  }

   364→   364→  private sampleBed(x: number, z: number): number {
   365→   365→    const fu = clamp((x / DX) - 0.5, 0, NX - 1.001);
   366→   366→    const fv = clamp((z + LZ / 2) / DZ - 0.5, 0, NZ - 1.001);
   367→   367→    const i = Math.floor(fu), j = Math.floor(fv);
   368→   368→    const fx = fu - i, fy = fv - j;
   369→   369→    const i1 = Math.min(i + 1, NX - 1), j1 = Math.min(j + 1, NZ - 1);
   370→   370→    const g = this.bedGrid;
   371→   371→    const a = g[j * NX + i] + (g[j * NX + i1] - g[j * NX + i]) * fx;
   372→   372→    const b = g[j1 * NX + i] + (g[j1 * NX + i1] - g[j1 * NX + i]) * fx;
   373→   373→    return a + (b - a) * fy;
   374→   374→  }

   610→   610→  private simSubstep(dt: number): void {
   611→   611→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   612→   612→    eu.uState.value = this.curState.texture;
   613→   613→    eu.uDt.value = dt;
   614→   614→    eu.uG.value = this.gravity;
   615→   615→    eu.uSrcRate.value = this.srcRate;
   616→   616→    this.renderTo(this.etaPass, this.rtEta);
   617→   617→
   618→   618→    const vu = (this.velPass.material as THREE.ShaderMaterial).uniforms;
   619→   619→    vu.uState.value = this.curState.texture;
   620→   620→    vu.uEtaNew.value = this.rtEta.texture;
   621→   621→    vu.uDt.value = dt;
   622→   622→    vu.uG.value = this.gravity;
   623→   623→    this.renderTo(this.velPass, this.nextState);
   624→   624→    this.swapState();
   625→   625→
   626→   626→    if (this.foamOn) {
   627→   627→      const fu = (this.foamPass.material as THREE.ShaderMaterial).uniforms;
   628→   628→      fu.uState.value = this.curState.texture;
   629→   629→      fu.uFoam.value = this.curFoam.texture;
   630→   630→      fu.uDt.value = dt;
   631→   631→      fu.uG.value = this.gravity;
   632→   632→      this.renderTo(this.foamPass, this.nextFoam);
   633→   633→      const t = this.curFoam;
   634→   634→      this.curFoam = this.nextFoam;
   635→   635→      this.nextFoam = t;
   636→   636→    }
   637→   637→
   638→   638→    if (this.sprayOn) {
   639→   639→      const su = (this.sprayPass.material as THREE.ShaderMaterial).uniforms;
   640→   640→      su.uSpray.value = this.curSpray.texture;
   641→   641→      su.uState.value = this.curState.texture;
   642→   642→      su.uDt.value = dt;
   643→   643→      su.uTime.value = this.simTime;
   644→   644→      this.renderTo(this.sprayPass, this.nextSpray);
   645→   645→      const t = this.curSpray;
   646→   646→      this.curSpray = this.nextSpray;
   647→   647→      this.nextSpray = t;
   648→   648→    }
   649→   649→  }

   696→   696→  runScenario(p: ScenarioParams): void {
   697→   697→    if (this.scenario) return;
   698→   698→    this.scenario = p;
   699→   699→    this.phase = 'scenario';
   700→   700→    this.scenT = 0;
   701→   701→    this.snapClock = 0;
   702→   702→    this.overtopT = 0;
   703→   703→    this.rainInflow = p.rain === 'none' ? 0 : p.rain === 'moderate' ? 26 : 64;
   704→   704→    this.breachSpan = null;
   705→   705→    this.breachT = null;
   706→   706→    this.breachDepth01 = 0;
   707→   707→    this.structDirty = true;
   708→   708→
   709→   709→    // reservoir set to the scenario level (drive takes it there quickly)
   710→   710→    const target = clamp(fracToSimLevel(p.levelFrac), 0, 24.4);
   711→   711→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   712→   712→    eu.uDriveEta.value = target;
   713→   713→    eu.uDriveOn.value = 1;
   714→   714→
   715→   715→    if (p.mechanism === 'overtopping') {
   716→   716→      // surge inflow pushes the lake above the crest; erosion follows
   717→   717→      eu.uDriveEta.value = Math.max(target, CREST + 1.0);
   718→   718→    } else {
   719→   719→      // schedule breach after the reservoir reaches level (~2-6 demo s)
   720→   720→      this.breachT = -clamp((target - this.lastLevel) / 1.4, 1.5, 5);
   721→   721→      this.breachTau = Math.max(p.formationMin / this.damProfile.timeMinPerSec, 4) / (p.mechanism === 'piping' ? 0.65 : 1) / this.breachRate;
   722→   722→      this.breachPre = p.mechanism === 'piping' ? 0.22 : 0;
   723→   723→      const span = this.breachSpanFor(p.breachWidthM, p.location);
   724→   724→      this.breachSpan = span;
   725→   725→      this.breachInverts = [];
   726→   726→      for (let b = span.start; b < span.start + span.count; b++) {
   727→   727→        const z0 = BLOCK_Z0 + b * BLOCK_W;
   728→   728→        this.breachInverts.push(this.blockInvert(z0 + BLOCK_W / 2));
   729→   729→      }
   730→   730→    }
   731→   731→    this.audio.burst(0.2);
   732→   732→  }

   815→   815→  private tick = (): void => {
   816→   816→    this.rafId = requestAnimationFrame(this.tick);
   817→   817→    const now = performance.now();
   818→   818→    const dtReal = Math.min((now - this.lastFrameTime) / 1000, 0.05);
   819→   819→    this.lastFrameTime = now;
   820→   820→    this.fpsEma = this.fpsEma * 0.95 + (1 / Math.max(dtReal, 1e-4)) * 0.05;
   821→   821→    const dtSim = this.paused ? 0 : dtReal * this.timeScale;
   822→   822→
   823→   823→    if (dtSim > 0 && !this.scrubbing) {
   824→   824→      this.stepScenario(dtSim);
   825→   825→
   826→   826→      const curInflow = this.baseInflow + this.rainInflow;
   827→   827→      this.srcRate = curInflow / (this.stripCells * DX * DZ);
   828→   828→
   829→   829→      if (this.structDirty) {
   830→   830→        this.applyStruct(this.breachSpan, this.gateElev);
   831→   831→        this.structDirty = false;
   832→   832→      }
   833→   833→
   834→   834→      const sub = clamp(Math.ceil(dtSim / DT_MAX), 1, 14);
   835→   835→      const h = dtSim / sub;
   836→   836→      for (let i = 0; i < sub; i++) this.simSubstep(h);
   837→   837→      this.simTime += dtSim;
   838→   838→      this.updateProps(dtSim);
   839→   839→
   840→   840→      // snapshots during a scenario for the time machine
   841→   841→      if (this.scenario) {
   842→   842→        this.snapClock += dtSim;
   843→   843→        if (this.snapClock >= SNAP_EVERY) {
   844→   844→          this.snapClock = 0;
   845→   845→          this.recordSnapshot();
   846→   846→        }
   847→   847→      }
   848→   848→    }
   849→   849→
   850→   850→    // readback for CPU physics + stats
   851→   851→    if (this.frame % 2 === 0) {
   852→   852→      (this.downPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   853→   853→      this.renderTo(this.downPass, this.rtDown);
   854→   854→      this.renderer.readRenderTargetPixels(this.rtDown, 0, 0, DOWN_W, DOWN_H, this.field.data);
   855→   855→      this.updateArrival();
   856→   856→    }
   857→   857→
   858→   858→    this.updateCamera(dtReal);
   859→   859→
   860→   860→    // shader uniforms
   861→   861→    this.waterMat.uniforms.uState.value = this.curState.texture;
   862→   862→    this.waterMat.uniforms.uFoam.value = this.curFoam.texture;
   863→   863→    this.waterMat.uniforms.uTime.value = this.simTime;
   864→   864→    this.waterMat.uniforms.uCamPos.value.copy(this.camera.position);
   865→   865→    this.sprayMat.uniforms.uSpray.value = this.curSpray.texture;
   866→   866→    this.sprayMat.uniforms.uState.value = this.curState.texture;
   867→   867→    (this.sprayPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   868→   868→
   869→   869→    // render with camera shake
   870→   870→    let shook: THREE.Vector3 | null = null;
   871→   871→    if (this.shake > 0.002) {
   872→   872→      shook = new THREE.Vector3(
   873→   873→        (Math.random() - 0.5) * this.shake * 0.9,
   874→   874→        (Math.random() - 0.5) * this.shake * 0.9,
   875→   875→        (Math.random() - 0.5) * this.shake * 0.9,
   876→   876→      );
   877→   877→      this.camera.position.add(shook);
   878→   878→    }
   879→   879→    this.composer.render();
   880→   880→    if (shook) {
   881→   881→      this.camera.position.sub(shook);
   882→   882→      this.shake *= Math.exp(-dtReal * 1.15);
   883→   883→    }
   884→   884→
   885→   885→    // stats ~10 Hz
   886→   886→    if (this.frame % 6 === 0 && this.onStats) {
   887→   887→      const s = this.computeStats();
   888→   888→      this.onStats(s);
   889→   889→      const v01 = clamp(s.vmax / 10, 0, 1);
   890→   890→      this.audio.setIntensity(v01);
   891→   891→    }
   892→   892→
   893→   893→    if (this.frame === 3) {
   894→   894→      (window as unknown as Record<string, unknown>).__damReady = true;
   895→   895→    }
   896→   896→    this.frame++;
   897→   897→  };

   899→   899→  // ------------------------------------------------ scenario + gate animation
   900→   900→  private stepScenario(dt: number): void {
   901→   901→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   902→   902→    const sc = this.scenario;
   903→   903→
   904→   904→    if (this.gateT !== null && this.gateT < 1) {
   905→   905→      this.gateT = Math.min(1, this.gateT + dt / 2.5);
   906→   906→      const e = easeInOut(this.gateT);
   907→   907→      this.gateElev = SPILL_CREST_CLOSED + (GATE_OPEN_ELEV - SPILL_CREST_CLOSED) * e;
   908→   908→      for (const g of this.dam.gates) g.position.y = 18.85 + 6.9 * e;
   909→   909→      this.structDirty = true;
   910→   910→    }
   911→   911→
   912→   912→    if (!sc) {
   913→   913→      // live monitoring: drive toward the user level target
   914→   914→      eu.uDriveOn.value = 1;
   915→   915→      return;
   916→   916→    }
   917→   917→
   918→   918→    this.scenT += dt;
   919→   919→
   920→   920→    if (sc.mechanism === 'overtopping') {
   921→   921→      this.overtopT += dt;
   922→   922→      // once overtopped for long enough, crest erosion begins
   923→   923→      if (this.overtopT > 7 && (this.breachT === null || this.breachT < 0)) {
   924→   924→        if (this.breachSpan === null) {
   925→   925→          const span = this.breachSpanFor(sc.breachWidthM, sc.location);
   926→   926→          this.breachSpan = span;
   927→   927→          this.breachTau = Math.max(sc.formationMin / this.damProfile.timeMinPerSec, 4) / this.breachRate;
   928→   928→          this.breachPre = 0;
   929→   929→          this.breachInverts = [];
   930→   930→          for (let b = span.start; b < span.start + span.count; b++) {
   931→   931→            const z0 = BLOCK_Z0 + b * BLOCK_W;
   932→   932→            this.breachInverts.push(this.blockInvert(z0 + BLOCK_W / 2));
   933→   933→          }
   934→   934→        }
   935→   935→        this.triggerBreachErosion();
   936→   936→      }
   937→   937→    }
   938→   938→
   939→   939→    // breach formation
   940→   940→    if (this.breachT !== null && this.breachSpan) {
   941→   941→      this.breachT += dt;
   942→   942→      if (this.breachT > 0) {
   943→   943→        const p = 1 - Math.exp(-Math.pow(this.breachT / this.breachTau, 1.7));
   944→   944→        this.breachDepth01 = clamp(this.breachPre + (1 - this.breachPre) * p, 0, 1);
   945→   945→        this.breachSpan.depth01 = this.breachDepth01;
   946→   946→        this.structDirty = true;
   947→   947→        // sink the failing blocks with the SAME curve as the struct field
   948→   948→        const bi = this.breachInverts;
   949→   949→        for (let k = 0; k < this.breachSpan.count; k++) {
   950→   950→          const blockIdx = this.breachSpan.start + k;
   951→   951→          const blk = this.dam.breachBlocks[blockIdx];
   952→   952→          if (!blk) continue;
   953→   953→          const drop = (CREST - bi[k]) * this.breachDepth01;
   954→   954→          blk.group.position.y = -drop;
   955→   955→          blk.group.rotation.z = -0.05 * this.breachDepth01;
   956→   956→        }
   957→   957→        if (p > 0.995 && this.breachT > this.breachTau) {
   958→   958→          this.breachT = null; // formation complete; struct stays at full depth
   959→   959→        }
   960→   960→      }
   961→   961→      // stop the level drive once the breach starts draining the lake
   962→   962→      if (this.breachDepth01 > 0.02 && sc.mechanism !== 'overtopping') {
   963→   963→        eu.uDriveOn.value = 0;
   964→   964→      }
   965→   965→    }
   966→   966→  }

   969→   969→  private updateArrival(): void {
   970→   970→    const d = this.field.data;
   971→   971→    let changed = false;

   991→   991→    }

  1040→  1040→  // ================================================================ props update
  1041→  1041→  private updateProps(dt: number): void {
  1042→  1042→    const f = this.field;
  1043→  1043→    const s = this.sample;
  1044→  1044→
  1045→  1045→    for (const fl of this.floaters) {
  1046→  1046→      f.sample(fl.pos.x, fl.pos.z, s);
  1047→  1047→      const bed = bedAt(fl.pos.x, fl.pos.z);
  1048→  1048→      const dep = s.eta - bed;
  1049→  1049→      const isBoat = !((fl.mesh as THREE.Mesh).isMesh); // boats are Groups, barrels are Meshes
  1050→  1050→      if (dep > 0.35) {
  1051→  1051→        const bob = Math.sin(this.simTime * 1.7 + fl.pos.x) * 0.06;
  1052→  1052→        fl.pos.y += (s.eta + 0.15 + bob - fl.pos.y) * Math.min(1, dt * 5);
  1053→  1053→        const follow = isBoat ? 0.0 : 0.92; // moored boats only bob, don't drift
  1054→  1054→        fl.vel.x += (s.u * follow - fl.vel.x) * Math.min(1, dt * 2.2);
  1055→  1055→        fl.vel.z += (s.v * follow - fl.vel.z) * Math.min(1, dt * 2.2);
  1056→  1056→        fl.pos.x += fl.vel.x * dt;
  1057→  1057→        fl.pos.z += fl.vel.z * dt;
  1058→  1058→      } else {
  1059→  1059→        fl.vel.multiplyScalar(1 - Math.min(1, dt * 2.5));
  1060→  1060→        fl.pos.y += (bed + 0.3 - fl.pos.y) * Math.min(1, dt * 4);
  1061→  1061→      }
  1062→  1062→      fl.pos.x = clamp(fl.pos.x, 1, LX - 1);
  1063→  1063→      fl.pos.z = clamp(fl.pos.z, -LZ / 2 + 1, LZ / 2 - 1);
  1064→  1064→      const spd = Math.sqrt(fl.vel.x * fl.vel.x + fl.vel.z * fl.vel.z);
  1065→  1065→      fl.mesh.position.copy(fl.pos);
  1066→  1066→      fl.mesh.rotation.x = clamp(fl.vel.z * 0.1, -0.5, 0.5) + Math.sin(this.simTime * 2.1 + fl.pos.z) * 0.04;
  1067→  1067→      fl.mesh.rotation.z = clamp(-fl.vel.x * 0.1, -0.5, 0.5);
  1068→  1068→      if (!isBoat) fl.mesh.rotation.y += (0.2 + spd * 0.3) * dt;
  1069→  1069→    }
  1070→  1070→
  1071→  1071→    for (const h of this.houses) {
  1072→  1072→      if (h.prog >= 1) continue;
  1073→  1073→      f.sample(h.group.position.x, h.group.position.z, s);
  1074→  1074→      const wd = s.eta - h.ground;
  1075→  1075→      if (wd > 0.55) {
  1076→  1076→        h.prog = Math.min(1, h.prog + dt * 0.4);
  1077→  1077→        const p = h.prog;
  1078→  1078→        const dir = Math.atan2(s.v, s.u);
  1079→  1079→        h.group.rotation.x = -Math.cos(dir) * p * 1.35;
  1080→  1080→        h.group.rotation.z = Math.sin(dir) * p * 1.35;
  1081→  1081→        h.group.position.x += s.u * dt * 0.45 * p;
  1082→  1082→        h.group.position.z += s.v * dt * 0.45 * p;
  1083→  1083→        h.group.position.y = h.ground - p * 0.5;
  1084→  1084→      }
  1085→  1085→    }
  1086→  1086→
  1087→  1087→    for (const t of this.trees) {
  1088→  1088→      if (t.prog >= 1) continue;
  1089→  1089→      f.sample(t.group.position.x, t.group.position.z, s);
  1090→  1090→      const wd = s.eta - t.ground;
  1091→  1091→      if (wd > 0.7) {
  1092→  1092→        t.prog = Math.min(1, t.prog + dt * 0.6);
  1093→  1093→        const p = t.prog;
  1094→  1094→        const dir = Math.atan2(s.v, s.u);
  1095→  1095→        t.group.rotation.x = -Math.cos(dir) * p * 1.3;
  1096→  1096→        t.group.rotation.z = Math.sin(dir) * p * 1.3;
  1097→  1097→        t.group.position.x += s.u * dt * 0.25 * p;
  1098→  1098→        t.group.position.z += s.v * dt * 0.25 * p;
  1099→  1099→      }
  1100→  1100→    }
  1101→  1101→
  1102→  1102→    for (const c of this.chunks) {
  1103→  1103→      if (!c.active) continue;
  1104→  1104→      c.vel.y -= 11 * dt;
  1105→  1105→      c.mesh.position.addScaledVector(c.vel, dt);
  1106→  1106→      c.mesh.rotation.x += c.vel.z * dt * 0.15;
  1107→  1107→      c.mesh.rotation.z -= c.vel.x * dt * 0.15;
  1108→  1108→      const ground = bedAt(c.mesh.position.x, c.mesh.position.z) + 0.25;
  1109→  1109→      if (c.mesh.position.y < ground) {
  1110→  1110→        c.mesh.position.y = ground;
  1111→  1111→        c.vel.y *= -0.35;
  1112→  1112→        c.vel.x *= 0.6;
  1113→  1113→        c.vel.z *= 0.6;
  1114→  1114→        if (c.vel.lengthSq() < 0.4) c.active = false;
  1115→  1115→      }
  1116→  1116→      if (c.mesh.position.x > LX - 1) c.active = false;
  1117→  1117→    }
  1118→  1118→
  1119→  1119→    // keep the algal stain band near the waterline
  1120→  1120→    this.dam.stainBand.position.y = this.lastLevel - 0.6;
  1121→  1121→  }

  1123→  1123→  // ================================================================ stats
  1124→  1124→  private computeStats(): DamStats {
  1125→  1125→    const d = this.field.data;
  1126→  1126→    const dzC = LZ / DOWN_H;
  1127→  1127→    const gauges: GaugeStat[] = [];
  1128→  1128→    let vmax = 0;
  1129→  1129→    let level = 0;
  1130→  1130→    let floodedCells = 0;
  1131→  1131→    for (let i = 0; i < DOWN_W * DOWN_H; i++) {
  1132→  1132→      const k = i * 4;
  1133→  1133→      const sp = Math.sqrt(d[k + 1] * d[k + 1] + d[k + 2] * d[k + 2]);
  1134→  1134→      vmax = Math.max(vmax, sp);
  1135→  1135→      const col = i % DOWN_W;
  1136→  1136→      // reservoir pool level: deepest wet cells (depth > 0.5 m) so shelf puddles
  1137→  1137→      // cannot distort the stage reading
  1138→  1138→      if (col < 26 && d[k + 3] > 0.5) level = Math.max(level, d[k]);
  1139→  1139→      if (d[k + 3] > 0.12) floodedCells++;
  1140→  1140→    }
  1141→  1141→    // gauges
  1142→  1142→    for (let gi = 0; gi < GAUGES.length; gi++) {
  1143→  1143→      const g = GAUGES[gi];
  1144→  1144→      const gx = clamp(Math.round((g.x / LX) * DOWN_W), 1, DOWN_W - 2);
  1145→  1145→      let q = 0;
  1146→  1146→      let stage = 0;
  1147→  1147→      let gv = 0;
  1148→  1148→      let froude = 0;
  1149→  1149→      for (let j = 0; j < DOWN_H; j++) {
  1150→  1150→        const k = (j * DOWN_W + gx) * 4;
  1151→  1151→        const h = Math.max(d[k + 3], 0);
  1152→  1152→        const u = d[k + 1];
  1153→  1153→        const v = d[k + 2];
  1154→  1154→        q += u * h * dzC;
  1155→  1155→        stage = Math.max(stage, h);
  1156→  1156→        gv = Math.max(gv, Math.sqrt(u * u + v * v));
  1157→  1157→        if (h > 0.05) froude = Math.max(froude, Math.abs(u) / Math.sqrt(this.gravity * h));
  1158→  1158→      }
  1159→  1159→      q = Math.max(q, 0);
  1160→  1160→      if (q > this.gaugePeaks[gi]) this.gaugePeaks[gi] = q;
  1161→  1161→      if (this.gaugeArr[gi] === null && stage > 0.2 && g.x > DAM_X) this.gaugeArr[gi] = this.simTime;
  1162→  1162→      gauges.push({
  1163→  1163→        id: g.id,
  1164→  1164→        label: g.label,
  1165→  1165→        q,
  1166→  1166→        qPeak: this.gaugePeaks[gi],
  1167→  1167→        depth: stage,
  1168→  1168→        vel: gv,
  1169→  1169→        arrivalMin: this.gaugeArr[gi] !== null ? this.gaugeArr[gi]! * this.damProfile.timeMinPerSec : null,
  1170→  1170→        distKm: (g.x - DAM_X) * this.damProfile.lenScale / 1000,
  1171→  1171→      });
  1172→  1172→    }
  1173→  1173→    const frG = gauges[0] ? gauges[0].q : 0;
  1174→  1174→    if (frG > this.qPeak) this.qPeak = frG;
  1175→  1175→
  1176→  1176→    // level rate of rise (demo m/s, ema over stats ticks)
  1177→  1177→    const dtSimS = this.simTime - this.lastStatSimT;
  1178→  1178→    const riseMs = dtSimS > 0.05 && this.lastStatSimT > 0
  1179→  1179→      ? Math.max((level - this.lastLevel) / dtSimS, -2)
  1180→  1180→      : 0;
  1181→  1181→    this.levelRise = this.levelRise * 0.7 + riseMs * 0.3;
  1182→  1182→    this.lastStatSimT = this.simTime;
  1183→  1183→    this.lastLevel = level;
  1184→  1184→
  1185→  1185→    // scenario progress + stage
  1186→  1186→    let progress01 = 0;
  1187→  1187→    let stage = 0;
  1188→  1188→    if (this.scenario) {
