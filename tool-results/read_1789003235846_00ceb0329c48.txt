   222→   222→  // ================================================================ terrain mesh
   223→   223→  private buildTerrainMesh(): void {
   224→   224→    // bed grid on cell centers
   225→   225→    this.bedGrid = new Float32Array(NX * NZ);
   226→   226→    for (let j = 0; j < NZ; j++)
   227→   227→      for (let i = 0; i < NX; i++)
   228→   228→        this.bedGrid[j * NX + i] = bedAt((i + 0.5) * DX, (j + 0.5) * DZ - LZ / 2);
   229→   229→
   230→   230→    const geo = new THREE.PlaneGeometry(LX, LZ, NX - 1, NZ - 1);
   231→   231→    geo.rotateX(-Math.PI / 2);
   232→   232→    geo.translate(LX / 2, 0, 0);
   233→   233→    const pos = geo.attributes.position as THREE.BufferAttribute;
   234→   234→    const colors = new Float32Array(pos.count * 3);
   235→   235→    const col = new THREE.Color();
   236→   236→    const tc = { r: 0, g: 0, b: 0 };
   237→   237→    for (let i = 0; i < pos.count; i++) {
   238→   238→      const x = pos.getX(i);
   239→   239→      const z = pos.getZ(i);
   240→   240→      const y = this.sampleBed(x, z);
   241→   241→      pos.setY(i, y);
   242→   242→      const e = 1.2;
   243→   243→      const sx = (this.sampleBed(x + e, z) - this.sampleBed(x - e, z)) / (2 * e);
   244→   244→      const sz = (this.sampleBed(x, z + e) - this.sampleBed(x, z - e)) / (2 * e);
   245→   245→      const slope = Math.sqrt(sx * sx + sz * sz);
   246→   246→      terrainColor(x, z, y, slope, tc);
   247→   247→      col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
   248→   248→      colors[i * 3] = col.r;
   249→   249→      colors[i * 3 + 1] = col.g;
   250→   250→      colors[i * 3 + 2] = col.b;
   251→   251→    }
   252→   252→    geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   253→   253→    geo.computeVertexNormals();
   254→   254→    const mesh = new THREE.Mesh(
   255→   255→      geo,
   256→   256→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.96, metalness: 0, envMapIntensity: 0.18 }),
   257→   257→    );
   258→   258→    mesh.receiveShadow = true;
   259→   259→    this.scene.add(mesh);
   260→   260→
   261→   261→    // surround plane so the domain doesn't float in the void
   262→   262→    const surround = new THREE.Mesh(
   263→   263→      new THREE.PlaneGeometry(3000, 3000),
   264→   264→      new THREE.MeshStandardMaterial({ color: 0xa8ad97, roughness: 1, metalness: 0, envMapIntensity: 0.1 }),
   265→   265→    );
   266→   266→    surround.rotation.x = -Math.PI / 2;
   267→   267→    surround.position.set(LX / 2, 1.2, 0);
   268→   268→    this.scene.add(surround);
   269→   269→  }

   271→   271→  private sampleBed(x: number, z: number): number {
   272→   272→    const fu = clamp((x / DX) - 0.5, 0, NX - 1.001);
   273→   273→    const fv = clamp((z + LZ / 2) / DZ - 0.5, 0, NZ - 1.001);
   274→   274→    const i = Math.floor(fu), j = Math.floor(fv);
   275→   275→    const fx = fu - i, fy = fv - j;
   276→   276→    const i1 = Math.min(i + 1, NX - 1), j1 = Math.min(j + 1, NZ - 1);
   277→   277→    const g = this.bedGrid;
   278→   278→    const a = g[j * NX + i] + (g[j * NX + i1] - g[j * NX + i]) * fx;
   279→   279→    const b = g[j1 * NX + i] + (g[j1 * NX + i1] - g[j1 * NX + i]) * fx;
   280→   280→    return a + (b - a) * fy;
   281→   281→  }

   476→   476→  private simSubstep(dt: number): void {
   477→   477→    const eu = (this.etaPass.material as THREE.ShaderMaterial).uniforms;
   478→   478→    eu.uState.value = this.curState.texture;
   479→   479→    eu.uDt.value = dt;
   480→   480→    eu.uG.value = this.gravity;
   481→   481→    eu.uSrcRate.value = this.srcRate;
   482→   482→    this.renderTo(this.etaPass, this.rtEta);
   483→   483→
   484→   484→    const vu = (this.velPass.material as THREE.ShaderMaterial).uniforms;
   485→   485→    vu.uState.value = this.curState.texture;
   486→   486→    vu.uEtaNew.value = this.rtEta.texture;
   487→   487→    vu.uDt.value = dt;
   488→   488→    vu.uG.value = this.gravity;
   489→   489→    this.renderTo(this.velPass, this.nextState);
   490→   490→    this.swapState();
   491→   491→
   492→   492→    if (this.foamOn) {
   493→   493→      const fu = (this.foamPass.material as THREE.ShaderMaterial).uniforms;
   494→   494→      fu.uState.value = this.curState.texture;
   495→   495→      fu.uFoam.value = this.curFoam.texture;
   496→   496→      fu.uDt.value = dt;
   497→   497→      fu.uG.value = this.gravity;
   498→   498→      this.renderTo(this.foamPass, this.nextFoam);
   499→   499→      const t = this.curFoam;
   500→   500→      this.curFoam = this.nextFoam;
   501→   501→      this.nextFoam = t;
   502→   502→    }
   503→   503→
   504→   504→    if (this.sprayOn) {
   505→   505→      const su = (this.sprayPass.material as THREE.ShaderMaterial).uniforms;
   506→   506→      su.uSpray.value = this.curSpray.texture;
   507→   507→      su.uState.value = this.curState.texture;
   508→   508→      su.uDt.value = dt;
   509→   509→      su.uTime.value = this.simTime;
   510→   510→      this.renderTo(this.sprayPass, this.nextSpray);
   511→   511→      const t = this.curSpray;
   512→   512→      this.curSpray = this.nextSpray;
   513→   513→      this.nextSpray = t;
   514→   514→    }
   515→   515→  }

   620→   620→  private tick = (): void => {
   621→   621→    this.rafId = requestAnimationFrame(this.tick);
   622→   622→    const now = performance.now();
   623→   623→    const dtReal = Math.min((now - this.lastFrameTime) / 1000, 0.05);
   624→   624→    this.lastFrameTime = now;
   625→   625→    this.fpsEma = this.fpsEma * 0.95 + (1 / Math.max(dtReal, 1e-4)) * 0.05;
   626→   626→    const dtSim = this.paused ? 0 : dtReal * this.timeScale;
   627→   627→
   628→   628→    if (dtSim > 0) {
   629→   629→      // scenario animation timers
   630→   630→      if (this.breachT !== null) {
   631→   631→        this.breachT += dtSim;
   632→   632→        const tau = 2.6 / this.breachRate;
   633→   633→        const p = 1 - Math.exp(-Math.pow(this.breachT / tau, 1.7));
   634→   634→        this.breachElev = CREST + (BREACH_BOTTOM - CREST) * p;
   635→   635→        this.dam.breachGroup.position.y = this.breachElev - CREST;
   636→   636→        this.dam.breachGroup.rotation.z = -0.05 * p;
   637→   637→        this.structDirty = true;
   638→   638→        if (p > 0.995) this.breachT = null;
   639→   639→      }
   640→   640→      if (this.gateT !== null && this.gateT < 1) {
   641→   641→        this.gateT = Math.min(1, this.gateT + dtSim / 2.5);
   642→   642→        const e = easeInOut(this.gateT);
   643→   643→        this.gateElev = SPILL_CREST_CLOSED + (GATE_OPEN_ELEV - SPILL_CREST_CLOSED) * e;
   644→   644→        this.dam.gate.position.y = 18.85 + 6.9 * e;
   645→   645→        this.structDirty = true;
   646→   646→      }
   647→   647→      if (this.overtopping) {
   648→   648→        this.overtopT += dtSim;
   649→   649→      }
   650→   650→      const curInflow = this.overtopping
   651→   651→        ? this.inflowQ + 272 * Math.min(this.overtopT / 18, 1)
   652→   652→        : this.inflowQ;
   653→   653→      this.srcRate = curInflow / (this.stripCells * DX * DZ);
   654→   654→
   655→   655→      if (this.structDirty) {
   656→   656→        this.applyStruct(this.breachElev, this.gateElev);
   657→   657→        this.structDirty = false;
   658→   658→      }
   659→   659→
   660→   660→      const sub = clamp(Math.ceil(dtSim / DT_MAX), 1, 8);
   661→   661→      const h = dtSim / sub;
   662→   662→      for (let i = 0; i < sub; i++) this.simSubstep(h);
   663→   663→      this.simTime += dtSim;
   664→   664→      this.updateProps(dtSim);
   665→   665→    }
   666→   666→
   667→   667→    // readback for CPU physics + stats
   668→   668→    if (this.frame % 2 === 0) {
   669→   669→      (this.downPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   670→   670→      this.renderTo(this.downPass, this.rtDown);
   671→   671→      this.renderer.readRenderTargetPixels(this.rtDown, 0, 0, DOWN_W, DOWN_H, this.field.data);
   672→   672→    }
   673→   673→
   674→   674→    this.updateCamera(dtReal);
   675→   675→
   676→   676→    // shader uniforms
   677→   677→    this.waterMat.uniforms.uState.value = this.curState.texture;
   678→   678→    this.waterMat.uniforms.uFoam.value = this.curFoam.texture;
   679→   679→    this.waterMat.uniforms.uTime.value = this.simTime;
   680→   680→    this.waterMat.uniforms.uCamPos.value.copy(this.camera.position);
   681→   681→    this.sprayMat.uniforms.uSpray.value = this.curSpray.texture;
   682→   682→    this.sprayMat.uniforms.uState.value = this.curState.texture;
   683→   683→    (this.sprayPass.material as THREE.ShaderMaterial).uniforms.uState.value = this.curState.texture;
   684→   684→
   685→   685→    // render with camera shake
   686→   686→    let shook: THREE.Vector3 | null = null;
   687→   687→    if (this.shake > 0.002) {
   688→   688→      shook = new THREE.Vector3(
   689→   689→        (Math.random() - 0.5) * this.shake * 0.9,
   690→   690→        (Math.random() - 0.5) * this.shake * 0.9,
   691→   691→        (Math.random() - 0.5) * this.shake * 0.9,
   692→   692→      );
   693→   693→      this.camera.position.add(shook);
   694→   694→    }
   695→   695→    this.renderer.render(this.scene, this.camera);
   696→   696→    if (shook) {
   697→   697→      this.camera.position.sub(shook);
   698→   698→      this.shake *= Math.exp(-dtReal * 1.15);
   699→   699→    }
   700→   700→
   701→   701→    // stats ~10 Hz
   702→   702→    if (this.frame % 6 === 0 && this.onStats) {
   703→   703→      const s = this.computeStats();
   704→   704→      this.onStats(s);
   705→   705→      const v01 = clamp(s.vmax / 10, 0, 1);
   706→   706→      this.audio.setIntensity(v01);
   707→   707→    }
   708→   708→
   709→   709→    if (this.frame === 3) {
   710→   710→      (window as unknown as Record<string, unknown>).__damReady = true;
   711→   711→    }
   712→   712→    this.frame++;
   713→   713→  };

   715→   715→  // ================================================================ props update
   716→   716→  private updateProps(dt: number): void {
   717→   717→    const f = this.field;
   718→   718→    const s = this.sample;
   719→   719→
   720→   720→    for (const b of this.barrels) {
   721→   721→      f.sample(b.pos.x, b.pos.z, s);
   722→   722→      const bed = bedAt(b.pos.x, b.pos.z);
   723→   723→      const dep = s.eta - bed;
   724→   724→      if (dep > 0.4) {
   725→   725→        const bob = Math.sin(this.simTime * 1.7 + b.pos.x) * 0.06;
   726→   726→        b.pos.y += (s.eta + 0.15 + bob - b.pos.y) * Math.min(1, dt * 5);
   727→   727→        b.vel.x += (s.u * 0.92 - b.vel.x) * Math.min(1, dt * 2.2);
   728→   728→        b.vel.z += (s.v * 0.92 - b.vel.z) * Math.min(1, dt * 2.2);
   729→   729→        b.pos.x += b.vel.x * dt;
   730→   730→        b.pos.z += b.vel.z * dt;
   731→   731→      } else {
   732→   732→        b.vel.multiplyScalar(1 - Math.min(1, dt * 2.5));
   733→   733→        b.pos.y += (bed + 0.3 - b.pos.y) * Math.min(1, dt * 4);
   734→   734→      }
   735→   735→      b.pos.x = clamp(b.pos.x, 1, LX - 1);
   736→   736→      b.pos.z = clamp(b.pos.z, -LZ / 2 + 1, LZ / 2 - 1);
   737→   737→      const spd = Math.sqrt(b.vel.x * b.vel.x + b.vel.z * b.vel.z);
   738→   738→      b.mesh.position.copy(b.pos);
   739→   739→      b.mesh.rotation.x = clamp(b.vel.z * 0.1, -0.5, 0.5) + Math.sin(this.simTime * 2.1 + b.pos.z) * 0.04;
   740→   740→      b.mesh.rotation.z = clamp(-b.vel.x * 0.1, -0.5, 0.5);
   741→   741→      b.mesh.rotation.y += (0.2 + spd * 0.3) * dt;
   742→   742→    }
   743→   743→
   744→   744→    for (const h of this.houses) {
   745→   745→      if (h.prog >= 1) continue;
   746→   746→      f.sample(h.group.position.x, h.group.position.z, s);
   747→   747→      const wd = s.eta - h.ground;
   748→   748→      if (wd > 0.55) {
   749→   749→        h.prog = Math.min(1, h.prog + dt * 0.4);
   750→   750→        const p = h.prog;
   751→   751→        const dir = Math.atan2(s.v, s.u);
   752→   752→        h.group.rotation.x = -Math.cos(dir) * p * 1.35;
   753→   753→        h.group.rotation.z = Math.sin(dir) * p * 1.35;
   754→   754→        h.group.position.x += s.u * dt * 0.45 * p;
   755→   755→        h.group.position.z += s.v * dt * 0.45 * p;
   756→   756→        h.group.position.y = h.ground - p * 0.5;
   757→   757→      }
   758→   758→    }
   759→   759→
   760→   760→    for (const t of this.trees) {
   761→   761→      if (t.prog >= 1) continue;
   762→   762→      f.sample(t.group.position.x, t.group.position.z, s);
   763→   763→      const wd = s.eta - t.ground;
   764→   764→      if (wd > 0.7) {
   765→   765→        t.prog = Math.min(1, t.prog + dt * 0.6);
   766→   766→        const p = t.prog;
   767→   767→        const dir = Math.atan2(s.v, s.u);
   768→   768→        t.group.rotation.x = -Math.cos(dir) * p * 1.3;
   769→   769→        t.group.rotation.z = Math.sin(dir) * p * 1.3;
   770→   770→        t.group.position.x += s.u * dt * 0.25 * p;
   771→   771→        t.group.position.z += s.v * dt * 0.25 * p;
   772→   772→      }
   773→   773→    }
   774→   774→
   775→   775→    for (const c of this.chunks) {
   776→   776→      if (!c.active) continue;
   777→   777→      c.vel.y -= 11 * dt;
   778→   778→      c.mesh.position.addScaledVector(c.vel, dt);
   779→   779→      c.mesh.rotation.x += c.vel.z * dt * 0.15;
   780→   780→      c.mesh.rotation.z -= c.vel.x * dt * 0.15;
   781→   781→      const ground = bedAt(c.mesh.position.x, c.mesh.position.z) + 0.25;
   782→   782→      if (c.mesh.position.y < ground) {
   783→   783→        c.mesh.position.y = ground;
   784→   784→        c.vel.y *= -0.35;
   785→   785→        c.vel.x *= 0.6;
   786→   786→        c.vel.z *= 0.6;
   787→   787→        if (c.vel.lengthSq() < 0.4) c.active = false;
   788→   788→      }
   789→   789→      if (c.mesh.position.x > LX - 1) c.active = false;
   790→   790→    }
   791→   791→  }

   793→   793→  // ================================================================ stats
   794→   794→  private computeStats(): DamStats {
   795→   795→    const d = this.field.data;
   796→   796→    const dzC = LZ / DOWN_H;
   797→   797→    const gx = 37;
   798→   798→    let q = 0;
   799→   799→    let stage = 0;
   800→   800→    let froude = 0;
   801→   801→    for (let j = 0; j < DOWN_H; j++) {
   802→   802→      const k = (j * DOWN_W + gx) * 4;
   803→   803→      const h = Math.max(d[k + 3], 0);
   804→   804→      const u = d[k + 1];
   805→   805→      q += u * h * dzC;
   806→   806→      stage = Math.max(stage, h);
   807→   807→      if (h > 0.05) {
   808→   808→        froude = Math.max(froude, Math.abs(u) / Math.sqrt(this.gravity * h));
   809→   809→      }
   810→   810→    }
   811→   811→    let vmax = 0;
   812→   812→    let level = 0;
   813→   813→    for (let i = 0; i < DOWN_W * DOWN_H; i++) {
   814→   814→      const k = i * 4;
   815→   815→      const sp = Math.sqrt(d[k + 1] * d[k + 1] + d[k + 2] * d[k + 2]);
   816→   816→      vmax = Math.max(vmax, sp);
   817→   817→      if (i % DOWN_W < 18 && d[k + 3] > 0.1) level = Math.max(level, d[k]);
   818→   818→    }
   819→   819→    if (q > this.qPeak) this.qPeak = q;
   820→   820→    return {
   821→   821→      t: this.simTime,
   822→   822→      q: Math.max(q, 0),
   823→   823→      qPeak: this.qPeak,
   824→   824→      level,
   825→   825→      vmax,
   826→   826→      froude,
   827→   827→      fps: this.fpsEma,
   828→   828→      phase: this.phase,
   829→   829→      overtopping: this.overtopping,
   830→   830→    };
   831→   831→  }

   833→   833→  // ================================================================ camera
   834→   834→  private updateCamera(dt: number): void {
   835→   835→    if (this.cinematic) {
   836→   836→      this.cinAngle += dt * 0.05;
   837→   837→      const R = 95;
   838→   838→      this.camera.position.set(
   839→   839→        105 + R * Math.cos(this.cinAngle),
   840→   840→        42 + 10 * Math.sin(this.cinAngle * 0.7),
   841→   841→        R * Math.sin(this.cinAngle),
   842→   842→      );
   843→   843→      this.camera.lookAt(105, 14, 0);
   844→   844→      return;
   845→   845→    }
   846→   846→    if (this.tweenT < 1) {
   847→   847→      this.tweenT = Math.min(1, this.tweenT + dt / 1.2);
   848→   848→      const e = easeInOut(this.tweenT);
   849→   849→      this.camera.position.lerpVectors(this.tweenFrom, this.tweenTo, e);
   850→   850→      this.controls.target.lerpVectors(this.tweenFromT, this.tweenToT, e);
   851→   851→    }
   852→   852→    this.controls.update();
   853→   853→  }
