    14→function hashRnd(i: number): number {
    15→  const h = Math.sin(i * 127.1 + 311.7) * 43758.5453;
    16→  return h - Math.floor(h);
    17→}

    19→function canvasTex(w: number, h: number, draw: (c: CanvasRenderingContext2D) => void): THREE.Texture {
    20→  const cv = document.createElement('canvas');
    21→  cv.width = w;
    22→  cv.height = h;
    23→  const ctx = cv.getContext('2d')!;
    24→  draw(ctx);
    25→  const tex = new THREE.CanvasTexture(cv);
    26→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    27→  tex.colorSpace = THREE.SRGBColorSpace;
    28→  tex.anisotropy = 4;
    29→  return tex;
    30→}

    52→function makePylon(h: number): THREE.Group {
    53→  const g = new THREE.Group();
    54→  const mat = matGalvanised();
    55→  const base = 1.5;
    56→  const top = 0.45;
    57→  // 4 splayed legs
    58→  for (const sx of [-1, 1]) {
    59→    for (const sz of [-1, 1]) {
    60→      const leg = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.09, h, 5), mat);
    61→      const lx = (sx * (base + (top - base) * 0.5)) / 2;
    62→      const lz = (sz * (base + (top - base) * 0.5)) / 2;
    63→      leg.position.set(lx, h / 2, lz);
    64→      leg.rotation.z = -sx * 0.115;
    65→      leg.rotation.x = sz * 0.115;
    66→      leg.castShadow = true;
    67→      g.add(leg);
    68→    }
    69→  }
    70→  // cross-braces on both faces (2 levels)
    71→  for (const lvl of [0.3, 0.62]) {
    72→    for (const sz of [-1, 1]) {
    73→      const brace = new THREE.Mesh(new THREE.BoxGeometry(base * (1 - lvl * 0.55), 0.06, 0.06), mat);
    74→      brace.position.set(0, h * lvl, (sz * base * (1 - lvl * 0.55)) / 2);
    75→      brace.rotation.z = sz * 0.5;
    76→      g.add(brace);
    77→    }
    78→  }
    79→  // cross-arms (two levels) + peak
    80→  const arm1 = new THREE.Mesh(new THREE.BoxGeometry(3.6, 0.09, 0.09), mat);
    81→  arm1.position.y = h * 0.8;
    82→  const arm2 = new THREE.Mesh(new THREE.BoxGeometry(2.7, 0.09, 0.09), mat);
    83→  arm2.position.y = h * 0.93;
    84→  const peak = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.05, h * 0.1, 5), mat);
    85→  peak.position.y = h * 1.04;
    86→  arm1.castShadow = arm2.castShadow = true;
    87→  g.add(arm1, arm2, peak);
    88→  return g;
    89→}

    91→function makeCable(a: THREE.Vector3, b: THREE.Vector3, sag: number, mat: THREE.Material): THREE.Mesh {
    92→  const mid = a.clone().add(b).multiplyScalar(0.5);
    93→  mid.y -= sag;
    94→  const curve = new THREE.QuadraticBezierCurve3(a, mid, b);
    95→  const geo = new THREE.TubeGeometry(curve, 20, 0.035, 5, false);
    96→  const m = new THREE.Mesh(geo, mat);
    97→  m.castShadow = false;
    98→  return m;
    99→}

   393→function arcPts(r: number, half: number, n: number): [number, number][] {
   394→  const pts: [number, number][] = [];
   395→  for (let i = 0; i <= n; i++) {
   396→    const a = -half + (2 * half * i) / n;
   397→    pts.push([TOWN_C[0] + r * Math.cos(a), TOWN_C[1] + r * Math.sin(a)]);
   398→  }
   399→  return pts;
   400→}
   401→function rayPts(a: number, r0: number, r1: number, n: number): [number, number][] {
   402→  const pts: [number, number][] = [];
   403→  for (let i = 0; i <= n; i++) {
   404→    const r = r0 + ((r1 - r0) * i) / n;
   405→    pts.push([TOWN_C[0] + r * Math.cos(a), TOWN_C[1] + r * Math.sin(a)]);
   406→  }
   407→  return pts;
   408→}

   414→  for (let i = 0; i < pts.length - 1; i++)

   443→function distToPaths(x: number, z: number): number {
   444→  let best = 1e9;
   445→  for (const s of CLEAR_SEGS) {
   446→    const dx = s.bx - s.ax, dz = s.bz - s.az;
   447→    const t = clamp(((x - s.ax) * dx + (z - s.az) * dz) / (dx * dx + dz * dz || 1), 0, 1);
   448→    const d = Math.hypot(x - (s.ax + dx * t), z - (s.az + dz * t));
   449→    if (d < best) best = d;
   450→  }
   451→  return best;
   452→}

   462→function clearOfSites(x: number, z: number, m = 1.4): boolean {
   463→  for (const [cx, cz, w, d] of CLEAR_RECTS)
   464→    if (Math.abs(x - cx) < w / 2 + m && Math.abs(z - cz) < d / 2 + m) return false;
   465→  return true;
   466→}

   468→// deck ribbon following an arbitrary elevation profile; returns the sampled
   469→// centreline so callers can hang piers / dashes off it
   470→function profileRibbon(
   471→  group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material,
   472→  yAt: (x: number, z: number, t: number) => number,
   473→  castShadow = true,
   474→): { x: number; z: number; y: number }[] {
   475→  const n = Math.max(10, pts.length * 5);
   476→  const P: { x: number; z: number; y: number }[] = [];
   477→  for (let k = 0; k <= n; k++) {
   478→    const t = k / n;
   479→    const seg = Math.min(Math.floor(t * (pts.length - 1)), pts.length - 2);
   480→    const lt = t * (pts.length - 1) - seg;
   481→    const x = pts[seg][0] + (pts[seg + 1][0] - pts[seg][0]) * lt;
   482→    const z = pts[seg][1] + (pts[seg + 1][1] - pts[seg][1]) * lt;
   483→    P.push({ x, z, y: yAt(x, z, t) });
   484→  }
   485→  // smooth the elevation profile (endpoints pinned)
   486→  for (let p = 0; p < 4; p++)
   487→    for (let i = 1; i < P.length - 1; i++)
   488→      P[i].y = (P[i - 1].y + 2 * P[i].y + P[i + 1].y) / 4;
   489→  const pos: number[] = [];
   490→  const idx: number[] = [];
   491→  for (let i = 0; i < P.length; i++) {
   492→    const a = P[Math.max(i - 1, 0)];
   493→    const b = P[Math.min(i + 1, P.length - 1)];
   494→    const dx = b.x - a.x, dz = b.z - a.z;
   495→    const dl = Math.hypot(dx, dz) || 1;
   496→    const nx = (-dz / dl) * w * 0.5, nz = (dx / dl) * w * 0.5;
   497→    pos.push(P[i].x - nx, P[i].y, P[i].z - nz, P[i].x + nx, P[i].y, P[i].z + nz);
   498→    if (i > 0) {
   499→      const q = (i - 1) * 2;
   500→      idx.push(q, q + 2, q + 1, q + 1, q + 2, q + 3);
   501→    }
   502→  }
   503→  const geo = new THREE.BufferGeometry();
   504→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   505→  geo.setIndex(idx);
   506→  geo.computeVertexNormals();
   507→  const mesh = new THREE.Mesh(geo, mat);
   508→  mesh.castShadow = castShadow;
   509→  mesh.receiveShadow = true;
   510→  group.add(mesh);
   511→  return P;
   512→}

   514→function deckPiers(
   515→  group: THREE.Group, P: { x: number; z: number; y: number }[],
   516→  r: number, mat: THREE.Material, minH = 1.4,
   517→): void {
   518→  for (let i = 3; i < P.length - 2; i += 4) {
   519→    const g = bedAt(P[i].x, P[i].z);
   520→    const hh = P[i].y - g;
   521→    if (hh < minH) continue;
   522→    const pier = new THREE.Mesh(new THREE.CylinderGeometry(r * 0.78, r, hh - 0.12, 8), mat);
   523→    pier.position.set(P[i].x, g + (hh - 0.12) / 2, P[i].z);
   524→    pier.castShadow = true;
   525→    group.add(pier);
   526→  }
   527→}

   529→// elevated expressway: deck on pylons following ground + hAbove, edge beams,
   530→// centre dashes, plus sloped ramps down to street level
   531→function buildExpressway(group: THREE.Group): void {
   532→  const deckMat = new THREE.MeshStandardMaterial({ color: 0x58585e, roughness: 0.88, metalness: 0.04, side: THREE.DoubleSide });
   533→  const beamMat = matConcreteDark();
   534→  const lineMat = new THREE.MeshStandardMaterial({ color: 0xd8d2b8, roughness: 0.8 });
   535→
   536→  const road = (pts: [number, number][], h0: number, h1: number, w: number): void => {
   537→    const P = profileRibbon(group, pts, w, deckMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t);
   538→    // edge beams slightly under the deck
   539→    profileRibbon(group, pts, w + 0.7, beamMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t - 0.16, false);
   540→    // pylons where the deck is well above ground (never in the river channel)
   541→    for (let i = 3; i < P.length - 2; i += 4) {
   542→      const g = bedAt(P[i].x, P[i].z);
   543→      const hh = P[i].y - g;
   544→      if (hh < 1.8 || Math.abs(P[i].z) < 13) continue;
   545→      const pylon = new THREE.Mesh(new THREE.BoxGeometry(0.72, hh - 0.1, 0.72), beamMat);
   546→      pylon.position.set(P[i].x, g + (hh - 0.1) / 2, P[i].z);
   547→      pylon.castShadow = true;
   548→      group.add(pylon);
   549→      const cap = new THREE.Mesh(new THREE.BoxGeometry(w + 0.5, 0.3, 1.0), beamMat);
   550→      cap.position.set(P[i].x, P[i].y - 0.3, P[i].z);
   551→      group.add(cap);
   552→    }
   553→    // centre dashes (instanced)
   554→    const dashT: THREE.Matrix4[] = [];
   555→    for (let i = 2; i < P.length - 2; i += 3) {
   556→      const a = P[i], b = P[i + 1];
   557→      const yaw = Math.atan2(b.x - a.x, b.z - a.z);
   558→      const mm = new THREE.Matrix4();
   559→      mm.makeRotationY(yaw);
   560→      mm.setPosition(a.x, a.y + 0.04, a.z);
   561→      dashT.push(mm);
   562→    }
   563→    if (dashT.length) {
   564→      const dashMesh = new THREE.InstancedMesh(new THREE.BoxGeometry(0.16, 0.04, 1.7), lineMat, dashT.length);
   565→      dashT.forEach((mm, i) => dashMesh.setMatrixAt(i, mm));
   566→      group.add(dashMesh);
   567→    }
   568→  };
   569→
   570→  road(EXP_A, 5.6, 5.6, 5.0);   // +z bank elevated highway
   571→  road(EXP_B, 5.2, 5.2, 5.0);   // signature river-crossing elevated highway
   572→  road(RAMP_A1, 5.6, 0.16, 3.6); // down-ramp to the outer ring
   573→  road(RAMP_A2, 5.6, 0.16, 3.6); // down-ramp near the dam bench
   574→  road(RAMP_B1, 5.2, 0.16, 3.6); // down-ramp on the left bank
   575→  road(RAMP_AB, 5.6, 5.2, 3.6);  // flyover connector A → B
   576→}

   578→function flatRibbon(group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material, lift = 0.14): void {
   579→  const pos: number[] = [];
   580→  const idx: number[] = [];
   581→  for (let i = 0; i < pts.length - 1; i++) {
   582→    for (let s = 0; s < 3; s++) {
   583→      const t0 = s / 3, t1 = (s + 1) / 3;
   584→      const x0 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t0;
   585→      const z0 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t0;
   586→      const x1 = pts[i][0] + (pts[i + 1][0] - pts[i][0]) * t1;
   587→      const z1 = pts[i][1] + (pts[i + 1][1] - pts[i][1]) * t1;
   588→      const dx = z1 - z0, dz = -(x1 - x0);
   589→      const dl = Math.hypot(dx, dz) || 1;
   590→      const nx = (dx / dl) * w * 0.5, nz = (dz / dl) * w * 0.5;
   591→      const a = bedAt(x0 - nx, z0 - nz) + lift;
   592→      const b = bedAt(x0 + nx, z0 + nz) + lift;
   593→      const c = bedAt(x1 - nx, z1 - nz) + lift;
   594→      const d = bedAt(x1 + nx, z1 + nz) + lift;
   595→      const base = pos.length / 3;
   596→      pos.push(x0 - nx, a, z0 - nz, x0 + nx, b, z0 + nz, x1 - nx, c, z1 - nz, x1 + nx, d, z1 + nz);
   597→      idx.push(base, base + 2, base + 1, base + 1, base + 2, base + 3);
   598→    }
   599→  }
   600→  const geo = new THREE.BufferGeometry();
   601→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   602→  geo.setIndex(idx);
   603→  geo.computeVertexNormals();
   604→  const mesh = new THREE.Mesh(geo, mat);
   605→  mesh.receiveShadow = true;
   606→  group.add(mesh);
   607→}

   707→export function buildTown(): { group: THREE.Group } {
   708→  const group = new THREE.Group();
   709→  const asphalt = new THREE.MeshStandardMaterial({ color: 0x45454a, roughness: 0.95 });
   710→  const paving = new THREE.MeshStandardMaterial({ color: 0x9b968c, roughness: 0.95 });
   711→
   712→  // old-town grid streets
   713→  for (const st of TOWN_STREETS) flatRibbon(group, st.pts, st.w, asphalt);
   714→
   715→  // ---- ring boulevards — arched decks over the river channel (bridges)
   716→  RING_PTS.forEach((pts, ri) => {
   717→    const P = profileRibbon(group, pts, RINGS[ri].w, asphalt, (x, z) => {
   718→      const g = bedAt(x, z);
   719→      const arch = Math.max(0, 1 - Math.abs(z) / 14);
   720→      return Math.max(g + 0.16, bedAt(x, 0) + 3.6 + 1.15 * arch * arch);
   721→    });
   722→    deckPiers(group, P, 0.4, matConcreteDark());
   723→  });
   724→
   725→  // radial avenues linking the rings to the old core
   726→  for (const pts of RADIAL_PTS) flatRibbon(group, pts, 2.5, asphalt);
   727→
   728→  // riverside promenades (trees + lamps line these)
   729→  flatRibbon(group, [[124, 12.4], [140, 12.2], [158, 12.2], [176, 12.4], [190, 12.8]], 1.9, paving);
   730→  flatRibbon(group, [[124, -12.4], [140, -12.3], [158, -12.3], [176, -12.5], [190, -12.9]], 1.9, paving);
   731→
   732→  // ---- elevated expressway + interchange
   733→  buildExpressway(group);
   734→
   735→  // ---- dense districts (instanced, three building classes)
   736→  const bldTex = makeBuildingTex();
   737→  const lowMat = new THREE.MeshStandardMaterial({ map: bldTex, roughness: 0.85, metalness: 0.05 });
   738→  const midMat = new THREE.MeshStandardMaterial({ map: makeTowerTex(false), roughness: 0.68, metalness: 0.1 });
   739→  const glassMat = new THREE.MeshStandardMaterial({ map: makeTowerTex(true), roughness: 0.38, metalness: 0.35, envMapIntensity: 0.35 });
   740→  interface BSpot { x: number; z: number; w: number; d: number; h: number; rot: number; tone: number }
   741→  const spotsL: BSpot[] = [];
   742→  const spotsM: BSpot[] = [];
   743→  const spotsH: BSpot[] = [];
   744→  const tones = [0xd9cdb4, 0xcbb99b, 0xd8d2c4, 0xc2b49a, 0xd3c1a6, 0xbfae94, 0xded6c6, 0xb7a68c];
   745→  const glassTones = [0x8fa4b2, 0xa3b8c4, 0x7f98a8, 0xb0c2cc];
   746→  // deterministic scatter across the whole valley floor — dense ring districts
   747→  let seed = 1;
   748→  const nextRnd = () => hashRnd(seed++ * 12.9898);
   749→  for (let gx = 121; gx <= 189; gx += 3.9) {
   750→    for (const zs of [1, -1]) {
   751→      for (let g = 0; g < 11; g++) {
   752→        const z = zs * (13.9 + g * 3.6 + nextRnd() * 1.6);
   753→        const x = gx + nextRnd() * 2.2;
   754→        if (Math.abs(z) > 49.5 || x > 189.5) continue;
   755→        const r = Math.hypot(x - TOWN_C[0], z);
   756→        if (r < 15.5) continue;
   757→        if (distToPaths(x, z) < 3.2) continue;
   758→        if (!clearOfSites(x, z)) continue;
   759→        const ground = bedAt(x, z);
   760→        if (ground < 3.3 || ground > 16.5) continue;
   761→        const slope = Math.abs(bedAt(x + 2, z) - ground) + Math.abs(bedAt(x, z + 2) - ground);
   762→        if (slope > 2.1) continue;
   763→        if (nextRnd() < 0.16) continue;
   764→        // downtown weight — towers cluster on the inner rings near the dam
   765→        const core = clamp(1 - (r - 16) / 27, 0, 1);
   766→        const roll = nextRnd();
   767→        const rot = (nextRnd() - 0.5) * 0.5;
   768→        if (core > 0.4 && roll < 0.07 + 0.2 * core) {
   769→          spotsH.push({ x, z, w: 3.9 + nextRnd() * 1.6, d: 3.5 + nextRnd() * 1.5, h: 13.5 + nextRnd() * 9.5, rot, tone: Math.floor(nextRnd() * glassTones.length) });
   770→        } else if (roll < 0.5) {
   771→          spotsM.push({ x, z, w: 3.1 + nextRnd() * 1.4, d: 2.9 + nextRnd() * 1.3, h: 6.8 + nextRnd() * 6.2, rot, tone: Math.floor(nextRnd() * tones.length) });
   772→        } else {
   773→          spotsL.push({ x, z, w: 2.6 + nextRnd() * 1.6, d: 2.4 + nextRnd() * 1.5, h: 2.8 + nextRnd() * 3.4, rot, tone: Math.floor(nextRnd() * tones.length) });
   774→        }
   775→      }
   776→    }
   777→  }
   778→  const boxGeo = new THREE.BoxGeometry(1, 1, 1);
   779→  const m4 = new THREE.Matrix4();
   780→  const q = new THREE.Quaternion();
   781→  const eu = new THREE.Euler();
   782→  const col = new THREE.Color();
   783→  const buildClass = (spots: BSpot[], mat: THREE.Material, tint: number[], roofsToo: boolean): void => {
   784→    const inst = new THREE.InstancedMesh(boxGeo, mat, spots.length);
   785→    spots.forEach((s, i) => {
   786→      eu.set(0, s.rot, 0);
   787→      q.setFromEuler(eu);
   788→      m4.compose(new THREE.Vector3(s.x, bedAt(s.x, s.z) + s.h / 2 - 0.12, s.z), q, new THREE.Vector3(s.w, s.h, s.d));
   789→      inst.setMatrixAt(i, m4);
   790→      inst.setColorAt(i, col.setHex(tint[s.tone]));

  1098→// ============================================================ FAR TERRAIN
  1099→// Visual-only continuation of the world beyond the solver domain: the valley
  1100→// runs on, then a broad amphitheater of SMOOTH ROLLING HILLS closes the
  1101→// horizon — the user asked for NO jagged mountains, so every wall is a soft
  1102→// exponential dome modulated by gentle fbm (no ridged-noise crest lines).
  1103→function farBed(x: number, z: number): number {
  1104→  const cx = clamp(x, 0.2, LX - 0.2);
  1105→  const cz = clamp(z, -LZ / 2 + 0.2, LZ / 2 - 0.2);
  1106→  let h = bedAt(cx, cz);
  1107→  // distance beyond each solver-domain edge
  1108→  const dUp = Math.max(-x, 0);
  1109→  const dDown = Math.max(x - LX, 0);
  1110→  const dNorth = Math.max(-LZ / 2 - z, 0);
  1111→  const dSouth = Math.max(z - LZ / 2, 0);
  1112→  // rolling amphitheater — each wall eases in exponentially and caps far below
  1113→  // the old jagged 260 m peaks: the north wall (behind the dam) is tallest
  1114→  // (~85 m over the valley), the upstream headwall ~65 m, the downstream east
  1115→  // wall ~75 m, and the south rim stays low (~40 m) so the default aerial
  1116→  // camera still looks over it into the city bowl
  1117→  const rise = Math.max(
  1118→    dUp > 0 ? (1 - Math.exp(-dUp / 55)) * 60 : 0,
  1119→    dDown > 0 ? (dDown > 34 ? (1 - Math.exp(-(dDown - 34) / 60)) * 70 : 0) : 0,
  1120→    dNorth > 0 ? (1 - Math.exp(-dNorth / 48)) * 72 : 0,
  1121→    dSouth > 0 ? (1 - Math.exp(-dSouth / 70)) * 38 : 0,
  1122→  );
  1123→  if (rise > 0) {
  1124→    // smooth fbm only — frequencies kept well below the terrain mesh sampling
  1125→    // rate (anything finer aliases into triangular spikes); the squared massif
  1126→    // mask gives broad dome-like ridges instead of sharp crests
  1127→    const m1 = fbm(x * 0.012 + 7.7, z * 0.012 + 2.9, 3); // broad massifs
  1128→    const m2 = fbm(x * 0.027 + 1.2, z * 0.027 + 8.8, 3); // gentle undulation
  1129→    const hills = 0.52 + 0.58 * m1 * m1; // 0.52..1.10 — rounded domes
  1130→    h += rise * hills * (0.7 + 0.55 * m2);
  1131→    h += (fbm(x * 0.033 + 3.3, z * 0.033 + 6.1, 2) - 0.5) * Math.min(2.2 + rise * 0.05, 5.5);
  1132→  }
  1133→  return h;
  1134→}

  1136→function farPatch(
  1137→  x0: number, x1: number, z0: number, z1: number, sx: number, sz: number, sink = 0,
  1138→): THREE.Group {
  1139→  const geo = new THREE.PlaneGeometry(x1 - x0, z1 - z0, sx, sz);
  1140→  geo.rotateX(-Math.PI / 2);
  1141→  geo.translate((x0 + x1) / 2, 0, (z0 + z1) / 2);
  1142→  const pos = geo.attributes.position as THREE.BufferAttribute;
  1143→  const colors = new Float32Array(pos.count * 3);
  1144→  const col = new THREE.Color();
  1145→  const tc = { r: 0, g: 0, b: 0 };
  1146→  for (let i = 0; i < pos.count; i++) {
  1147→    const x = pos.getX(i);
  1148→    const z = pos.getZ(i);
  1149→    const y = farBed(x, z) - sink;
  1150→    pos.setY(i, y);
  1151→    const e = 3.2;
  1152→    const sl = Math.hypot(farBed(x + e, z) - farBed(x - e, z), farBed(x, z + e) - farBed(x, z - e)) / (2 * e);
  1153→    terrainColor(x, z, y, sl, tc);
  1154→    col.setRGB(tc.r, tc.g, tc.b, THREE.SRGBColorSpace);
  1155→    colors[i * 3] = col.r;
  1156→    colors[i * 3 + 1] = col.g;
  1157→    colors[i * 3 + 2] = col.b;
  1158→  }
  1159→
  1160→  geo.setAttribute('color', new THREE.BufferAttribute(colors, 3));
  1161→  geo.computeVertexNormals();
  1162→
  1163→  // Perimeter skirt: neighbouring patches share edge heights but have different
  1164→  // vertex spacings, so T-junction interpolation opens hairline cracks that
  1165→  // read as white specks from afar. Dropping a terrain-colored ribbon around
  1166→  // every edge hides any gap, whatever the viewing angle.
  1167→  const SKIRT = 7;
  1168→  const gridW = sx + 1;
  1169→  const gridH = sz + 1;
  1170→  const boundary: number[] = [];
  1171→  for (let ix = 0; ix < gridW; ix++) boundary.push(ix); // north row (z0)
  1172→  for (let iz = 1; iz < gridH; iz++) boundary.push(iz * gridW + gridW - 1); // east col
  1173→  for (let ix = gridW - 2; ix >= 0; ix--) boundary.push((gridH - 1) * gridW + ix); // south row
  1174→  for (let iz = gridH - 2; iz >= 1; iz--) boundary.push(iz * gridW); // west col
  1175→  const baseCount = pos.count;
  1176→  const skirtVerts = boundary.length * 2;
  1177→  const posArr = new Float32Array((baseCount + skirtVerts) * 3);
  1178→  posArr.set(pos.array as Float32Array);
  1179→  const colArr = new Float32Array((baseCount + skirtVerts) * 3);
  1180→  colArr.set(colors);
  1181→  for (let k = 0; k < boundary.length; k++) {
  1182→    const b = boundary[k];
  1183→    for (let c = 0; c < 3; c++) {
  1184→      posArr[(baseCount + k * 2) * 3 + c] = posArr[b * 3 + c];
  1185→      colArr[(baseCount + k * 2) * 3 + c] = colArr[b * 3 + c];
  1186→      posArr[(baseCount + k * 2 + 1) * 3 + c] = posArr[b * 3 + c];
  1187→      colArr[(baseCount + k * 2 + 1) * 3 + c] = colArr[b * 3 + c];
  1188→    }
  1189→    posArr[(baseCount + k * 2 + 1) * 3 + 1] -= SKIRT;
  1190→  }
  1191→  const idx = new Uint32Array(boundary.length * 6);
  1192→  for (let k = 0; k < boundary.length; k++) {
  1193→    const k2 = (k + 1) % boundary.length;
  1194→    const tA = baseCount + k * 2, bA = tA + 1, tB = baseCount + k2 * 2, bB = tB + 1;
  1195→    idx[k * 6] = tA; idx[k * 6 + 1] = tB; idx[k * 6 + 2] = bA;
  1196→    idx[k * 6 + 3] = bA; idx[k * 6 + 4] = tB; idx[k * 6 + 5] = bB;
  1197→  }
  1198→  const skirtGeo = new THREE.BufferGeometry();
  1199→  skirtGeo.setAttribute('position', new THREE.BufferAttribute(posArr, 3));
  1200→  skirtGeo.setAttribute('color', new THREE.BufferAttribute(colArr, 3));
  1201→  skirtGeo.setIndex(new THREE.BufferAttribute(idx, 1));
  1202→  skirtGeo.computeVertexNormals();
  1203→
  1204→  const mat = new THREE.MeshStandardMaterial({
  1205→    vertexColors: true, roughness: 0.97, metalness: 0, envMapIntensity: 0.18, side: THREE.DoubleSide,
  1206→  });
  1207→  const top = new THREE.Mesh(geo, mat);
  1208→  const skirt = new THREE.Mesh(skirtGeo, mat);
  1209→  skirt.receiveShadow = false;
  1210→  const group = new THREE.Group();
  1211→  group.add(top, skirt);
  1212→  return group;
  1213→}

  1215→export function buildFarTerrain(): { group: THREE.Group } {
  1216→  const group = new THREE.Group();
  1217→  const E = 260; // how far the ring extends sideways
  1218→  const Z0 = -LZ / 2;
  1219→  const Z1 = LZ / 2;
  1220→  // Seam apron: the domain terrain mesh samples bedAt at CELL CENTERS while a
  1221→  // far patch at the exact edge samples the analytic bed — the half-cell offset
  1222→  // used to open hairline cracks (white streaks) along every domain edge once
  1223→  // the jagged peaks that hid them were removed. Each patch now extends ~2.5 m
  1224→  // INTO the domain, sunk 0.8 m below the bed, so the domain surface always
  1225→  // renders on top of a tucked-under apron — no crack can ever open.
  1226→  const OV = 2.5;
  1227→  const SINK = 0.8;
  1228→  // upstream continuation — rises into the headwall range that hugs the lake
  1229→  // (also covers both far corners)
  1230→  group.add(farPatch(-260, OV, -E, E, 60, 100, SINK));
  1231→  // downstream continuation toward the horizon (also covers both far corners)
  1232→  group.add(farPatch(LX - OV, LX + 260, -E, E, 60, 100, SINK));
  1233→  // side ranges — inset in x so they meet, never overlap, the upstream and
  1234→  // downstream patches
  1235→  group.add(farPatch(OV, LX - OV, Z0 - E, Z0 + OV, 76, 64, SINK));
  1236→  group.add(farPatch(OV, LX - OV, Z1 - OV, Z1 + E, 76, 64, SINK));
  1237→  return { group };
  1238→}
