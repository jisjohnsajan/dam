    14→function hashRnd(i: number): number {
    15→  const h = Math.sin(i * 127.1 + 311.7) * 43758.5453;
    16→  return h - Math.floor(h);
    17→}

    19→function canvasTex(w: number, h: number, draw: (c: CanvasRenderingContext2D) => void): THREE.Texture {
    20→  const cv = document.createElement('canvas');
    21→  cv.width = w;
    22→  cv.height = h;
    23→  const ctx = cv.getContext('2d')!;
    24→  draw(ctx);
    25→  const tex = new THREE.CanvasTexture(cv);
    26→  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
    27→  tex.colorSpace = THREE.SRGBColorSpace;
    28→  tex.anisotropy = 4;
    29→  return tex;
    30→}

    52→function makePylon(h: number): THREE.Group {
    53→  const g = new THREE.Group();
    54→  const mat = matGalvanised();
    55→  const base = 1.5;
    56→  const top = 0.45;
    57→  // 4 splayed legs
    58→  for (const sx of [-1, 1]) {
    59→    for (const sz of [-1, 1]) {
    60→      const leg = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.09, h, 5), mat);
    61→      const lx = (sx * (base + (top - base) * 0.5)) / 2;
    62→      const lz = (sz * (base + (top - base) * 0.5)) / 2;
    63→      leg.position.set(lx, h / 2, lz);
    64→      leg.rotation.z = -sx * 0.115;
    65→      leg.rotation.x = sz * 0.115;
    66→      leg.castShadow = true;
    67→      g.add(leg);
    68→    }
    69→  }
    70→  // cross-braces on both faces (2 levels)
    71→  for (const lvl of [0.3, 0.62]) {
    72→    for (const sz of [-1, 1]) {
    73→      const brace = new THREE.Mesh(new THREE.BoxGeometry(base * (1 - lvl * 0.55), 0.06, 0.06), mat);
    74→      brace.position.set(0, h * lvl, (sz * base * (1 - lvl * 0.55)) / 2);
    75→      brace.rotation.z = sz * 0.5;
    76→      g.add(brace);
    77→    }
    78→  }
    79→  // cross-arms (two levels) + peak
    80→  const arm1 = new THREE.Mesh(new THREE.BoxGeometry(3.6, 0.09, 0.09), mat);
    81→  arm1.position.y = h * 0.8;
    82→  const arm2 = new THREE.Mesh(new THREE.BoxGeometry(2.7, 0.09, 0.09), mat);
    83→  arm2.position.y = h * 0.93;
    84→  const peak = new THREE.Mesh(new THREE.CylinderGeometry(0.03, 0.05, h * 0.1, 5), mat);
    85→  peak.position.y = h * 1.04;
    86→  arm1.castShadow = arm2.castShadow = true;
    87→  g.add(arm1, arm2, peak);
    88→  return g;
    89→}

    91→function makeCable(a: THREE.Vector3, b: THREE.Vector3, sag: number, mat: THREE.Material): THREE.Mesh {
    92→  const mid = a.clone().add(b).multiplyScalar(0.5);
    93→  mid.y -= sag;
    94→  const curve = new THREE.QuadraticBezierCurve3(a, mid, b);
    95→  const geo = new THREE.TubeGeometry(curve, 20, 0.035, 5, false);
    96→  const m = new THREE.Mesh(geo, mat);
    97→  m.castShadow = false;
    98→  return m;
    99→}

   395→function arcPts(r: number, half: number, n: number): [number, number][] {
   396→  const pts: [number, number][] = [];
   397→  for (let i = 0; i <= n; i++) {
   398→    const a = -half + (2 * half * i) / n;
   399→    pts.push([TOWN_C[0] + r * Math.cos(a), TOWN_C[1] + r * Math.sin(a)]);
   400→  }
   401→  return pts;
   402→}
   403→function rayPts(a: number, r0: number, r1: number, n: number): [number, number][] {
   404→  const pts: [number, number][] = [];

   445→function distToPaths(x: number, z: number): number {
   446→  let best = 1e9;
   447→  for (const s of CLEAR_SEGS) {
   448→    const dx = s.bx - s.ax, dz = s.bz - s.az;
   449→    const t = clamp(((x - s.ax) * dx + (z - s.az) * dz) / (dx * dx + dz * dz || 1), 0, 1);
   450→    const d = Math.hypot(x - (s.ax + dx * t), z - (s.az + dz * t));
   451→    if (d < best) best = d;
   452→  }
   453→  return best;
   454→}

   470→// deck ribbon following an arbitrary elevation profile; returns the sampled
   471→// centreline so callers can hang piers / dashes off it
   472→function profileRibbon(
   473→  group: THREE.Group, pts: [number, number][], w: number, mat: THREE.Material,
   474→  yAt: (x: number, z: number, t: number) => number,
   475→  castShadow = true,
   476→): { x: number; z: number; y: number }[] {
   477→  const n = Math.max(10, pts.length * 5);
   478→  const P: { x: number; z: number; y: number }[] = [];
   479→  for (let k = 0; k <= n; k++) {
   480→    const t = k / n;
   481→    const seg = Math.min(Math.floor(t * (pts.length - 1)), pts.length - 2);
   482→    const lt = t * (pts.length - 1) - seg;
   483→    const x = pts[seg][0] + (pts[seg + 1][0] - pts[seg][0]) * lt;
   484→    const z = pts[seg][1] + (pts[seg + 1][1] - pts[seg][1]) * lt;
   485→    P.push({ x, z, y: yAt(x, z, t) });
   486→  }
   487→  // smooth the elevation profile (endpoints pinned)
   488→  for (let p = 0; p < 4; p++)
   489→    for (let i = 1; i < P.length - 1; i++)
   490→      P[i].y = (P[i - 1].y + 2 * P[i].y + P[i + 1].y) / 4;
   491→  const pos: number[] = [];
   492→  const idx: number[] = [];
   493→  for (let i = 0; i < P.length; i++) {
   494→    const a = P[Math.max(i - 1, 0)];
   495→    const b = P[Math.min(i + 1, P.length - 1)];
   496→    const dx = b.x - a.x, dz = b.z - a.z;
   497→    const dl = Math.hypot(dx, dz) || 1;
   498→    const nx = (-dz / dl) * w * 0.5, nz = (dx / dl) * w * 0.5;
   499→    pos.push(P[i].x - nx, P[i].y, P[i].z - nz, P[i].x + nx, P[i].y, P[i].z + nz);
   500→    if (i > 0) {
   501→      const q = (i - 1) * 2;
   502→      idx.push(q, q + 2, q + 1, q + 1, q + 2, q + 3);
   503→    }
   504→  }
   505→  const geo = new THREE.BufferGeometry();
   506→  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(pos), 3));
   507→  geo.setIndex(idx);
   508→  geo.computeVertexNormals();
   509→  const mesh = new THREE.Mesh(geo, mat);
   510→  mesh.castShadow = castShadow;
   511→  mesh.receiveShadow = true;
   512→  group.add(mesh);
   513→  return P;
   514→}

   533→function buildExpressway(group: THREE.Group): void {
   534→  const deckMat = new THREE.MeshStandardMaterial({ color: 0x58585e, roughness: 0.88, metalness: 0.04, side: THREE.DoubleSide });
   535→  const beamMat = matConcreteDark();

   537→
   538→  const road = (pts: [number, number][], h0: number, h1: number, w: number): void => {
   539→    const P = profileRibbon(group, pts, w, deckMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t);
   540→    // edge beams slightly under the deck
   541→    profileRibbon(group, pts, w + 0.7, beamMat, (x, z, t) => bedAt(x, z) + h0 + (h1 - h0) * t - 0.16, false);
   542→    // pylons where the deck is well above ground (never in the river channel)
   543→    for (let i = 3; i < P.length - 2; i += 4) {

   545→      const hh = P[i].y - g;
